   1: import assert from "node:assert/strict";
   2: import fs from "node:fs/promises";
   3: import os from "node:os";
   4: import path from "node:path";
   5: import { after, before, test } from "node:test";
   6: 
   7: import express from "express";
   8: import supertest from "supertest";
   9: 
  10: import { uploadProjectMedia } from "../../src/middleware/project/projectMediaUploadMiddleware.js";
  11: import { validateProjectMedia } from "../../src/middleware/project/projectMediaValidationMiddleware.js";
  12: import { getProjectMediaDirectory } from "../../src/services/Storage/projectMediaStorageService.js";
  13: 
  14: const USER_ID = "validation-test-user";
  15: const PROJECT_ID = "validation-test-project";
  16: 
  17: let temporaryDirectory;
  18: let originalStoragePath;
  19: 
  20: before(async () => {
  21:   temporaryDirectory = await fs.mkdtemp(
  22:     path.join(os.tmpdir(), "project-media-validation-test-"),
  23:   );
  24: 
  25:   originalStoragePath = process.env.PROJECT_MEDIA_STORAGE_PATH;
  26:   process.env.PROJECT_MEDIA_STORAGE_PATH = temporaryDirectory;
  27: });
  28: 
  29: after(async () => {
  30:   if (originalStoragePath === undefined) {
  31:     delete process.env.PROJECT_MEDIA_STORAGE_PATH;
  32:   } else {
  33:     process.env.PROJECT_MEDIA_STORAGE_PATH = originalStoragePath;
  34:   }
  35: 
  36:   if (temporaryDirectory) {
  37:     await fs.rm(temporaryDirectory, {
  38:       recursive: true,
  39:       force: true,
  40:     });
  41:   }
  42: });
  43: 
  44: function createApp() {
  45:   const app = express();
  46: 
  47:   app.post(
  48:     "/projects/:projectId/media",
  49:     (request, response, next) => {
  50:       request.auth = { userId: USER_ID };
  51:       request.authorizedProject = { id: PROJECT_ID };
  52:       next();
  53:     },
  54:     uploadProjectMedia,
  55:     validateProjectMedia,
  56:     (request, response) => {
  57:       response.status(201).json({
  58:         verifiedMimeType: request.projectMediaFile?.verifiedMimeType,
  59:         verifiedSize: request.projectMediaFile?.verifiedSize,
  60:       });
  61:     },
  62:   );
  63: 
  64:   app.use((error, request, response, next) => {
  65:     response.status(error.status || 500).json({
  66:       code: error.code || "INTERNAL_ERROR",
  67:     });
  68:   });
  69: 
  70:   return app;
  71: }
  72: 
  73: async function getStoredFiles() {
  74:   const directory = getProjectMediaDirectory(USER_ID, PROJECT_ID);
  75: 
  76:   const relativePath = path.relative(temporaryDirectory, directory);
  77: 
  78:   assert.ok(
  79:     relativePath &&
  80:       relativePath !== ".." &&
  81:       !relativePath.startsWith(`..${path.sep}`) &&
  82:       !path.isAbsolute(relativePath),
  83:     "Test storage directory must be inside temporary storage.",
  84:   );
  85: 
  86:   try {
  87:     return await fs.readdir(directory);
  88:   } catch (error) {
  89:     if (error.code === "ENOENT") {
  90:       return [];
  91:     }
  92: 
  93:     throw error;
  94:   }
  95: }

  97: test("rejects a request without a media file", async () => {
  98:   const response = await supertest(createApp())
  99:     .post(`/projects/${PROJECT_ID}/media`)
 100:     .field("description", "No file supplied");
 101: 
 102:   assert.equal(response.status, 400);
 103:   assert.equal(response.body.code, "PROJECT_MEDIA_FILE_REQUIRED");
 104:   assert.deepEqual(await getStoredFiles(), []);

 141: test("accepts a valid PNG image and preserves its contents", async () => {
 142:   const pngBytes = Buffer.from(
 143:     "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+/lXcAAAAASUVORK5CYII=",
 144:     "base64",
 145:   );
 146: 
 147:   const response = await supertest(createApp())
 148:     .post(`/projects/${PROJECT_ID}/media`)
 149:     .attach("media", pngBytes, {
 150:       filename: "valid.png",
 151:       contentType: "image/png",
 152:     });
 153: 
 154:   assert.equal(response.status, 201);
 155:   assert.equal(response.body.verifiedMimeType, "image/png");
 156:   assert.equal(response.body.verifiedSize, pngBytes.length);
 157: 
 158:   const projectDirectory = getProjectMediaDirectory(
 159:     USER_ID,
 160:     PROJECT_ID,
 161:   );
 162: 
 163:   const storedFiles = await getStoredFiles();
 164: 
 165:   assert.equal(storedFiles.length, 1);
 166:   assert.match(storedFiles[0], /^[a-f0-9-]{36}\.png$/);
 167: 
 168:   const storedBytes = await fs.readFile(
 169:     path.join(projectDirectory, storedFiles[0]),
 170:   );
 171: 
 172:   assert.deepEqual(storedBytes, pngBytes);
 173: });

 175: test("rejects PNG images exceeding 20 MiB and removes the file", async () => {

 177:     "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+/lXcAAAAASUVORK5CYII=",
