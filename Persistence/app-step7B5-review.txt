import cors from "cors";
import express from "express";
import helmet from "helmet";

import userRoutes from "./routes/userRoutes.js";

import { checkDatabaseHealth } from "./config/database.js";

import { getEnvironment, isDatabaseConfigured } from "./config/environment.js";

import aiRoutes from "./routes/aiRoutes.js";
import authRoutes from "./routes/authRoutes.js";
import profileRoutes from "./routes/profileRoutes.js";

import {
  globalErrorHandler,
  notFoundHandler,
} from "./middleware/errorMiddleware.js";

import supportRoutes from "./routes/supportRoutes.js";

/*
 * =========================================
 * Environment and Application
 * =========================================
 */

const environment = getEnvironment();

const app = express();

/*
 * =========================================
 * Express Information Header
 * =========================================
 */

app.disable("x-powered-by");

/*
 * =========================================
 * Security Headers
 * =========================================
 */

app.use(helmet());

/*
 * =========================================
 * Allowed Frontend Origins
 * =========================================
 */

const allowedOrigins = environment.FRONTEND_URL.split(",")
  .map((origin) => origin.trim())
  .filter(Boolean);

/*
 * =========================================
 * CORS
 * =========================================
 */

app.use(
  cors({
    origin(origin, callback) {
      /*
       * Postman, curl and server-to-server
       * requests may not include Origin.
       */

      if (!origin) {
        return callback(null, true);
      }

      if (allowedOrigins.includes(origin)) {
        return callback(null, true);
      }

      const error = new Error(`Origin not allowed by CORS: ${origin}`);

      error.name = "CorsOriginError";

      error.code = "CORS_ORIGIN_DENIED";

      error.status = 403;

      error.publicMessage = "This website is not allowed to access the API.";

      return callback(error);
    },

    credentials: true,

    methods: ["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],

    allowedHeaders: ["Content-Type", "Authorization"],
  }),
);

/*
 * =========================================
 * Public Profile Pictures
 * =========================================
 */

if (environment.PROFILE_PICTURE_STORAGE_PATH) {
  app.use(
    "/uploads/profile-pictures",
    (request, response, next) => {
      /*
       * Profile pictures are intentionally public assets.
       * Allow the portfolio frontend to display them even
       * though the frontend and API use different origins.
       */

      response.setHeader("Cross-Origin-Resource-Policy", "cross-origin");

      next();
    },
    express.static(environment.PROFILE_PICTURE_STORAGE_PATH, {
      fallthrough: true,
      index: false,
      dotfiles: "deny",
    }),
  );
}

/*
 * =========================================
 * Request Body
 * =========================================
 */

app.use(
  express.json({
    limit: "1mb",
  }),
);

/*
 * =========================================
 * API Health Check
 * =========================================
 */

app.get("/api/health", (req, res) => {
  return res.status(200).json({
    status: "ok",

    service: "portfolio-platform-api",

    environment: environment.NODE_ENV,

    databaseConfigured: isDatabaseConfigured(environment),

    timestamp: new Date().toISOString(),
  });
});

/*
 * =========================================
 * Database Health Check
 * =========================================
 */

app.get("/api/health/database", async (req, res) => {
  const health = await checkDatabaseHealth();

  const isHealthy = health.status === "ok";

  let message = "Database connection is unavailable.";

  if (isHealthy) {
    message = "Database connection is healthy.";
  } else if (health.status === "not-configured") {
    message = "Database service is not configured.";
  }

  return res.status(isHealthy ? 200 : 503).json({
    status: health.status,

    service: "portfolio-platform-database",

    configured: health.configured,

    responseTimeMs: health.responseTimeMs,

    message,

    timestamp: new Date().toISOString(),
  });
});

/*
 * =========================================
 * Application Routes
 * =========================================
 */
app.use("/api/auth", authRoutes);

app.use("/api/ai", aiRoutes);

app.use("/api/users", userRoutes);

app.use("/api/support", supportRoutes);

app.use("/api/profile", profileRoutes);
/*
 * The Profile and Skill routes remain disabled
 * until their repositories and authentication
 * middleware are implemented.
 *
 * Later:
 *
 * app.use("/api/profile", profileRoutes);
 * app.use("/api/skills", skillRoutes);
 */

/*
 * =========================================
 * Route Not Found
 * =========================================
 */

app.use(notFoundHandler);

/*
 * =========================================
 * Global Error Handler
 * =========================================
 */

app.use(globalErrorHandler);

export default app;
