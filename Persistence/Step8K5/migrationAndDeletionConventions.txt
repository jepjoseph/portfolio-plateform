const currentDirectory = path.dirname(fileURLToPath(import.meta.url));

const migrationsDirectory = path.join(currentDirectory, "migrations");

/*
 * =========================================
 * Known Migration Checksum Exception
 * =========================================
 *
 * Migration 005 was applied to Azure SQL before its
 * current file version was committed.
 *
 * The expected schema has been inspected in Azure.
 * This exception is restricted to the two known hashes.
 *
 * Remove only after the migration history has been
 * properly reconciled.
 */

const knownChecksumExceptions = new Map([
  [
    "005_profile_details.sql",
    {
      recordedChecksum:
        "f436e5fcefd0bcbd0203f816bbaa30595b1fa59811ab58d84cf781b2687ec049",

      currentChecksum:
        "3817285481562c3ecb195e3f31736429c82f0879c48dac38244b77416aee365b",
    },
  ],
]);

const migrationFilePattern = /^\d{3,}_[a-z0-9_]+\.sql$/;

/*
 * =========================================
 * Migration Checksum
 * =========================================
 */

function createChecksum(contents) {
  return createHash("sha256").update(contents, "utf8").digest("hex");
}

/*
 * =========================================
 * Load Migration Files
 * =========================================
 */

async function loadMigrationFiles() {
  const entries = await readdir(migrationsDirectory, {
    withFileTypes: true,
  });

  const fileNames = entries
    .filter((entry) => entry.isFile() && migrationFilePattern.test(entry.name))
    .map((entry) => entry.name)
    .sort((first, second) => first.localeCompare(second, "en"));

  return Promise.all(
    fileNames.map(async (fileName) => {
      const filePath = path.join(migrationsDirectory, fileName);

      const contents = await readFile(filePath, "utf8");

      if (!contents.trim()) {
        throw new Error(`Migration file is empty: ${fileName}`);
      }

      /*
       * GO is understood by SSMS, but it is not
       * a SQL Server statement and cannot be sent
       * through the Node MSSQL driver.
       */

      if (/^\s*GO\s*;?\s*$/im.test(contents)) {
        throw new Error(
          `Migration ${fileName} contains GO. Remove GO because it is a client-side batch separator.`,
        );
      }

      return {
        id: fileName,

        contents,

        checksum: createChecksum(contents),
      };
    }),
  );
}

/*
 * =========================================
 * Migration Tracking Table
 * =========================================
 */

async function ensureMigrationTable(pool) {
  await pool.request().batch(`
    IF OBJECT_ID(N'dbo.schema_migrations', N'U') IS NULL
    BEGIN
      CREATE TABLE dbo.schema_migrations
      (
        migration_id NVARCHAR(255) NOT NULL,

        checksum CHAR(64) NOT NULL,

        applied_at DATETIME2(3) NOT NULL
          CONSTRAINT DF_schema_migrations_applied_at
          DEFAULT SYSUTCDATETIME(),

        CONSTRAINT PK_schema_migrations
          PRIMARY KEY CLUSTERED (migration_id)
      );
    END;
  `);
}

/*
 * =========================================
 * Migration Lock
 * =========================================
 */

async function acquireMigrationLock(request) {
  const result = await request.query(`
    DECLARE @lock_result INT;

    EXEC @lock_result = sys.sp_getapplock
      @Resource = N'portfolio-platform-schema-migrations',
      @LockMode = N'Exclusive',
      @LockOwner = N'Transaction',
      @LockTimeout = 60000;

    SELECT @lock_result AS lock_result;
  `);

  const lockResult = Number(result.recordset[0]?.lock_result);

  if (lockResult < 0) {
    throw new Error(
      `Unable to acquire the database migration lock. Result: ${lockResult}`,
    );
  }
}

/*
 * =========================================
 * Applied Migration Lookup
 * =========================================
 */

async function getAppliedMigrations(request) {
  const result = await request.query(`
    SELECT
      migration_id,
      checksum,
      applied_at
    FROM dbo.schema_migrations
    ORDER BY migration_id;
  `);

  return new Map(
    result.recordset.map((migration) => [migration.migration_id, migration]),
  );
}

/*
 * =========================================
 * Run Migrations
 * =========================================
 */

async function runMigrations() {
  const environment = getEnvironment();

  if (!isDatabaseConfigured(environment)) {
    throw new Error("Database configuration is required to run migrations.");
  }

  const migrations = await loadMigrationFiles();

  if (migrations.length === 0) {
    console.log("No database migration files were found.");

    return;
  }

  const pool = await getDatabasePool();

  await ensureMigrationTable(pool);

  const transaction = new sql.Transaction(pool);

  await transaction.begin(sql.ISOLATION_LEVEL.SERIALIZABLE);

  try {
    const request = new sql.Request(transaction);

    await acquireMigrationLock(request);

    const appliedMigrations = await getAppliedMigrations(request);

    let appliedCount = 0;

    for (const migration of migrations) {
      const existingMigration = appliedMigrations.get(migration.id);

      if (existingMigration) {
        if (existingMigration.checksum !== migration.checksum) {
          const exception = knownChecksumExceptions.get(migration.id);

          const isKnownMismatch =
            exception !== undefined &&
            existingMigration.checksum === exception.recordedChecksum &&
            migration.checksum === exception.currentChecksum;

          if (!isKnownMismatch) {
            throw new Error(
              `Migration ${migration.id} was modified after it was applied. Create a new migration instead.`,
            );
          }

          console.warn(
            `WARNING: Accepted documented checksum mismatch for ${migration.id}. ` +
              "The migration was previously applied; its SQL will not be executed again.",
          );
        }

        console.log(`Already applied: ${migration.id}`);

        continue;
      }

      console.log(`Applying migration: ${migration.id}`);

      await new sql.Request(transaction).batch(migration.contents);

      await new sql.Request(transaction)
        .input("migrationId", sql.NVarChar(255), migration.id)
        .input("checksum", sql.Char(64), migration.checksum).query(`
          INSERT INTO dbo.schema_migrations
          (
            migration_id,
            checksum
          )
          VALUES
          (
            @migrationId,
            @checksum
          );
        `);

      appliedCount += 1;

      console.log(`Applied migration: ${migration.id}`);
    }

    await transaction.commit();

    console.log(
      appliedCount === 0
        ? "Database schema is already up to date."
        : `Database migration completed. Applied ${appliedCount} migration(s).`,
    );
  } catch (error) {
    try {
      await transaction.rollback();
    } catch (rollbackError) {
      console.error("Database migration rollback failed:", rollbackError);
    }

    throw error;
  }
}

/*
 * =========================================
 * Execute Migration Process
 * =========================================
 */

runMigrations()
  .catch((error) => {
    console.error("Database migration failed:", error);

    process.exitCode = 1;
  })
  .finally(async () => {
    try {
      await closeDatabase();
    } catch (error) {
      console.error("Unable to close the database connection:", error);

      process.exitCode = 1;
    }
  });

import { getDatabasePool, sql } from "../config/database.js";

const PROJECT_COLUMNS = `
  id,
  user_id,
  title,
  category,
  lifecycle_status,
  record_status,
  is_featured,
  project_data,
  created_at,
  updated_at,
  row_version
`;

function mapProjectRow(row) {
  if (!row) {
    return null;
  }

  const projectData =
    typeof row.project_data === "string"
      ? JSON.parse(row.project_data)
      : row.project_data;

  return {
    ...projectData,
    id: row.id,
    title: row.title,
    category: row.category,
    lifecycleStatus: row.lifecycle_status,
    recordStatus: row.record_status,
    presentation: {
      ...(projectData.presentation || {}),
      isFeatured: Boolean(row.is_featured),
    },
    createdAt: row.created_at,
    updatedAt: row.updated_at,
    rowVersion: Buffer.from(row.row_version).toString("base64"),
  };
}

function bindProjectParameters(request, userId, project) {
  return request
    .input("id", sql.NVarChar(200), project.id)
    .input("userId", sql.UniqueIdentifier, userId)
    .input("title", sql.NVarChar(500), project.title)
    .input("category", sql.NVarChar(100), project.category)
    .input("lifecycleStatus", sql.NVarChar(50), project.lifecycleStatus)
    .input("recordStatus", sql.NVarChar(20), project.recordStatus)
    .input("isFeatured", sql.Bit, Boolean(project.presentation?.isFeatured))
    .input("projectData", sql.NVarChar(sql.MAX), JSON.stringify(project));
}

export async function listProjectsByUser(userId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
      ORDER BY updated_at DESC, id ASC;
    `);

  return result.recordset.map(mapProjectRow);
}

export async function findProjectById(userId, projectId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
        AND id = @id;
    `);

  return mapProjectRow(result.recordset[0]);
}

export async function createProject(userId, project) {
  const pool = await getDatabasePool();

  await bindProjectParameters(pool.request(), userId, project).query(`
    INSERT INTO dbo.projects (
      id,
      user_id,
      title,
      category,
      lifecycle_status,
      record_status,
      is_featured,
      project_data
    )
    VALUES (
      @id,
      @userId,
      @title,
      @category,
      @lifecycleStatus,
      @recordStatus,
      @isFeatured,
      @projectData
    );
  `);

  return findProjectById(userId, project.id);
}

export async function updateProject(userId, project, expectedRowVersion) {
  const pool = await getDatabasePool();

  const request = bindProjectParameters(pool.request(), userId, project).input(
    "expectedRowVersion",
    sql.Binary(8),
    Buffer.from(expectedRowVersion, "base64"),
  );

  const result = await request.query(`
    UPDATE dbo.projects
    SET
      title = @title,
      category = @category,
      lifecycle_status = @lifecycleStatus,
      record_status = @recordStatus,
      is_featured = @isFeatured,
      project_data = @projectData,
      updated_at = SYSUTCDATETIME()
    WHERE id = @id
      AND user_id = @userId
      AND row_version = @expectedRowVersion;
  `);

  if (result.rowsAffected[0] === 0) {
    return null;
  }

  return findProjectById(userId, project.id);
}

export async function deleteProject(userId, projectId, expectedRowVersion) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId)
    .input(
      "expectedRowVersion",
      sql.Binary(8),
      Buffer.from(expectedRowVersion, "base64"),
    ).query(`
      DELETE FROM dbo.projects
      WHERE id = @id
        AND user_id = @userId
        AND row_version = @expectedRowVersion;
    `);

  return result.rowsAffected[0] > 0;
}

import {
  listProjectsByUser,
  findProjectById,
  createProject,
  updateProject,
  deleteProject,
} from "../../repositories/projectRepository.js";

import {
  parseProjectInput,
  parseProjectRowVersion,
} from "../../schemas/projectSchemas.js";

const SERVER_OWNED_FIELDS = [
  "userId",
  "user_id",
  "createdAt",
  "updatedAt",
  "rowVersion",
];

function createProjectError({ message, code, status, publicMessage }) {
  const error = new Error(message);
  error.name = "ProjectServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;
  return error;
}

function assertUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw createProjectError({
      message: "An authenticated user ID is required.",
      code: "PROJECT_AUTHENTICATION_REQUIRED",
      status: 401,
      publicMessage: "You must sign in to access your Projects.",
    });
  }

  return userId.trim();
}

function assertProjectId(projectId) {
  if (
    typeof projectId !== "string" ||
    !projectId.trim() ||
    projectId.length > 200
  ) {
    throw createProjectError({
      message: "A valid Project ID is required.",
      code: "PROJECT_ID_INVALID",
      status: 400,
      publicMessage: "The Project identifier is invalid.",
    });
  }

  return projectId.trim();
}

function projectNotFound() {
  return createProjectError({
    message: "Project not found for authenticated user.",
    code: "PROJECT_NOT_FOUND",
    status: 404,
    publicMessage: "The requested Project could not be found.",
  });
}

function projectVersionConflict() {
  return createProjectError({
    message: "Project version conflict.",
    code: "PROJECT_VERSION_CONFLICT",
    status: 409,
    publicMessage:
      "This Project was changed elsewhere. Reload it before saving.",
  });
}

function projectIdMismatch() {
  return createProjectError({
    message: "The Project body ID does not match the requested Project ID.",
    code: "PROJECT_ID_MISMATCH",
    status: 400,
    publicMessage: "The Project identifiers do not match.",
  });
}

function withoutServerFields(project) {
  const result = { ...project };

  for (const field of SERVER_OWNED_FIELDS) {
    delete result[field];
  }

  return result;
}

export async function getAuthenticatedProjects({ userId }) {
  return listProjectsByUser(assertUserId(userId));
}

export async function getAuthenticatedProject({ userId, projectId }) {
  const project = await findProjectById(
    assertUserId(userId),
    assertProjectId(projectId),
  );

  if (!project) {
    throw projectNotFound();
  }

  return project;
}

export async function createAuthenticatedProject({ userId, project }) {
  const authenticatedUserId = assertUserId(userId);

  // Reject invalid data before any database operation.
  // Client-supplied server-owned fields remain forbidden.
  const validatedProject = parseProjectInput(project);

  try {
    return await createProject(authenticatedUserId, validatedProject);
  } catch (error) {
    if (error.number === 2627 || error.number === 2601) {
      throw createProjectError({
        message: "Project ID already exists.",
        code: "PROJECT_ALREADY_EXISTS",
        status: 409,
        publicMessage: "A Project with this identifier already exists.",
      });
    }

    throw error;
  }
}

export async function updateAuthenticatedProject({
  userId,
  projectId,
  project,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);

  // Validate the supplied version and full Project body.
  const expectedRowVersion = parseProjectRowVersion(rowVersion);
  const validatedProject = parseProjectInput(project);

  if (validatedProject.id !== normalizedProjectId) {
    throw projectIdMismatch();
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function setAuthenticatedProjectArchiveStatus({
  userId,
  projectId,
  rowVersion,
  archived,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  if (typeof archived !== "boolean") {
    throw createProjectError({
      message: "Archive status must be a boolean.",
      code: "PROJECT_ARCHIVE_STATUS_INVALID",
      status: 400,
      publicMessage: "A valid archive status is required.",
    });
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  // Database-managed metadata is not editable Project content.
  const editableProject = withoutServerFields(existingProject);

  const validatedProject = parseProjectInput({
    ...editableProject,
    recordStatus: archived ? "archived" : "active",
  });

  // SQL Server must match the caller's row version.
  // Do not substitute the version from the latest database read.
  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function deleteAuthenticatedProject({
  userId,
  projectId,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const deleted = await deleteProject(
    authenticatedUserId,
    normalizedProjectId,
    expectedRowVersion,
  );

  if (!deleted) {
    throw projectVersionConflict();
  }

  return { deleted: true };
}
