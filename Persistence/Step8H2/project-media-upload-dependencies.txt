import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import { getEnvironment } from "../../config/environment.js";

function requireIdentifier(value, label) {
  const normalized = String(value ?? "").trim();

  if (!normalized) {
    throw new Error(`${label} is required for Project media storage.`);
  }

  return normalized;
}

function hashIdentifier(value) {
  return crypto.createHash("sha256").update(value).digest("hex");
}

function validateFileName(fileName) {
  const normalized = String(fileName ?? "").trim();

  if (
    !normalized ||
    normalized === "." ||
    normalized === ".." ||
    normalized.includes("/") ||
    normalized.includes("\\") ||
    path.basename(normalized) !== normalized ||
    !/^[a-f0-9-]{36}\.(?:jpg|jpeg|png|webp|avif|gif|mp4|webm)$/i.test(
      normalized,
    )
  ) {
    throw new Error("Invalid Project media filename.");
  }

  return normalized;
}

export function getProjectMediaStorageRoot() {
  const environment = getEnvironment();

  const configuredPath = String(
    environment.PROJECT_MEDIA_STORAGE_PATH || "",
  ).trim();

  if (!configuredPath) {
    throw new Error("PROJECT_MEDIA_STORAGE_PATH is not configured.");
  }

  return path.resolve(configuredPath);
}

export function getProjectMediaUserDirectory(userId) {
  return hashIdentifier(requireIdentifier(userId, "User ID"));
}

export function getProjectMediaProjectDirectory(projectId) {
  return hashIdentifier(requireIdentifier(projectId, "Project ID"));
}

export function getProjectMediaDirectory(userId, projectId) {
  return path.join(
    getProjectMediaStorageRoot(),
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
  );
}

export function createProjectMediaStorageKey(userId, projectId, fileName) {
  const safeFileName = validateFileName(fileName);

  return [
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
    safeFileName,
  ].join("/");
}

export function resolveOwnedProjectMediaPath(userId, projectId, storageKey) {
  const normalizedKey = String(storageKey ?? "").trim();

  const expectedUserDirectory = getProjectMediaUserDirectory(userId);
  const expectedProjectDirectory = getProjectMediaProjectDirectory(projectId);

  const parts = normalizedKey.split("/");

  if (
    parts.length !== 3 ||
    parts[0] !== expectedUserDirectory ||
    parts[1] !== expectedProjectDirectory
  ) {
    throw new Error(
      "Project media storage key does not belong to this user and Project.",
    );
  }

  const safeFileName = validateFileName(parts[2]);

  const mediaDirectory = path.resolve(
    getProjectMediaDirectory(userId, projectId),
  );

  const filePath = path.resolve(mediaDirectory, safeFileName);

  if (path.dirname(filePath) !== mediaDirectory) {
    throw new Error("Project media path is outside the expected directory.");
  }

  return filePath;
}

async function assertSafeProjectMediaDirectory(userId, projectId) {
  const root = getProjectMediaStorageRoot();
  const userDirectory = path.join(root, getProjectMediaUserDirectory(userId));
  const projectDirectory = path.join(
    userDirectory,
    getProjectMediaProjectDirectory(projectId),
  );

  const directories = [root, userDirectory, projectDirectory];

  for (const directory of directories) {
    const stats = await fs.lstat(directory);

    if (stats.isSymbolicLink() || !stats.isDirectory()) {
      throw new Error(
        "Project media storage directory is not a regular directory.",
      );
    }
  }

  const canonicalRoot = await fs.realpath(root);
  const canonicalProjectDirectory = await fs.realpath(projectDirectory);

  const relative = path.relative(
    canonicalRoot,
    canonicalProjectDirectory,
  );

  if (
    relative === "" ||
    relative === ".." ||
    relative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(relative)
  ) {
    throw new Error("Project media directory is outside the storage root.");
  }
}

export async function deleteOwnedProjectMedia(userId, projectId, storageKey) {
  const filePath = resolveOwnedProjectMediaPath(userId, projectId, storageKey);

  try {
    await assertSafeProjectMediaDirectory(userId, projectId);

    const stats = await fs.lstat(filePath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw new Error("Project media target is not a regular file.");
    }

    await fs.unlink(filePath);

    return {
      deleted: true,
      storageKey,
    };
  } catch (error) {
    if (error?.code === "ENOENT") {
      return {
        deleted: false,
        storageKey,
      };
    }

    throw error;
  }
}

import crypto from "node:crypto";
import fs from "node:fs";

import multer from "multer";

import { getProfilePictureUserStoragePath } from "../../services/Storage/profilePictureStorageService.js";

/*
 * =========================================
 * Configuration
 * =========================================
 */

const MAX_PROFILE_PICTURE_SIZE = 20 * 1024 * 1024;

const ALLOWED_IMAGE_TYPES = new Map([
  ["image/jpeg", ".jpg"],
  ["image/jpg", ".jpg"],
  ["image/png", ".png"],
  ["image/webp", ".webp"],
  ["image/avif", ".avif"],
  ["image/gif", ".gif"],
]);

/*
 * =========================================
 * Upload Error
 * =========================================
 */

function createUploadError({
  message,
  code = "PROFILE_PICTURE_UPLOAD_FAILED",
  status = 400,
  publicMessage = "The profile picture could not be uploaded.",
}) {
  const error = new Error(message);

  error.name = "ProfilePictureUploadError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;

  return error;
}

/*
 * =========================================
 * Storage Configuration
 * =========================================
 */

const storage = multer.diskStorage({
  destination(request, file, callback) {
    try {
      const userId = String(request.auth?.userId || "").trim();

      if (!userId) {
        return callback(
          createUploadError({
            message:
              "Authenticated user ID is unavailable during profile-picture upload.",
            code: "PROFILE_PICTURE_UPLOAD_USER_INVALID",
            status: 401,
            publicMessage:
              "Your session could not be verified for this upload.",
          }),
        );
      }

      const userStoragePath = getProfilePictureUserStoragePath(userId);

      fs.mkdirSync(userStoragePath, {
        recursive: true,
      });

      return callback(null, userStoragePath);
    } catch (error) {
      return callback(error);
    }
  },

  filename(request, file, callback) {
    const extension = ALLOWED_IMAGE_TYPES.get(file.mimetype);

    if (!extension) {
      return callback(
        createUploadError({
          message: `Unsupported profile-picture MIME type: ${file.mimetype}`,
          code: "PROFILE_PICTURE_TYPE_NOT_ALLOWED",
          status: 400,
          publicMessage:
            "The profile picture must be a JPEG, PNG, WebP, or AVIF image.",
        }),
      );
    }

    const fileName = `${crypto.randomUUID()}${extension}`;

    return callback(null, fileName);
  },
});

/*
 * =========================================
 * File Type Filter
 * =========================================
 */

function profilePictureFileFilter(request, file, callback) {
  if (!ALLOWED_IMAGE_TYPES.has(file.mimetype)) {
    return callback(
      createUploadError({
        message: `Unsupported profile-picture MIME type: ${file.mimetype}`,
        code: "PROFILE_PICTURE_TYPE_NOT_ALLOWED",
        status: 400,
        publicMessage:
          "The profile picture must be a JPEG, PNG, WebP, or AVIF image.",
      }),
    );
  }

  return callback(null, true);
}

/*
 * =========================================
 * Multer Upload
 * =========================================
 */

const profilePictureUpload = multer({
  storage,

  limits: {
    fileSize: MAX_PROFILE_PICTURE_SIZE,
    files: 1,
  },

  fileFilter: profilePictureFileFilter,
}).single("picture");

/*
 * =========================================
 * Upload Middleware
 * =========================================
 */

export function uploadProfilePicture(request, response, next) {
  profilePictureUpload(request, response, (error) => {
    if (!error) {
      return next();
    }

    if (error instanceof multer.MulterError) {
      if (error.code === "LIMIT_FILE_SIZE") {
        return next(
          createUploadError({
            message: "Profile picture exceeds the 20 MiB upload limit.",
            code: "PROFILE_PICTURE_TOO_LARGE",
            status: 413,
            publicMessage: "The profile picture must be 20 MB or smaller.",
          }),
        );
      }

      if (error.code === "LIMIT_FILE_COUNT") {
        return next(
          createUploadError({
            message: "More than one profile-picture file was uploaded.",
            code: "PROFILE_PICTURE_TOO_MANY_FILES",
            status: 400,
            publicMessage: "Please upload only one profile picture.",
          }),
        );
      }

      if (error.code === "LIMIT_UNEXPECTED_FILE") {
        return next(
          createUploadError({
            message: "An unexpected profile-picture upload field was received.",
            code: "PROFILE_PICTURE_UNEXPECTED_FILE",
            status: 400,
            publicMessage:
              'The profile picture must be uploaded using the "picture" field.',
          }),
        );
      }

      return next(
        createUploadError({
          message: `Multer profile-picture upload error: ${error.code}`,
          code: "PROFILE_PICTURE_UPLOAD_INVALID",
          status: 400,
          publicMessage: "The profile picture upload could not be processed.",
        }),
      );
    }

    return next(error);
  });
}
