 * Boolean Environment Values
 * =========================================

const TRUE_VALUES = new Set(["1", "true", "yes", "on"]);

const FALSE_VALUES = new Set(["0", "false", "no", "off"]);

/*
 * =========================================
 * Optional Text Normalization
 * =========================================
 */

function normalizeOptionalText(value) {
  if (typeof value !== "string") {
    return value;
  }

  const normalizedValue = value.trim();

  return normalizedValue || undefined;
}

/*
 * =========================================
 * Boolean Normalization
 * =========================================
 */

function normalizeBooleanValue(value) {
  if (typeof value === "boolean") {
    return value;
  }

  const normalizedValue = String(value ?? "")
    .trim()
    .toLowerCase();

  if (TRUE_VALUES.has(normalizedValue)) {
    return true;
  }

  if (FALSE_VALUES.has(normalizedValue)) {
    return false;
  }

  return value;
}

let cachedEnvironment;

/*
 * =========================================
 * Read and Validate Environment
 * =========================================
 */

export function getEnvironment() {
  if (cachedEnvironment) {
    return cachedEnvironment;
  }

  const result = environmentSchema.safeParse(process.env);

  if (!result.success) {
    const details = result.error.issues
      .map(
        (issue) => `${issue.path.join(".") || "environment"}: ${issue.message}`,
      )
      .join("\n");

    throw new Error(`Invalid environment configuration:\n${details}`);
  }

  cachedEnvironment = Object.freeze(result.data);

  return cachedEnvironment;
}

/*
 * =========================================
 * Database Configuration Status
 * =========================================
 */

export function isDatabaseConfigured(environment = getEnvironment()) {
  return Boolean(
    environment.DB_HOST &&
    environment.DB_NAME &&
    environment.DB_USER &&
    environment.DB_PASSWORD,
  );
}

/*
 * =========================================
 * Authentication Secret Access
 * =========================================
 *
 * This function prevents authentication
 * operations from running without all three
 * security secrets.
 */

export function getAuthenticationSecrets(environment = getEnvironment()) {
  const secrets = {
    otp: environment.AUTH_OTP_HMAC_SECRET,

    token: environment.AUTH_TOKEN_HMAC_SECRET,

    ip: environment.AUTH_IP_HMAC_SECRET,
  };

  const missingSecrets = Object.entries(secrets)
    .filter(([, value]) => !value)
    .map(([name]) => name);

  if (missingSecrets.length > 0) {
    const error = new Error(
      `Authentication security secrets are missing: ${missingSecrets.join(", ")}.`,
    );

    error.name = "AuthenticationConfigurationError";

    error.code = "AUTHENTICATION_NOT_CONFIGURED";

    error.status = 503;

    error.publicMessage = "Authentication service is temporarily unavailable.";

    throw error;
  }

  return Object.freeze(secrets);
}

import crypto from "node:crypto";
import fs from "node:fs";

import multer from "multer";

import { getProfilePictureUserStoragePath } from "../../services/Storage/profilePictureStorageService.js";

/*
 * =========================================
 * Configuration
 * =========================================
 */

const MAX_PROFILE_PICTURE_SIZE = 20 * 1024 * 1024;

const ALLOWED_IMAGE_TYPES = new Map([
  ["image/jpeg", ".jpg"],
  ["image/jpg", ".jpg"],
  ["image/png", ".png"],
  ["image/webp", ".webp"],
  ["image/avif", ".avif"],
  ["image/gif", ".gif"],
]);

/*
 * =========================================
 * Upload Error
 * =========================================
 */

function createUploadError({
  message,
  code = "PROFILE_PICTURE_UPLOAD_FAILED",
  status = 400,
  publicMessage = "The profile picture could not be uploaded.",
}) {
  const error = new Error(message);

  error.name = "ProfilePictureUploadError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;

  return error;
}

/*
 * =========================================
 * Storage Configuration
 * =========================================
 */

const storage = multer.diskStorage({
  destination(request, file, callback) {
    try {
      const userId = String(request.auth?.userId || "").trim();

      if (!userId) {
        return callback(
          createUploadError({
            message:
              "Authenticated user ID is unavailable during profile-picture upload.",
            code: "PROFILE_PICTURE_UPLOAD_USER_INVALID",
            status: 401,
            publicMessage:
              "Your session could not be verified for this upload.",
          }),
        );
      }

      const userStoragePath = getProfilePictureUserStoragePath(userId);

      fs.mkdirSync(userStoragePath, {
        recursive: true,
      });

      return callback(null, userStoragePath);
    } catch (error) {
      return callback(error);
    }
  },

  filename(request, file, callback) {
    const extension = ALLOWED_IMAGE_TYPES.get(file.mimetype);

    if (!extension) {
      return callback(
        createUploadError({
          message: `Unsupported profile-picture MIME type: ${file.mimetype}`,
          code: "PROFILE_PICTURE_TYPE_NOT_ALLOWED",
          status: 400,
          publicMessage:
            "The profile picture must be a JPEG, PNG, WebP, or AVIF image.",
        }),
      );
    }

    const fileName = `${crypto.randomUUID()}${extension}`;

    return callback(null, fileName);
  },
});

/*
 * =========================================
 * File Type Filter
 * =========================================
 */

function profilePictureFileFilter(request, file, callback) {
  if (!ALLOWED_IMAGE_TYPES.has(file.mimetype)) {
    return callback(
      createUploadError({
        message: `Unsupported profile-picture MIME type: ${file.mimetype}`,
        code: "PROFILE_PICTURE_TYPE_NOT_ALLOWED",
        status: 400,
        publicMessage:
          "The profile picture must be a JPEG, PNG, WebP, or AVIF image.",
      }),
    );
  }

  return callback(null, true);
}

/*
 * =========================================
 * Multer Upload
 * =========================================
 */

const profilePictureUpload = multer({
  storage,

  limits: {
    fileSize: MAX_PROFILE_PICTURE_SIZE,
    files: 1,
  },

  fileFilter: profilePictureFileFilter,
}).single("picture");

/*
 * =========================================
 * Upload Middleware
 * =========================================
 */

export function uploadProfilePicture(request, response, next) {
  profilePictureUpload(request, response, (error) => {
    if (!error) {
      return next();
    }

    if (error instanceof multer.MulterError) {
      if (error.code === "LIMIT_FILE_SIZE") {
        return next(
          createUploadError({
            message: "Profile picture exceeds the 20 MiB upload limit.",
            code: "PROFILE_PICTURE_TOO_LARGE",
            status: 413,
            publicMessage: "The profile picture must be 20 MB or smaller.",
          }),
        );
      }

      if (error.code === "LIMIT_FILE_COUNT") {
        return next(
          createUploadError({
            message: "More than one profile-picture file was uploaded.",
            code: "PROFILE_PICTURE_TOO_MANY_FILES",
            status: 400,
            publicMessage: "Please upload only one profile picture.",
          }),
        );
      }

      if (error.code === "LIMIT_UNEXPECTED_FILE") {
        return next(
          createUploadError({
            message: "An unexpected profile-picture upload field was received.",
            code: "PROFILE_PICTURE_UNEXPECTED_FILE",
            status: 400,
            publicMessage:
              'The profile picture must be uploaded using the "picture" field.',
          }),
        );
      }

      return next(
        createUploadError({
          message: `Multer profile-picture upload error: ${error.code}`,
          code: "PROFILE_PICTURE_UPLOAD_INVALID",
          status: 400,
          publicMessage: "The profile picture upload could not be processed.",
        }),
      );
    }

    return next(error);
  });
}

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import { getEnvironment } from "../../config/environment.js";

const environment = getEnvironment();

/*
 * =========================================
 * User Storage Identity
 * =========================================
 */

export function getProfilePictureUserDirectory(userId) {
  const normalizedUserId = String(userId || "").trim();

  if (!normalizedUserId) {
    throw new Error(
      "An authenticated user ID is required for profile-picture storage.",
    );
  }

  return crypto.createHash("sha256").update(normalizedUserId).digest("hex");
}

/*
 * =========================================
 * Storage Root
 * =========================================
 */

export function getProfilePictureStorageRoot() {
  const storageRoot = String(
    environment.PROFILE_PICTURE_STORAGE_PATH || "",
  ).trim();

  if (!storageRoot) {
    throw new Error("PROFILE_PICTURE_STORAGE_PATH is not configured.");
  }

  return path.resolve(storageRoot);
}

/*
 * =========================================
 * User Storage Path
 * =========================================
 */

export function getProfilePictureUserStoragePath(userId) {
  return path.join(
    getProfilePictureStorageRoot(),
    getProfilePictureUserDirectory(userId),
  );
}

/*
 * =========================================
 * Storage Key
 * =========================================
 */

export function createProfilePictureStorageKey(userId, fileName) {
  const safeFileName = path.basename(String(fileName || "").trim());

  if (!safeFileName) {
    throw new Error("A profile-picture filename is required.");
  }

  return [getProfilePictureUserDirectory(userId), safeFileName].join("/");
}

export function resolveOwnedProfilePicturePath(userId, storageKey) {
  const normalizedStorageKey = String(storageKey || "")
    .trim()
    .replaceAll("\\", "/");

  if (!normalizedStorageKey) {
    throw new Error("A profile-picture storage key is required.");
  }

  const userDirectory = getProfilePictureUserDirectory(userId);

  const expectedPrefix = `${userDirectory}/`;

  if (!normalizedStorageKey.startsWith(expectedPrefix)) {
    throw new Error(
      "The profile-picture storage key does not belong to the authenticated user.",
    );
  }

  const relativeFileName = normalizedStorageKey.slice(expectedPrefix.length);

  /*
   * A profile picture must be one file directly inside
   * the authenticated user's storage directory.
   */

  if (
    !relativeFileName ||
    relativeFileName.includes("/") ||
    relativeFileName === "." ||
    relativeFileName === ".." ||
    path.basename(relativeFileName) !== relativeFileName
  ) {
    throw new Error("The profile-picture storage key is invalid.");
  }

  const userStoragePath = path.resolve(
    getProfilePictureUserStoragePath(userId),
  );

  const resolvedFilePath = path.resolve(userStoragePath, relativeFileName);

  /*
   * Defense in depth:
   * even after validating the storage key above, verify that
   * the resulting file remains directly inside this user's
   * storage directory.
   */

  if (path.dirname(resolvedFilePath) !== userStoragePath) {
    throw new Error(
      "The profile-picture path is outside the authenticated user's storage directory.",
    );
  }

  return resolvedFilePath;
}

export async function deleteOwnedProfilePicture(userId, storageKey) {
  const filePath = resolveOwnedProfilePicturePath(userId, storageKey);

  try {
    await fs.unlink(filePath);

    return {
      deleted: true,
      storageKey,
    };
  } catch (error) {
    /*
     * If the file is already gone, cleanup has still reached
     * the desired state. Treat deletion as idempotent.
     */

    if (error?.code === "ENOENT") {
      return {
        deleted: false,
        storageKey,
      };
    }

    throw error;
  }
}
