import {
  listProjectsByUser,
  findProjectById,
  createProject,
  updateProject,
  deleteProject,
} from "../../repositories/projectRepository.js";

import {
  parseProjectInput,
  parseProjectRowVersion,
} from "../../schemas/projectSchemas.js";

const SERVER_OWNED_FIELDS = [
  "userId",
  "user_id",
  "createdAt",
  "updatedAt",
  "rowVersion",
];

function createProjectError({ message, code, status, publicMessage }) {
  const error = new Error(message);
  error.name = "ProjectServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;
  return error;
}

function assertUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw createProjectError({
      message: "An authenticated user ID is required.",
      code: "PROJECT_AUTHENTICATION_REQUIRED",
      status: 401,
      publicMessage: "You must sign in to access your Projects.",
    });
  }

  return userId.trim();
}

function assertProjectId(projectId) {
  if (
    typeof projectId !== "string" ||
    !projectId.trim() ||
    projectId.length > 200
  ) {
    throw createProjectError({
      message: "A valid Project ID is required.",
      code: "PROJECT_ID_INVALID",
      status: 400,
      publicMessage: "The Project identifier is invalid.",
    });
  }

  return projectId.trim();
}

function projectNotFound() {
  return createProjectError({
    message: "Project not found for authenticated user.",
    code: "PROJECT_NOT_FOUND",
    status: 404,
    publicMessage: "The requested Project could not be found.",
  });
}

function projectVersionConflict() {
  return createProjectError({
    message: "Project version conflict.",
    code: "PROJECT_VERSION_CONFLICT",
    status: 409,
    publicMessage:
      "This Project was changed elsewhere. Reload it before saving.",
  });
}

function projectIdMismatch() {
  return createProjectError({
    message: "The Project body ID does not match the requested Project ID.",
    code: "PROJECT_ID_MISMATCH",
    status: 400,
    publicMessage: "The Project identifiers do not match.",
  });
}

function withoutServerFields(project) {
  const result = { ...project };

  for (const field of SERVER_OWNED_FIELDS) {
    delete result[field];
  }

  return result;
}

export async function getAuthenticatedProjects({ userId }) {
  return listProjectsByUser(assertUserId(userId));
}

export async function getAuthenticatedProject({ userId, projectId }) {
  const project = await findProjectById(
    assertUserId(userId),
    assertProjectId(projectId),
  );

  if (!project) {
    throw projectNotFound();
  }

  return project;
}

export async function createAuthenticatedProject({ userId, project }) {
  const authenticatedUserId = assertUserId(userId);

  // Reject invalid data before any database operation.
  // Client-supplied server-owned fields remain forbidden.
  const validatedProject = parseProjectInput(project);

  try {
    return await createProject(authenticatedUserId, validatedProject);
  } catch (error) {
    if (error.number === 2627 || error.number === 2601) {
      throw createProjectError({
        message: "Project ID already exists.",
        code: "PROJECT_ALREADY_EXISTS",
        status: 409,
        publicMessage: "A Project with this identifier already exists.",
      });
    }

    throw error;
  }
}

export async function updateAuthenticatedProject({
  userId,
  projectId,
  project,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);

  // Validate the supplied version and full Project body.
  const expectedRowVersion = parseProjectRowVersion(rowVersion);
  const validatedProject = parseProjectInput(project);

  if (validatedProject.id !== normalizedProjectId) {
    throw projectIdMismatch();
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function setAuthenticatedProjectArchiveStatus({
  userId,
  projectId,
  rowVersion,
  archived,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  if (typeof archived !== "boolean") {
    throw createProjectError({
      message: "Archive status must be a boolean.",
      code: "PROJECT_ARCHIVE_STATUS_INVALID",
      status: 400,
      publicMessage: "A valid archive status is required.",
    });
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  // Database-managed metadata is not editable Project content.
  const editableProject = withoutServerFields(existingProject);

  const validatedProject = parseProjectInput({
    ...editableProject,
    recordStatus: archived ? "archived" : "active",
  });

  // SQL Server must match the caller's row version.
  // Do not substitute the version from the latest database read.
  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function deleteAuthenticatedProject({
  userId,
  projectId,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const deleted = await deleteProject(
    authenticatedUserId,
    normalizedProjectId,
    expectedRowVersion,
  );

  if (!deleted) {
    throw projectVersionConflict();
  }

  return { deleted: true };
}

import { validateNestedProject } from "./projectNestedValidation.js";

const PROJECT_ID_PATTERN =
  /^project-(?:[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}|[0-9]{10,17}-[a-z0-9]{1,20})$/i;

const CATEGORIES = new Set([
  "software",
  "web-application",
  "mobile-application",
  "artificial-intelligence",
  "data",
  "cloud-infrastructure",
  "networking",
  "cybersecurity",
  "embedded-systems",
  "electronics",
  "automation",
  "internet-of-things",
  "research",
  "design",
  "business",
  "academic",
  "community",
  "other",
]);

const LIFECYCLE_STATUSES = new Set([
  "planned",
  "in-progress",
  "completed",
  "maintained",
  "paused",
  "cancelled",
]);

const RECORD_STATUSES = new Set(["active", "archived"]);

const SERVER_OWNED_FIELDS = [
  "userId",
  "user_id",
  "createdAt",
  "updatedAt",
  "rowVersion",
];

function createValidationError(errors) {
  const error = new Error("Project validation failed.");

  error.name = "ProjectValidationError";
  error.code = "PROJECT_VALIDATION_FAILED";
  error.status = 400;
  error.publicMessage = "Please correct the Project information.";
  error.errors = errors;

  error.fieldErrors = {};

  for (const item of errors) {
    if (!error.fieldErrors[item.field]) {
      error.fieldErrors[item.field] = item.message;
    }
  }

  return error;
}

function addError(errors, field, code, message) {
  errors.push({ field, code, message });
}

function isPlainObject(value) {
  return value !== null && typeof value === "object" && !Array.isArray(value);
}

function isValidProjectDate(value) {
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

export function parseProjectInput(value) {
  const errors = [];

  if (!isPlainObject(value)) {
    throw createValidationError([
      {
        field: "project",
        code: "PROJECT_OBJECT_REQUIRED",
        message: "Project data must be an object.",
      },
    ]);
  }

  for (const field of SERVER_OWNED_FIELDS) {
    if (Object.hasOwn(value, field)) {
      addError(
        errors,
        field,
        "PROJECT_SERVER_FIELD_FORBIDDEN",
        `${field} cannot be supplied by the client.`,
      );
    }
  }

  if (typeof value.id !== "string" || !PROJECT_ID_PATTERN.test(value.id)) {
    addError(
      errors,
      "id",
      "PROJECT_ID_INVALID",
      "A valid Project identifier is required.",
    );
  }

  if (
    typeof value.title !== "string" ||
    !value.title.trim() ||
    value.title.trim().length > 200
  ) {
    addError(
      errors,
      "title",
      "PROJECT_TITLE_INVALID",
      "Project title must contain 1 to 200 characters.",
    );
  }

  if (
    typeof value.role !== "string" ||
    !value.role.trim() ||
    value.role.trim().length > 160
  ) {
    addError(
      errors,
      "role",
      "PROJECT_ROLE_INVALID",
      "Project role must contain 1 to 160 characters.",
    );
  }

  if (!CATEGORIES.has(value.category)) {
    addError(
      errors,
      "category",
      "PROJECT_CATEGORY_INVALID",
      "Select a valid Project category.",
    );
  }

  if (!LIFECYCLE_STATUSES.has(value.lifecycleStatus)) {
    addError(
      errors,
      "lifecycleStatus",
      "PROJECT_LIFECYCLE_INVALID",
      "Select a valid lifecycle status.",
    );
  }

  if (!RECORD_STATUSES.has(value.recordStatus)) {
    addError(
      errors,
      "recordStatus",
      "PROJECT_RECORD_STATUS_INVALID",
      "Select a valid record status.",
    );
  }

  if (!isPlainObject(value.dates)) {
    addError(
      errors,
      "dates",
      "PROJECT_DATES_INVALID",
      "Project dates must be an object.",
    );
  } else {
    if (!isValidProjectDate(value.dates.startDate)) {
      addError(
        errors,
        "dates.startDate",
        "PROJECT_START_DATE_INVALID",
        "A valid Project start date is required.",
      );
    }

    if (value.dates.endDate && !isValidProjectDate(value.dates.endDate)) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_INVALID",
        "Enter a valid Project end date.",
      );
    }

    if (value.dates.isCurrent === true && value.dates.endDate) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_CURRENT_END_DATE_CONFLICT",
        "An ongoing Project cannot have an end date.",
      );
    }

    if (
      value.dates.isCurrent === true &&
      !["in-progress", "maintained"].includes(value.lifecycleStatus)
    ) {
      addError(
        errors,
        "lifecycleStatus",
        "PROJECT_CURRENT_LIFECYCLE_CONFLICT",
        "An ongoing Project must be In Progress or Maintained.",
      );
    }

    if (
      ["completed", "cancelled"].includes(value.lifecycleStatus) &&
      value.dates.isCurrent !== true &&
      !value.dates.endDate
    ) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_REQUIRED",
        "A completion or cancellation date is required.",
      );
    }

    if (
      isValidProjectDate(value.dates.startDate) &&
      isValidProjectDate(value.dates.endDate)
    ) {
      const start = new Date(
        value.dates.startDate.length === 7
          ? `${value.dates.startDate}-01`
          : value.dates.startDate,
      );

      const end = new Date(
        value.dates.endDate.length === 7
          ? `${value.dates.endDate}-01`
          : value.dates.endDate,
      );

      if (end < start) {
        addError(
          errors,
          "dates.endDate",
          "PROJECT_DATE_ORDER_INVALID",
          "End date cannot be earlier than start date.",
        );
      }
    }
  }

  if (
    !isPlainObject(value.presentation) ||
    typeof value.presentation.isFeatured !== "boolean"
  ) {
    addError(
      errors,
      "presentation.isFeatured",
      "PROJECT_FEATURED_INVALID",
      "Project featured status must be a boolean.",
    );
  }

  const urlFields = [
    ["organization.website", value.organization?.website],
    ...[
      "liveUrl",
      "repositoryUrl",
      "documentationUrl",
      "caseStudyUrl",
      "videoUrl",
    ].map((name) => [`links.${name}`, value.links?.[name]]),
  ];

  for (const [field, urlValue] of urlFields) {
    if (urlValue === undefined || urlValue === "") continue;

    let valid = false;

    if (typeof urlValue === "string") {
      try {
        const parsed = new URL(urlValue);
        valid = ["http:", "https:"].includes(parsed.protocol);
      } catch {
        valid = false;
      }
    }

    if (!valid) {
      addError(
        errors,
        field,
        "PROJECT_URL_INVALID",
        "Enter a valid HTTP or HTTPS URL.",
      );
    }
  }

  const collectionLimits = [
    ["problem.objectives", value.problem?.objectives, 25],
    ["problem.constraints", value.problem?.constraints, 25],
    ["solution.features", value.solution?.features, 40],
    ["solution.contributions", value.solution?.contributions, 40],
    ["results.metrics", value.results?.metrics, 25],
    ["skillRelationships", value.skillRelationships, 60],
    ["technologies", value.technologies, 60],
    ["awards", value.awards, 20],
    ["media", value.media, 30],
    ["supportingDocuments", value.supportingDocuments, 30],
    ...[
      "experienceRelationships",
      "educationRelationships",
      "trainingRelationships",
      "certificationRelationships",
    ].map((name) => [
      `relatedRecords.${name}`,
      value.relatedRecords?.[name],
      50,
    ]),
  ];

  for (const [field, items, maximum] of collectionLimits) {
    if (items === undefined) continue;

    if (!Array.isArray(items) || items.length > maximum) {
      addError(
        errors,
        field,
        "PROJECT_COLLECTION_INVALID",
        `This collection must contain at most ${maximum} items.`,
      );
    }
  }
  errors.push(...validateNestedProject(value));

  if (errors.length > 0) {
    throw createValidationError(errors);
  }

  return structuredClone(value);
}

export function parseProjectRowVersion(value) {
  if (typeof value !== "string" || !/^[A-Za-z0-9+/]{11}=$/.test(value)) {
    throw createValidationError([
      {
        field: "rowVersion",
        code: "PROJECT_VERSION_INVALID",
        message: "A valid Project version is required.",
      },
    ]);
  }

  return value;
}

const OPTIONS = {

  source: ["manual", "import", "integration", "ai-assisted"],
  proficiency: ["", "foundational", "intermediate", "advanced", "expert"],
  recognitionType: [

function object(value) {
  return value !== null &&
    typeof value === "object" &&
    !Array.isArray(value);
}

function fieldValue(source, path) {
  return path.split(".").reduce(
    (current, key) => object(current) ? current[key] : undefined,
    source,
  );
}

function error(errors, field, code, message) {
  errors.push({ field, code, message });
}

function checkText(errors, field, value, maximum) {
  if (value === undefined) return;

  if (typeof value !== "string" || value.trim().length > maximum) {
    error(
      errors, field, "PROJECT_FIELD_INVALID",
      `This field must be text with at most ${maximum} characters.`,
    );
  }
}

function checkOption(errors, field, value, options) {
  if (value === undefined) return;

  if (!options.includes(value)) {
    error(errors, field, "PROJECT_OPTION_INVALID", "Invalid option.");
  }
}

function checkUrl(errors, field, value, maximum = 2000) {
  if (value === undefined) return;

  checkText(errors, field, value, maximum);

  if (typeof value !== "string" || !value.trim()) return;

  try {
    const url = new URL(value);
    if (!["http:", "https:"].includes(url.protocol)) throw new Error();
  } catch {
    error(errors, field, "PROJECT_URL_INVALID", "Invalid HTTP or HTTPS URL.");
  }
}

function checkItems(errors, source, path, callback) {
  const items = fieldValue(source, path);
  if (items === undefined || !Array.isArray(items)) return;

  items.forEach((item, index) => {
    const base = `${path}.${index}`;

    if (!object(item)) {
      error(errors, base, "PROJECT_ITEM_INVALID", "Expected an object.");
      return;
    }

    callback(item, base);
  });
}

function checkOrderedItems(errors, source, path, maximum) {
  checkItems(errors, source, path, (item, base) => {
    checkText(errors, `${base}.id`, item.id, 200);
    checkText(errors, `${base}.text`, item.text, maximum);
  });
}

export function validateNestedProject(source) {
  const errors = [];

  for (const [field, maximum] of Object.entries(TEXT_LIMITS)) {
    checkText(errors, field, fieldValue(source, field), maximum);
  }

  checkOption(errors, "ownership", source.ownership, OPTIONS.ownership);
  checkOption(errors, "source", source.source, OPTIONS.source);

  if (source.modelVersion !== undefined && source.modelVersion !== 2) {
    error(errors, "modelVersion", "PROJECT_VERSION_INVALID",
      "Unsupported Project model version.");
  }

  for (const field of [
    "liveUrl", "repositoryUrl", "documentationUrl",
    "caseStudyUrl", "videoUrl",
  ]) {
    checkUrl(errors, `links.${field}`, source.links?.[field]);
  }

  checkUrl(errors, "organization.website", source.organization?.website, 1000);

  for (const [path, maximum] of COLLECTION_FIELDS) {
    checkOrderedItems(errors, source, path, maximum);
  }

  checkItems(errors, source, "results.metrics", (item, base) => {
    checkText(errors, `${base}.label`, item.label, 200);
    checkText(errors, `${base}.value`, item.value, 200);
    checkText(errors, `${base}.description`, item.description, 500);
  });

  checkItems(errors, source, "skillRelationships", (item, base) => {
    checkText(errors, `${base}.nameSnapshot`, item.nameSnapshot, 200);
    checkText(errors, `${base}.usageDescription`, item.usageDescription, 1000);
    checkOption(errors, `${base}.demonstratedProficiency`,
      item.demonstratedProficiency, OPTIONS.proficiency);

    if (!item.skillId && !item.nameSnapshot) {
      error(errors, base, "PROJECT_SKILL_IDENTITY_REQUIRED",
        "A Skill identifier or snapshot is required.");
    }
  });

  checkItems(errors, source, "technologies", (item, base) => {
    checkText(errors, `${base}.name`, item.name, 150);
    checkText(errors, `${base}.category`, item.category, 100);
  });

  const relationshipTypes = [
    ["experienceRelationships", "experienceId"],
    ["educationRelationships", "educationId"],
    ["trainingRelationships", "trainingId"],
    ["certificationRelationships", "certificationId"],
  ];

  for (const [name, idField] of relationshipTypes) {
    checkItems(errors, source, `relatedRecords.${name}`, (item, base) => {
      checkText(errors, `${base}.${idField}`, item[idField], 200);
      checkText(errors, `${base}.description`, item.description, 1000);

      if (!item[idField]) {
        error(errors, `${base}.${idField}`, "PROJECT_RELATIONSHIP_ID_REQUIRED",
          "A related record identifier is required.");
      }

      if (item.snapshot !== undefined && !object(item.snapshot)) {
        error(errors, `${base}.snapshot`, "PROJECT_SNAPSHOT_INVALID",
          "Snapshot must be an object.");
      } else {
        checkText(errors, `${base}.snapshot.name`, item.snapshot?.name, 200);
        checkText(errors, `${base}.snapshot.secondaryLabel`,
          item.snapshot?.secondaryLabel, 200);
      }
    });
  }

  const documentIds = new Set(
    Array.isArray(source.supportingDocuments)
      ? source.supportingDocuments.map((item) => item?.id)
      : [],
  );

  checkItems(errors, source, "awards", (item, base) => {
    checkText(errors, `${base}.name`, item.name, 250);
    checkText(errors, `${base}.organization`, item.organization, 250);
    checkText(errors, `${base}.description`, item.description, 1500);
    checkOption(errors, `${base}.recognitionType`,
      item.recognitionType, OPTIONS.recognitionType);
    checkOption(errors, `${base}.placement`, item.placement, OPTIONS.placement);
    checkUrl(errors, `${base}.url`, item.url);

    if (item.supportingDocumentIds !== undefined) {
      if (!Array.isArray(item.supportingDocumentIds) ||
          item.supportingDocumentIds.length > 10) {
        error(errors, `${base}.supportingDocumentIds`,
          "PROJECT_AWARD_DOCUMENT_LIMIT", "Maximum 10 document references.");
      } else {
        for (const id of item.supportingDocumentIds) {
          if (!documentIds.has(id)) {
            error(errors, `${base}.supportingDocumentIds`,
              "PROJECT_AWARD_DOCUMENT_NOT_FOUND",
              "Referenced document does not exist in this Project.");
          }
        }
      }
    }
  });

  checkItems(errors, source, "media", (item, base) => {
    checkOption(errors, `${base}.type`, item.type, OPTIONS.mediaType);
    checkOption(errors, `${base}.visibility`,
      item.visibility, OPTIONS.visibility);
    checkOption(errors, `${base}.status`,
      item.status, OPTIONS.recordStatus);
    checkText(errors, `${base}.name`, item.name, 255);
    checkText(errors, `${base}.caption`, item.caption, 1000);
    checkText(errors, `${base}.altText`, item.altText, 500);
    checkUrl(errors, `${base}.externalUrl`, item.externalUrl);

    if (!item.storageKey && !item.externalUrl) {
      error(errors, base, "PROJECT_MEDIA_SOURCE_REQUIRED",
        "A storage key or external URL is required.");
    }
  });

  checkItems(errors, source, "supportingDocuments", (item, base) => {
    checkOption(errors, `${base}.documentType`,
      item.documentType, OPTIONS.documentType);
    checkOption(errors, `${base}.visibility`,
      item.visibility, OPTIONS.visibility);
    checkOption(errors, `${base}.status`,
      item.status, OPTIONS.recordStatus);
    checkText(errors, `${base}.name`, item.name, 255);
    checkText(errors, `${base}.description`, item.description, 1000);
    checkUrl(errors, `${base}.externalUrl`, item.externalUrl);

    if (!item.storageKey && !item.externalUrl) {
      error(errors, base, "PROJECT_DOCUMENT_SOURCE_REQUIRED",
        "A storage key or external URL is required.");
    }
  });

  if (source.visibility !== undefined) {
    if (!object(source.visibility)) {
      error(errors, "visibility", "PROJECT_VISIBILITY_INVALID",
        "Visibility must be an object.");
    } else {
      for (const [name, value] of Object.entries(source.visibility)) {
        if (typeof value !== "boolean") {
          error(errors, `visibility.${name}`, "PROJECT_VISIBILITY_INVALID",
            "Visibility must be true or false.");
        }
      }
    }
  }

  if (source.presentation?.featuredMediaId) {
    const featuredId = source.presentation.featuredMediaId;
    const eligible = Array.isArray(source.media) &&
      source.media.some((item) =>
        item?.id === featuredId && item.status !== "archived" &&
        item.visibility !== "private" &&
        (item.storageKey || item.externalUrl),
      );

    if (!eligible) {
      error(errors, "presentation.featuredMediaId",
        "PROJECT_FEATURED_MEDIA_INVALID",
        "Featured media must be active, public, and available.");
    }
  }

  return errors;
}

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import { getEnvironment } from "../../config/environment.js";

function requireIdentifier(value, label) {
  const normalized = String(value ?? "").trim();

  if (!normalized) {
    throw new Error(`${label} is required for Project media storage.`);
  }

  return normalized;
}

function hashIdentifier(value) {
  return crypto.createHash("sha256").update(value).digest("hex");
}

function validateFileName(fileName) {
  const normalized = String(fileName ?? "").trim();

  if (
    !normalized ||
    normalized === "." ||
    normalized === ".." ||
    normalized.includes("/") ||
    normalized.includes("\\") ||
    path.basename(normalized) !== normalized ||
    !/^[a-f0-9-]{36}\.(?:jpg|jpeg|png|webp|avif|gif|mp4|webm)$/i.test(
      normalized,
    )
  ) {
    throw new Error("Invalid Project media filename.");
  }

  return normalized;
}

export function getProjectMediaStorageRoot() {
  const environment = getEnvironment();

  const configuredPath = String(
    environment.PROJECT_MEDIA_STORAGE_PATH || "",
  ).trim();

  if (!configuredPath) {
    throw new Error("PROJECT_MEDIA_STORAGE_PATH is not configured.");
  }

  return path.resolve(configuredPath);
}

export function getProjectMediaUserDirectory(userId) {
  return hashIdentifier(requireIdentifier(userId, "User ID"));
}

export function getProjectMediaProjectDirectory(projectId) {
  return hashIdentifier(requireIdentifier(projectId, "Project ID"));
}

export function getProjectMediaDirectory(userId, projectId) {
  return path.join(
    getProjectMediaStorageRoot(),
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
  );
}

export function createProjectMediaStorageKey(userId, projectId, fileName) {
  const safeFileName = validateFileName(fileName);

  return [
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
    safeFileName,
  ].join("/");
}

export function resolveOwnedProjectMediaPath(userId, projectId, storageKey) {
  const normalizedKey = String(storageKey ?? "").trim();

  const expectedUserDirectory = getProjectMediaUserDirectory(userId);
  const expectedProjectDirectory = getProjectMediaProjectDirectory(projectId);

  const parts = normalizedKey.split("/");

  if (
    parts.length !== 3 ||
    parts[0] !== expectedUserDirectory ||
    parts[1] !== expectedProjectDirectory
  ) {
    throw new Error(
      "Project media storage key does not belong to this user and Project.",
    );
  }

  const safeFileName = validateFileName(parts[2]);

  const mediaDirectory = path.resolve(
    getProjectMediaDirectory(userId, projectId),
  );

  const filePath = path.resolve(mediaDirectory, safeFileName);

  if (path.dirname(filePath) !== mediaDirectory) {
    throw new Error("Project media path is outside the expected directory.");
  }

  return filePath;
}

export async function assertSafeProjectMediaDirectory(userId, projectId) {
  const root = getProjectMediaStorageRoot();
  const userDirectory = path.join(root, getProjectMediaUserDirectory(userId));
  const projectDirectory = path.join(
    userDirectory,
    getProjectMediaProjectDirectory(projectId),
  );

  const directories = [root, userDirectory, projectDirectory];

  for (const directory of directories) {
    const stats = await fs.lstat(directory);

    if (stats.isSymbolicLink() || !stats.isDirectory()) {
      throw new Error(
        "Project media storage directory is not a regular directory.",
      );
    }
  }

  const canonicalRoot = await fs.realpath(root);
  const canonicalProjectDirectory = await fs.realpath(projectDirectory);

  const relative = path.relative(
    canonicalRoot,
    canonicalProjectDirectory,
  );

  if (
    relative === "" ||
    relative === ".." ||
    relative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(relative)
  ) {
    throw new Error("Project media directory is outside the storage root.");
  }
}

export async function deleteOwnedProjectMedia(userId, projectId, storageKey) {
  const filePath = resolveOwnedProjectMediaPath(userId, projectId, storageKey);

  try {
    await assertSafeProjectMediaDirectory(userId, projectId);

    const stats = await fs.lstat(filePath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw new Error("Project media target is not a regular file.");
    }

    await fs.unlink(filePath);

    return {
      deleted: true,
      storageKey,
    };
  } catch (error) {
    if (error?.code === "ENOENT") {
      return {
        deleted: false,
        storageKey,
      };
    }

    throw error;
  }
}

function setPrivateHeaders(response) {
  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",
    Pragma: "no-cache",
    Expires: "0",
  });
}

function projectRequestError(message) {
  const error = new Error(message);
  error.name = "ProjectRequestError";
  error.code = "PROJECT_REQUEST_INVALID";
  error.status = 400;
  error.publicMessage = message;
  return error;
}

function requireObject(value, name) {
  if (value === null || typeof value !== "object" || Array.isArray(value)) {
    throw projectRequestError(`${name} must be a JSON object.`);
  }

  return value;
}

function requireFields(body, allowedFields) {
  const input = requireObject(body, "Request body");

  const unexpected = Object.keys(input).filter(
    (field) => !allowedFields.includes(field),
  );

  if (unexpected.length > 0) {
    throw projectRequestError(
      `Unexpected request fields: ${unexpected.join(", ")}.`,
    );
  }

  return input;
}
