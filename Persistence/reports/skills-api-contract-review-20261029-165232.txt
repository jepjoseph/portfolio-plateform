  138:           "Years of experience can contain no more than one decimal place.",

  198: function validateSkillRelationships(skill, context) {
  199:   const isLanguageType = skill.type === "language";
  200: 
  201:   const isLanguageCategory = skill.category === "language";
  202: 
  203:   const languageProficiency = skill.language?.proficiency || "";
  204: 
  205:   if (isLanguageType && !isLanguageCategory) {
  206:     context.addIssue({
  207:       code: z.ZodIssueCode.custom,
  208:       path: ["category"],
  209:       message: 'A language skill must use the "Languages" category.',
  210:     });
  211:   }
  212: 
  213:   if (isLanguageCategory && !isLanguageType) {
  214:     context.addIssue({
  215:       code: z.ZodIssueCode.custom,
  216:       path: ["type"],
  217:       message:
  218:         'A skill in the "Languages" category must use the "Language Skill" type.',
  219:     });
  220:   }
  221: 
  222:   if (!isLanguageType && languageProficiency) {
  223:     context.addIssue({
  224:       code: z.ZodIssueCode.custom,
  225:       path: ["language", "proficiency"],
  226:       message: "Language proficiency can only be assigned to a language skill.",
  227:     });
  228:   }
  229: 
  230:   const normalizedName = skill.name
  231:     .normalize("NFKC")
  232:     .trim()
  233:     .toLocaleLowerCase()
  234:     .replace(/[._/\\-]+/g, " ")
  235:     .replace(/\s+/g, " ");
  236: 
  237:   const seenAliases = new Set();
  238: 
  239:   skill.aliases.forEach((alias, index) => {
  240:     const normalizedAlias = alias
  241:       .normalize("NFKC")
  242:       .trim()
  243:       .toLocaleLowerCase()
  244:       .replace(/[._/\\-]+/g, " ")
  245:       .replace(/\s+/g, " ");
  246: 
  247:     if (normalizedAlias === normalizedName) {
  248:       context.addIssue({
  249:         code: z.ZodIssueCode.custom,
  250:         path: ["aliases", index],
  251:         message: "An alias cannot be the same as the primary skill name.",
  252:       });
  253:     }
  254: 
  255:     if (seenAliases.has(normalizedAlias)) {
  256:       context.addIssue({
  257:         code: z.ZodIssueCode.custom,
  258:         path: ["aliases", index],
  259:         message: "This skill alias has already been added.",
  260:       });
  261:     }
  262: 
  263:     seenAliases.add(normalizedAlias);
  264:   });
  265: }

  367: function parseBooleanQuery(value) {
  368:   if (value === true || value === "true" || value === "1") {
  369:     return true;
  370:   }
  371: 
  372:   if (
  373:     value === false ||
  374:     value === "false" ||
  375:     value === "0" ||
  376:     value === undefined

  378:     return false;
  379:   }
  380: 
  381:   return value;
  382: }
