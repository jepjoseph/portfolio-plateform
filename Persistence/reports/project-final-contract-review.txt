function createValidationError(errors) {
  const error = new Error("Project validation failed.");

  error.name = "ProjectValidationError";
  error.code = "PROJECT_VALIDATION_FAILED";
  error.status = 400;
  error.publicMessage = "Please correct the Project information.";
  error.errors = errors;

  error.fieldErrors = {};

  for (const item of errors) {
    if (!error.fieldErrors[item.field]) {
      error.fieldErrors[item.field] = item.message;
    }
  }

  return error;
}

function addError(errors, field, code, message) {
  errors.push({ field, code, message });
}

function isPlainObject(value) {
  return value !== null && typeof value === "object" && !Array.isArray(value);
}

function isValidProjectDate(value) {
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

export function parseProjectInput(value) {
  const errors = [];

  if (!isPlainObject(value)) {
    throw createValidationError([
      {
        field: "project",
        code: "PROJECT_OBJECT_REQUIRED",
        message: "Project data must be an object.",
      },
    ]);
  }

  for (const field of SERVER_OWNED_FIELDS) {
    if (Object.hasOwn(value, field)) {
      addError(
        errors,
        field,
        "PROJECT_SERVER_FIELD_FORBIDDEN",
        `${field} cannot be supplied by the client.`,
      );
    }
  }

  if (typeof value.id !== "string" || !PROJECT_ID_PATTERN.test(value.id)) {
    addError(
      errors,
      "id",
      "PROJECT_ID_INVALID",
      "A valid Project identifier is required.",
    );
  }

  if (
    typeof value.title !== "string" ||
    !value.title.trim() ||
    value.title.trim().length > 200
  ) {
    addError(
      errors,
      "title",
      "PROJECT_TITLE_INVALID",
      "Project title must contain 1 to 200 characters.",
    );
  }

  if (
    typeof value.role !== "string" ||
    !value.role.trim() ||
    value.role.trim().length > 160
  ) {
    addError(
      errors,
      "role",
      "PROJECT_ROLE_INVALID",
      "Project role must contain 1 to 160 characters.",
    );
  }

  if (!CATEGORIES.has(value.category)) {
    addError(
      errors,
      "category",
      "PROJECT_CATEGORY_INVALID",
      "Select a valid Project category.",
    );
  }

  if (!LIFECYCLE_STATUSES.has(value.lifecycleStatus)) {
    addError(
      errors,
      "lifecycleStatus",
      "PROJECT_LIFECYCLE_INVALID",
      "Select a valid lifecycle status.",
    );
  }

  if (!RECORD_STATUSES.has(value.recordStatus)) {
    addError(
      errors,
      "recordStatus",
      "PROJECT_RECORD_STATUS_INVALID",
      "Select a valid record status.",
    );
  }

  if (!isPlainObject(value.dates)) {
    addError(
      errors,
      "dates",
      "PROJECT_DATES_INVALID",
      "Project dates must be an object.",
    );
  } else {
    if (!isValidProjectDate(value.dates.startDate)) {
      addError(
        errors,
        "dates.startDate",
        "PROJECT_START_DATE_INVALID",
        "A valid Project start date is required.",
      );
    }

    if (value.dates.endDate && !isValidProjectDate(value.dates.endDate)) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_INVALID",
        "Enter a valid Project end date.",
      );
    }

    if (value.dates.isCurrent === true && value.dates.endDate) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_CURRENT_END_DATE_CONFLICT",
        "An ongoing Project cannot have an end date.",
      );
    }

    if (
      value.dates.isCurrent === true &&
      !["in-progress", "maintained"].includes(value.lifecycleStatus)
    ) {
      addError(
        errors,
        "lifecycleStatus",
        "PROJECT_CURRENT_LIFECYCLE_CONFLICT",
        "An ongoing Project must be In Progress or Maintained.",
      );
    }

    if (
      ["completed", "cancelled"].includes(value.lifecycleStatus) &&
      value.dates.isCurrent !== true &&
      !value.dates.endDate
    ) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_REQUIRED",
        "A completion or cancellation date is required.",
      );
    }

    if (
      isValidProjectDate(value.dates.startDate) &&
      isValidProjectDate(value.dates.endDate)
    ) {
      const start = new Date(
        value.dates.startDate.length === 7
          ? `${value.dates.startDate}-01`
          : value.dates.startDate,
      );

      const end = new Date(
        value.dates.endDate.length === 7
          ? `${value.dates.endDate}-01`
          : value.dates.endDate,
      );

      if (end < start) {
        addError(
          errors,
          "dates.endDate",
          "PROJECT_DATE_ORDER_INVALID",
          "End date cannot be earlier than start date.",
        );
      }
    }
  }

  if (
    !isPlainObject(value.presentation) ||
    typeof value.presentation.isFeatured !== "boolean"
  ) {
    addError(
      errors,
      "presentation.isFeatured",
      "PROJECT_FEATURED_INVALID",
      "Project featured status must be a boolean.",
    );
  }

  const urlFields = [
    ["organization.website", value.organization?.website],
    ...[
      "liveUrl",
      "repositoryUrl",
      "documentationUrl",
      "caseStudyUrl",
      "videoUrl",
    ].map((name) => [`links.${name}`, value.links?.[name]]),
  ];

  for (const [field, urlValue] of urlFields) {
    if (urlValue === undefined || urlValue === "") continue;

    let valid = false;

    if (typeof urlValue === "string") {
      try {
        const parsed = new URL(urlValue);
        valid = ["http:", "https:"].includes(parsed.protocol);
      } catch {
        valid = false;
      }
    }

    if (!valid) {
      addError(
        errors,
        field,
        "PROJECT_URL_INVALID",
        "Enter a valid HTTP or HTTPS URL.",
      );
    }
  }

  const collectionLimits = [
    ["problem.objectives", value.problem?.objectives, 25],
    ["problem.constraints", value.problem?.constraints, 25],
    ["solution.features", value.solution?.features, 40],
    ["solution.contributions", value.solution?.contributions, 40],
    ["results.metrics", value.results?.metrics, 25],
    ["skillRelationships", value.skillRelationships, 60],
    ["technologies", value.technologies, 60],
    ["awards", value.awards, 20],
    ["media", value.media, 30],
    ["supportingDocuments", value.supportingDocuments, 30],
    ...[
      "experienceRelationships",
      "educationRelationships",
      "trainingRelationships",
      "certificationRelationships",
    ].map((name) => [
      `relatedRecords.${name}`,
      value.relatedRecords?.[name],
      50,
    ]),
  ];

  for (const [field, items, maximum] of collectionLimits) {
    if (items === undefined) continue;

    if (!Array.isArray(items) || items.length > maximum) {
      addError(
        errors,
        field,
        "PROJECT_COLLECTION_INVALID",
        `This collection must contain at most ${maximum} items.`,
      );
    }
  }
  errors.push(...validateNestedProject(value));

  if (errors.length > 0) {
    throw createValidationError(errors);
  }

  return structuredClone(value);
}

const PROJECT_COLUMNS = `
  id,
  user_id,
  title,
  category,
  lifecycle_status,
  record_status,
  is_featured,
  project_data,
  created_at,
  updated_at,

function mapProjectRow(row) {
  if (!row) {
    return null;
  }

  const projectData =
    typeof row.project_data === "string"
      ? JSON.parse(row.project_data)
      : row.project_data;

  return {
    ...projectData,
    id: row.id,
    title: row.title,
    category: row.category,
    lifecycleStatus: row.lifecycle_status,
    recordStatus: row.record_status,
    presentation: {
      ...(projectData.presentation || {}),
      isFeatured: Boolean(row.is_featured),
    },
    createdAt: row.created_at,
    updatedAt: row.updated_at,
    rowVersion: Buffer.from(row.row_version).toString("base64"),
  };
}

function bindProjectParameters(request, userId, project) {
  return request
    .input("id", sql.NVarChar(200), project.id)
    .input("userId", sql.UniqueIdentifier, userId)
    .input("title", sql.NVarChar(500), project.title)
    .input("category", sql.NVarChar(100), project.category)
    .input("lifecycleStatus", sql.NVarChar(50), project.lifecycleStatus)
    .input("recordStatus", sql.NVarChar(20), project.recordStatus)
    .input("isFeatured", sql.Bit, Boolean(project.presentation?.isFeatured))
    .input("projectData", sql.NVarChar(sql.MAX), JSON.stringify(project));
}

export async function listProjectsByUser(userId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
      ORDER BY updated_at DESC, id ASC;
    `);

  return result.recordset.map(mapProjectRow);
}

export async function findProjectById(userId, projectId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
        AND id = @id;
    `);

  return mapProjectRow(result.recordset[0]);
}

export async function createProject(userId, project) {

      id,
      user_id,
      title,
      category,
      lifecycle_status,
      record_status,
      is_featured,
      project_data
    )
    VALUES (
      @id,
      @userId,
      @title,
      @category,
      @lifecycleStatus,
      @recordStatus,
      @isFeatured,
      @projectData
    );

export async function updateProject(userId, project, expectedRowVersion) {
  const pool = await getDatabasePool();

  const request = bindProjectParameters(pool.request(), userId, project).input(
    "expectedRowVersion",
    sql.Binary(8),
    Buffer.from(expectedRowVersion, "base64"),
  );

  const result = await request.query(`
    UPDATE dbo.projects
    SET
      title = @title,
      category = @category,
      lifecycle_status = @lifecycleStatus,
      record_status = @recordStatus,
      is_featured = @isFeatured,
      project_data = @projectData,
      updated_at = SYSUTCDATETIME()
    WHERE id = @id
      AND user_id = @userId
      AND row_version = @expectedRowVersion;
  `);

  if (result.rowsAffected[0] === 0) {
    return null;
  }

  return findProjectById(userId, project.id);
}

export async function deleteProject(userId, projectId, expectedRowVersion) {
  const pool = await getDatabasePool();

    .request()

    ).query(`
      DELETE FROM dbo.projects
      WHERE id = @id
        AND user_id = @userId
        AND row_version = @expectedRowVersion;

}

function isValidUrl(value) {
  const normalizedValue = getText(value);

  if (!normalizedValue) {
    return true;
  }

  try {
    const url = new URL(normalizedValue);

    return url.protocol === "http:" || url.protocol === "https:";
  } catch {
    return false;
  }
}

function parseProjectDate(value) {
  const normalizedValue = getText(value);

  if (!normalizedValue) {
    return null;
  }

  const match = normalizedValue.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return null;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const parsedDate = new Date(Date.UTC(year, month - 1, day));

  if (
    match[3] &&
    (parsedDate.getUTCFullYear() !== year ||
      parsedDate.getUTCMonth() !== month - 1 ||
      parsedDate.getUTCDate() !== day)
  ) {
    return null;
  }

  return parsedDate.getTime();
}

export function getProjectValidationSummary(value) {
  const project = normalizeProject(value);

  const activePublicMedia = project.media.filter(
    (mediaItem) =>
      mediaItem.status !== "archived" && mediaItem.visibility === "public",
  );

  const checks = [
    {
      id: "identity",
      complete: Boolean(project.title && project.role),
    },
    {
      id: "timeline",
      complete: Boolean(
        project.dates.startDate &&
        (project.dates.isCurrent ||
          project.dates.endDate ||
          !["completed", "cancelled"].includes(project.lifecycleStatus)),
      ),
    },
    {
      id: "problem",
      complete: Boolean(project.problem.statement),
    },
    {
      id: "importance",
      complete: Boolean(project.problem.importance),
    },
    {
      id: "solution",
      complete: Boolean(project.solution.overview),
    },
    {
      id: "contributions",
      complete: project.solution.contributions.length > 0,
    },
    {
      id: "results",
      complete: Boolean(
        project.results.outcome || project.lifecycleStatus !== "completed",
      ),
    },
    {
      id: "evidence",
      complete:
        project.results.metrics.length > 0 ||
        project.supportingDocuments.length > 0,
    },
    {
      id: "skills",
      complete: project.skillRelationships.length > 0,
    },
    {
      id: "technologies",
      complete: project.technologies.length > 0,
    },
    {
      id: "summary",
      complete: Boolean(project.presentation.shortSummary),
    },
    {
      id: "media",
      complete: activePublicMedia.length > 0,
    },
  ];

  const completedCount = checks.filter((check) => check.complete).length;

  const percentage = Math.round((completedCount / checks.length) * 100);

  let level = "Basic";

  if (percentage >= 90) {
    level = "Portfolio Ready";
  } else if (percentage >= 70) {
    level = "Strong";
  } else if (percentage >= 50) {
    level = "Developing";
  }

  return `${completedCount} of ${checks.length} professional elements completed Â· ${level}`;
}
