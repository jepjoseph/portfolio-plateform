 127: function createRequest({
 128:   refreshToken = REFRESH_TOKEN,
 129:   includeCookie = true,
 130: } = {}) {
 131:   return {
 132:     headers: includeCookie

 144: /*
 145:  * =========================================
 146:  * Response Factory
 147:  * =========================================
 148:  */
 149: 
 150: function createResponse() {
 151:   const response = {
 152:     headers: {},
 153: 
 154:     clearedCookies: [],
 155: 
 156:     variedHeaders: [],
 157: 
 158:     set(values) {
 159:       for (const [name, value] of Object.entries(values)) {
 160:         this.headers[name.toLowerCase()] = value;
 161:       }
 162: 
 163:       return this;
 164:     },
 165: 
 166:     setHeader(name, value) {
 167:       this.headers[String(name).toLowerCase()] = value;
 168: 
 169:       return this;
 170:     },
 171: 
 172:     vary(fieldName) {
 173:       this.variedHeaders.push(fieldName);
 174: 
 175:       return this;
 176:     },
 177: 
 178:     clearCookie(name, options) {
 179:       this.clearedCookies.push({
 180:         name,
 181: 
 182:         options,
 183:       });
 184: 
 185:       return this;
 186:     },
 187:   };
 188: 
 189:   return response;
 190: }
