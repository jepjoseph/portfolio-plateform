const TRAINING_COLUMNS = `
  id,
  user_id,
  title,
  training_type,
  provider_name,
  status,
  training_data,
  created_at,
  updated_at,
  row_version
`;

function mapTrainingRow(row) {
  if (!row) return null;

  const data = typeof row.training_data === "string"
    ? JSON.parse(row.training_data)
    : row.training_data;

  if (!data || typeof data !== "object" || Array.isArray(data)) {
    throw new Error("Stored Training data is invalid.");
  }

  return {
    ...data,
    id: row.id,
    title: row.title,
    trainingType: row.training_type,
    provider: {
      ...(data.provider || {}),
      name: row.provider_name,
    },
    status: row.status,
    createdAt: new Date(row.created_at).toISOString(),
    updatedAt: new Date(row.updated_at).toISOString(),
    rowVersion: Buffer.from(row.row_version).toString("base64"),
  };
}

function bindTraining(request, userId, training) {
  return request
    .input("id", sql.NVarChar(200), training.id)
    .input("userId", sql.UniqueIdentifier, userId)
    .input("title", sql.NVarChar(250), training.title)
    .input("trainingType", sql.NVarChar(100), training.trainingType)
    .input("providerName", sql.NVarChar(500), training.provider.name)
    .input("status", sql.NVarChar(20), training.status)
    .input(
      "trainingData",
      sql.NVarChar(sql.MAX),
      JSON.stringify(training),
    );
}

function decodeRowVersion(value) {
  if (
    typeof value !== "string" ||
    !/^[A-Za-z0-9+/]{11}=$/.test(value)
  ) {
    throw new TypeError("A valid rowVersion is required.");
  }

  const buffer = Buffer.from(value, "base64");

  if (
    buffer.length !== 8 ||
    buffer.toString("base64") !== value
  ) {
    throw new TypeError("Invalid rowVersion.");
  }

  return buffer;
}

function certificationError(message, code, status, errors) {
  const error = new Error(message);
  error.name = "CertificationServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = message;

  if (errors) error.errors = errors;

  return error;
}

function validate(schema, value) {
  const result = schema.safeParse(value);

  if (!result.success) {
    throw certificationError(
      "Certification validation failed.",
      "CERTIFICATION_VALIDATION_FAILED",
      400,
      formatCertificationSchemaErrors(result.error),
    );
  }

  return result.data;
}

function requireUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw certificationError(
      "Authentication is required.",
      "CERTIFICATION_AUTH_REQUIRED",
      401,
    );
  }

  return userId;
}

function requireNoDocuments(certification) {
  if (certification.supportingDocuments.length > 0) {
    throw certificationError(
      "Certification document uploads are not available yet.",
      "CERTIFICATION_DOCUMENTS_NOT_READY",
      400,
    );
  }
}

function normalizeRecord(record) {

  return {
    ...record,
    createdAt: new Date(record.createdAt).toISOString(),
    updatedAt: new Date(record.updatedAt).toISOString(),
  };
}

export async function createAuthenticatedCertification({
  userId,
  certification,
}) {
  requireUserId(userId);

  if (
    certification === null ||
    typeof certification !== "object" ||
    Array.isArray(certification)
  ) {
    throw certificationError(
      "Certification must be a JSON object.",
      "CERTIFICATION_REQUEST_INVALID",
      400,
    );
  }

  const now = new Date().toISOString();

  const candidate = {
    ...certification,
    id: randomUUID(),
    createdAt: now,
    updatedAt: now,
  };

  const validated = validate(
    createCertificationSchema,
    candidate,
  );

  requireNoDocuments(validated);

  try {
    const created = await createCertificationRecord(
      userId,
      validated,
    );

    return normalizeRecord(created);
  } catch (error) {
    const number =
      error.number ??
      error.originalError?.number ??
      error.cause?.number;

    if (number === 2601 || number === 2627) {
      throw certificationError(
        "A Certification identifier already exists.",
        "CERTIFICATION_ID_CONFLICT",
        409,
      );
    }

    throw error;
  }
}

    }
  }
}

    completeCertificationSchema,
    merged,

    userId,
    complete,

function assertUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {

function assertProjectId(projectId) {

    message: "Project version conflict.",
    code: "PROJECT_VERSION_CONFLICT",
    status: 409,

  });

function projectIdMismatch() {

    publicMessage: "The Project identifiers do not match.",
  });
}

function withoutServerFields(project) {
  const result = { ...project };

  for (const field of SERVER_OWNED_FIELDS) {
    delete result[field];
  }

  return result;
}

export async function createAuthenticatedProject({ userId, project }) {
  const authenticatedUserId = assertUserId(userId);

  // Reject invalid data before any database operation.
  // Client-supplied server-owned fields remain forbidden.
  const validatedProject = parseProjectInput(project);

  try {
    return await createProject(authenticatedUserId, validatedProject);
  } catch (error) {
    if (error.number === 2627 || error.number === 2601) {
      throw createProjectError({
        message: "Project ID already exists.",
        code: "PROJECT_ALREADY_EXISTS",
        status: 409,
        publicMessage: "A Project with this identifier already exists.",
      });
    }

    throw error;
  }
}

export async function updateAuthenticatedProject({
  userId,
  projectId,
  project,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);

  // Validate the supplied version and full Project body.
  const expectedRowVersion = parseProjectRowVersion(rowVersion);
  const validatedProject = parseProjectInput(project);

  if (validatedProject.id !== normalizedProjectId) {
    throw projectIdMismatch();
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function setAuthenticatedProjectArchiveStatus({
  userId,
  projectId,
  rowVersion,
  archived,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  if (typeof archived !== "boolean") {
    throw createProjectError({
      message: "Archive status must be a boolean.",
      code: "PROJECT_ARCHIVE_STATUS_INVALID",
      status: 400,
      publicMessage: "A valid archive status is required.",
    });
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  // Database-managed metadata is not editable Project content.
  const editableProject = withoutServerFields(existingProject);

  const validatedProject = parseProjectInput({
    ...editableProject,
    recordStatus: archived ? "archived" : "active",
  });

  // SQL Server must match the caller's row version.
  // Do not substitute the version from the latest database read.
  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function deleteAuthenticatedProject({
  userId,
  projectId,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const deleted = await deleteProject(
    authenticatedUserId,
    normalizedProjectId,
    expectedRowVersion,
  );

  if (!deleted) {
    throw projectVersionConflict();
  }

  return { deleted: true };
}

function validateCertificationRules(value, context) {
  const { dates, credential, supportingDocuments } = value;

  function issue(path, message) {
    context.addIssue({
      code: "custom",
      path,
      message,
    });
  }

  if (credential.state !== "planned" && !dates.issueDate) {
    issue(["dates", "issueDate"], "Issue date is required.");
  }

  if (dates.issueDate && dates.expirationDate &&
      dates.expirationDate < dates.issueDate) {
    issue(["dates", "expirationDate"],
      "Expiration date cannot precede issue date.");
  }

  if (dates.issueDate && dates.lastRenewedDate &&
      dates.lastRenewedDate < dates.issueDate) {
    issue(["dates", "lastRenewedDate"],
      "Renewal date cannot precede issue date.");
  }

  if (dates.lastRenewedDate && dates.nextRenewalDate &&
      dates.nextRenewalDate < dates.lastRenewedDate) {
    issue(["dates", "nextRenewalDate"],
      "Next renewal cannot precede last renewal.");
  }

  if (dates.doesNotExpire &&
      (dates.expirationDate || dates.nextRenewalDate)) {
    issue(["dates", "doesNotExpire"],
      "Non-expiring certifications cannot have expiration or next-renewal dates.");
  }

  const active = supportingDocuments.filter(
    (document) => document.status !== "archived",
  );

  const primary = active.filter(
    (document) => document.isPrimary,
  );

  if (active.length && primary.length !== 1) {
    issue(["supportingDocuments"],
      "Exactly one active document must be primary.");
  }

  const seenDocuments = new Set();

  supportingDocuments.forEach((document, index) => {
    if (seenDocuments.has(document.id)) {
      issue(["supportingDocuments", index, "id"],
        "Duplicate document ID.");
    }
    seenDocuments.add(document.id);

    if (document.status !== "archived" && !document.storageKey) {
      issue(["supportingDocuments", index, "storageKey"],
        "Active documents require a storage key.");
    }

    const extension = document.fileName.toLowerCase().slice(
      document.fileName.lastIndexOf("."),
    );

    if (extension !== document.extension) {
      issue(["supportingDocuments", index, "extension"],
        "Document extension does not match the filename.");
    }
  });

  for (const [records, field, key] of [
    [value.skillRelationships, "skillRelationships", "skillId"],
    [value.relatedRecords.trainingRelationships,
      "relatedRecords.trainingRelationships", "trainingId"],
  ]) {
    const seen = new Set();

    records.forEach((record, index) => {
      if (seen.has(record[key])) {
        issue([field, index, key], "Duplicate relationship.");
      }
      seen.add(record[key]);
    });
  }

  const educationIds = value.relatedRecords.educationIds;
  if (new Set(educationIds).size !== educationIds.length) {
    issue(["relatedRecords", "educationIds"],
      "Duplicate Education relationship.");
  }

function isValidProjectDate(value) {
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

export function parseProjectInput(value) {

    );

    );

  if (!isPlainObject(value.dates)) {
    addError(
      errors,
      "dates",
      "PROJECT_DATES_INVALID",
      "Project dates must be an object.",

    if (!isValidProjectDate(value.dates.startDate)) {
      addError(
        errors,

        "A valid Project start date is required.",
      );

    if (value.dates.endDate && !isValidProjectDate(value.dates.endDate)) {
      addError(

        "A completion or cancellation date is required.",
      );
    }

    if (
      isValidProjectDate(value.dates.startDate) &&

      const start = new Date(

/*
 * =========================================
 * Global Error Handler
 * =========================================
 */

export function globalErrorHandler(error, request, response, next) {
  /*
   * Express delegates to its default handler
   * if a response has already started.
   */

  if (response.headersSent) {
    return next(error);
  }

  const isMalformedJson =
    error?.type === "entity.parse.failed" && error instanceof SyntaxError;

  const status = isMalformedJson ? 400 : normalizeStatus(error?.status);

  const code = isMalformedJson
    ? "INVALID_JSON"
    : normalizePublicCode(error, status);

  const message = isMalformedJson
    ? "The request body contains invalid JSON."
    : getPublicMessage(error, status);

  /*
   * Log complete technical information only
   * on the server.
   */

  console.error("Unhandled API error:", {
    method: request.method,

    url: request.originalUrl,

    name: error?.name,

    code: error?.code,

    message: error?.message,

    status,

    provider: error?.provider,

    stack: process.env.NODE_ENV === "development" ? error?.stack : undefined,
  });

  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",

    Pragma: "no-cache",

    Expires: "0",
  });

  const fieldErrors = normalizeFieldErrors(error?.fieldErrors);

  const validationErrors = normalizeValidationErrors(
    error?.errors || error?.validationErrors,
  );

  return response.status(status).json({
    status: "error",

    code,

    message,

    ...(Object.keys(fieldErrors).length > 0
      ? {
          fieldErrors,
        }
      : {}),

    ...(validationErrors.length > 0
      ? {
          errors: validationErrors,
        }
      : {}),

    /*
     * Development details are useful while
     * building the API but are excluded from
     * production responses.
     */

    ...(process.env.NODE_ENV === "development"
      ? {
          debug: {
            name: error?.name || "Error",

            technicalMessage: error?.message || "Unknown error",

            provider: error?.provider || null,
          },
        }
      : {}),
  });
}

function normalizeStatus(status) {
  const numericStatus = Number(status);

  if (
    Number.isInteger(numericStatus) &&
    numericStatus >= 400 &&
    numericStatus <= 599
  ) {
    return numericStatus;
  }

  return 500;
}

/*
 * =========================================
 * Public Error Code
 * =========================================
 *
 * Internal database and provider codes should
 * not automatically be returned to clients.
 * Application errors are public only when
 * they also provide a publicMessage.
 */

function normalizePublicCode(error, status) {
  const code = typeof error?.code === "string" ? error.code.trim() : "";

  const isSafeCode = /^[A-Z][A-Z0-9_]{1,99}$/.test(code);

  if (error?.publicMessage && isSafeCode) {
    return code;
  }

  if (status === 400) {
    return "BAD_REQUEST";
  }

  if (status === 401) {
    return "AUTHENTICATION_REQUIRED";
  }

  if (status === 403) {
    return "ACCESS_FORBIDDEN";
  }

  if (status === 404) {
    return "RESOURCE_NOT_FOUND";
  }

  if (status === 409) {
    return "RESOURCE_CONFLICT";
  }

  if (status === 429) {
    return "RATE_LIMIT_EXCEEDED";
  }

  if (status === 503) {
    return "SERVICE_UNAVAILABLE";
  }

  return "INTERNAL_SERVER_ERROR";
}

/*
 * =========================================
 * Public Error Message
 * =========================================
 */

function getPublicMessage(error, status) {
  if (typeof error?.publicMessage === "string" && error.publicMessage.trim()) {
    return error.publicMessage.trim();
  }

  if (status === 400) {
    return "The request could not be processed.";
  }

  if (status === 401) {
    return "Authentication is required.";
  }

  if (status === 403) {
    return "You do not have permission to perform this operation.";
  }

  if (status === 404) {
    return "The requested resource was not found.";
  }

  if (status === 409) {
    return "The request conflicts with the current resource state.";
  }

  if (status === 429) {
    return "Too many requests were submitted. Please try again later.";
  }

  if (status === 503) {
    return "The service is temporarily unavailable.";
  }

  return "An unexpected server error occurred.";
}

/*
 * =========================================
 * Field Error Normalization
 * =========================================
 */

function normalizeFieldErrors(fieldErrors) {
  if (
    !fieldErrors ||
    typeof fieldErrors !== "object" ||
    Array.isArray(fieldErrors)
  ) {
    return {};
  }

  const normalized = {};

  for (const [fieldName, messages] of Object.entries(fieldErrors)) {
    const safeFieldName = String(fieldName).slice(0, 100);

    const messageValues = Array.isArray(messages) ? messages : [messages];

    const safeMessages = messageValues
      .filter((message) => typeof message === "string")
      .map((message) => message.trim().slice(0, 500))
      .filter(Boolean);

    if (safeMessages.length > 0) {
      normalized[safeFieldName] = safeMessages;
    }
  }

  return normalized;
}

/*
 * =========================================
 * Validation Error Normalization
 * =========================================
 */

function normalizeValidationErrors(errors) {
  if (!Array.isArray(errors)) {
    return [];
  }

  return errors
    .slice(0, 50)
    .map((validationError) => {
      if (typeof validationError === "string") {
        return {
          field: "",

          code: "VALIDATION_ERROR",

          message: validationError.trim().slice(0, 500),
        };
      }

      if (!validationError || typeof validationError !== "object") {
        return null;
      }

      const message =
        typeof validationError.message === "string"
          ? validationError.message.trim().slice(0, 500)
          : "";

      if (!message) {
        return null;
      }

      return {
        field:
          typeof validationError.field === "string"
            ? validationError.field.trim().slice(0, 100)
            : "",

        code:
          typeof validationError.code === "string"
            ? validationError.code.trim().slice(0, 100)
            : "VALIDATION_ERROR",

        message,
      };
    })
    .filter(Boolean);
}
