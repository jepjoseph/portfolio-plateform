import { getAuthenticatedSession } from "../services/Service/sessionService.js";

import {
  clearRefreshTokenCookie,
  getRefreshTokenFromRequest,
} from "../security/sessionCookieSecurity.js";

/*
 * =========================================
 * Authentication Error
 * =========================================
 */

function createAuthenticationError({
  message,
  code = "AUTHENTICATION_REQUIRED",
  status = 401,
  publicMessage = "You must sign in to access this resource.",
}) {
  const error = new Error(message);

  error.name = "AuthenticationError";

  error.code = code;

  error.status = status;

  error.publicMessage = publicMessage;

  return error;
}

/*
 * =========================================
 * Invalid Session Errors
 * =========================================
 */

function shouldClearSessionCookie(error) {
  const invalidSessionCodes = new Set([
    "SESSION_REFRESH_TOKEN_INVALID",
    "SESSION_NOT_FOUND",
    "SESSION_REFRESH_CONFLICT",
  ]);

  return invalidSessionCodes.has(error?.code);
}

/*
 * =========================================
 * Safely Clear Refresh Cookie
 * =========================================
 */

function safelyClearRefreshTokenCookie(response) {
  try {
    clearRefreshTokenCookie(response);
  } catch (error) {
    console.error(
      "Unable to clear refresh-token cookie during authentication:",
      error,
    );
  }
}

/*
 * =========================================
 * Normalize Roles
 * =========================================
 */

function normalizeRoles(roles) {
  if (!Array.isArray(roles)) {
    return [];
  }

  return [
    ...new Set(
      roles
        .map((role) =>
          typeof role === "string" ? role.trim().toLowerCase() : "",
        )
        .filter(Boolean),
    ),
  ];
}

/*
 * =========================================
 * Create Request Authentication
 * =========================================
 */

function createRequestAuthentication({ user, roles, session }) {
  if (!user?.id || !session?.id) {
    throw createAuthenticationError({
      message:
        "The resolved session does not contain a valid user and session.",

      code: "AUTHENTICATION_CONTEXT_INVALID",

      status: 500,

      publicMessage: "Your session could not be verified.",
    });
  }

  const normalizedRoles = Object.freeze(normalizeRoles(roles));

  return Object.freeze({
    isAuthenticated: true,

    userId: user.id,

    sessionId: session.id,

    user,

    roles: normalizedRoles,

    session,
  });
}

/*
 * =========================================
 * Require Authentication
 * =========================================
 *
 * Resolves the HttpOnly refresh-token cookie,
 * validates the database session and places
 * public-safe identity information on:
 *
 * request.auth
 *
 * This middleware validates the session but
 * does not rotate its refresh token.
 */

export async function requireAuthentication(request, response, next) {
  /*
   * Protected responses must not be cached
   * or shared between authenticated users.
   */

  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",

    Pragma: "no-cache",

    Expires: "0",
  });

  response.vary("Cookie");

  try {
    const refreshToken = getRefreshTokenFromRequest(request);

    if (!refreshToken) {
      throw createAuthenticationError({
        message: "The request does not contain a refresh-token cookie.",

        code: "AUTHENTICATION_REQUIRED",

        status: 401,

        publicMessage: "You must sign in to access this resource.",
      });
    }

    const authentication = await getAuthenticatedSession({
      refreshToken,
    });

    request.auth = createRequestAuthentication(authentication);

    return next();
  } catch (error) {
    if (shouldClearSessionCookie(error)) {
      safelyClearRefreshTokenCookie(response);
    }

    return next(error);
  }
}

import { getDatabasePool, sql } from "../config/database.js";

const PROJECT_COLUMNS = `
  id,
  user_id,
  title,
  category,
  lifecycle_status,
  record_status,
  is_featured,
  project_data,
  created_at,
  updated_at,
  row_version
`;

function mapProjectRow(row) {
  if (!row) {
    return null;
  }

  const projectData =
    typeof row.project_data === "string"
      ? JSON.parse(row.project_data)
      : row.project_data;

  return {
    ...projectData,
    id: row.id,
    title: row.title,
    category: row.category,
    lifecycleStatus: row.lifecycle_status,
    recordStatus: row.record_status,
    presentation: {
      ...(projectData.presentation || {}),
      isFeatured: Boolean(row.is_featured),
    },
    createdAt: row.created_at,
    updatedAt: row.updated_at,
    rowVersion: Buffer.from(row.row_version).toString("base64"),
  };
}

function bindProjectParameters(request, userId, project) {
  return request
    .input("id", sql.NVarChar(200), project.id)
    .input("userId", sql.UniqueIdentifier, userId)
    .input("title", sql.NVarChar(500), project.title)
    .input("category", sql.NVarChar(100), project.category)
    .input("lifecycleStatus", sql.NVarChar(50), project.lifecycleStatus)
    .input("recordStatus", sql.NVarChar(20), project.recordStatus)
    .input("isFeatured", sql.Bit, Boolean(project.presentation?.isFeatured))
    .input("projectData", sql.NVarChar(sql.MAX), JSON.stringify(project));
}

export async function listProjectsByUser(userId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
      ORDER BY updated_at DESC, id ASC;
    `);

  return result.recordset.map(mapProjectRow);
}

export async function findProjectById(userId, projectId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
        AND id = @id;
    `);

  return mapProjectRow(result.recordset[0]);
}

export async function createProject(userId, project) {
  const pool = await getDatabasePool();

  await bindProjectParameters(pool.request(), userId, project).query(`
    INSERT INTO dbo.projects (
      id,
      user_id,
      title,
      category,
      lifecycle_status,
      record_status,
      is_featured,
      project_data
    )
    VALUES (
      @id,
      @userId,
      @title,
      @category,
      @lifecycleStatus,
      @recordStatus,
      @isFeatured,
      @projectData
    );
  `);

  return findProjectById(userId, project.id);
}

export async function updateProject(userId, project, expectedRowVersion) {
  const pool = await getDatabasePool();

  const request = bindProjectParameters(pool.request(), userId, project).input(
    "expectedRowVersion",
    sql.Binary(8),
    Buffer.from(expectedRowVersion, "base64"),
  );

  const result = await request.query(`
    UPDATE dbo.projects
    SET
      title = @title,
      category = @category,
      lifecycle_status = @lifecycleStatus,
      record_status = @recordStatus,
      is_featured = @isFeatured,
      project_data = @projectData,
      updated_at = SYSUTCDATETIME()
    WHERE id = @id
      AND user_id = @userId
      AND row_version = @expectedRowVersion;
  `);

  if (result.rowsAffected[0] === 0) {
    return null;
  }

  return findProjectById(userId, project.id);
}

export async function deleteProject(userId, projectId, expectedRowVersion) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId)
    .input(
      "expectedRowVersion",
      sql.Binary(8),
      Buffer.from(expectedRowVersion, "base64"),
    ).query(`
      DELETE FROM dbo.projects
      WHERE id = @id
        AND user_id = @userId
        AND row_version = @expectedRowVersion;
    `);

  return result.rowsAffected[0] > 0;
}

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import { getEnvironment } from "../../config/environment.js";

const environment = getEnvironment();

/*
 * =========================================
 * User Storage Identity
 * =========================================
 */

export function getProfilePictureUserDirectory(userId) {
  const normalizedUserId = String(userId || "").trim();

  if (!normalizedUserId) {
    throw new Error(
      "An authenticated user ID is required for profile-picture storage.",
    );
  }

  return crypto.createHash("sha256").update(normalizedUserId).digest("hex");
}

/*
 * =========================================
 * Storage Root
 * =========================================
 */

export function getProfilePictureStorageRoot() {
  const storageRoot = String(
    environment.PROFILE_PICTURE_STORAGE_PATH || "",
  ).trim();

  if (!storageRoot) {
    throw new Error("PROFILE_PICTURE_STORAGE_PATH is not configured.");
  }

  return path.resolve(storageRoot);
}

/*
 * =========================================
 * User Storage Path
 * =========================================
 */

export function getProfilePictureUserStoragePath(userId) {
  return path.join(
    getProfilePictureStorageRoot(),
    getProfilePictureUserDirectory(userId),
  );
}

/*
 * =========================================
 * Storage Key
 * =========================================
 */

export function createProfilePictureStorageKey(userId, fileName) {
  const safeFileName = path.basename(String(fileName || "").trim());

  if (!safeFileName) {
    throw new Error("A profile-picture filename is required.");
  }

  return [getProfilePictureUserDirectory(userId), safeFileName].join("/");
}

export function resolveOwnedProfilePicturePath(userId, storageKey) {
  const normalizedStorageKey = String(storageKey || "")
    .trim()
    .replaceAll("\\", "/");

  if (!normalizedStorageKey) {
    throw new Error("A profile-picture storage key is required.");
  }

  const userDirectory = getProfilePictureUserDirectory(userId);

  const expectedPrefix = `${userDirectory}/`;

  if (!normalizedStorageKey.startsWith(expectedPrefix)) {
    throw new Error(
      "The profile-picture storage key does not belong to the authenticated user.",
    );
  }

  const relativeFileName = normalizedStorageKey.slice(expectedPrefix.length);

  /*
   * A profile picture must be one file directly inside
   * the authenticated user's storage directory.
   */

  if (
    !relativeFileName ||
    relativeFileName.includes("/") ||
    relativeFileName === "." ||
    relativeFileName === ".." ||
    path.basename(relativeFileName) !== relativeFileName
  ) {
    throw new Error("The profile-picture storage key is invalid.");
  }

  const userStoragePath = path.resolve(
    getProfilePictureUserStoragePath(userId),
  );

  const resolvedFilePath = path.resolve(userStoragePath, relativeFileName);

  /*
   * Defense in depth:
   * even after validating the storage key above, verify that
   * the resulting file remains directly inside this user's
   * storage directory.
   */

  if (path.dirname(resolvedFilePath) !== userStoragePath) {
    throw new Error(
      "The profile-picture path is outside the authenticated user's storage directory.",
    );
  }

  return resolvedFilePath;
}

export async function deleteOwnedProfilePicture(userId, storageKey) {
  const filePath = resolveOwnedProfilePicturePath(userId, storageKey);

  try {
    await fs.unlink(filePath);

    return {
      deleted: true,
      storageKey,
    };
  } catch (error) {
    /*
     * If the file is already gone, cleanup has still reached
     * the desired state. Treat deletion as idempotent.
     */

    if (error?.code === "ENOENT") {
      return {
        deleted: false,
        storageKey,
      };
    }

    throw error;
  }
}
