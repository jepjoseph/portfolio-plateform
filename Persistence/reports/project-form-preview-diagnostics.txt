  392:         ) {
  393:           transaction.abort();
  394: 
  395:           reject(
  396:             createProjectAssetStorageError({
  397:               code: "PROJECT_ASSET_STORAGE_KEY_CONFLICT",
  398: 
  399:               message:
  400:                 "The requested asset storage key belongs to another Project.",
  401: 
  402:               publicMessage:
  403:                 "The Project file could not be saved because its storage identifier conflicts with another file.",
  404:             }),
  405:           );
  406: 
  407:           return;
  408:         }
  409: 
  410:         const blob = file.slice(
  411:           0,
  412:           file.size,
  413:           file.type || "application/octet-stream",
  414:         );
  415: 
  416:         const record = {
  417:           storageKey: resolvedStorageKey,
  418: 
  419:           projectId: normalizedProjectId,
  420: 
  421:           assetId: resolvedAssetId,
  422: 
  423:           kind: normalizedKind,
  424: 
  425:           fileName: getText(file.name),
  426: 
  427:           mimeType: getText(file.type) || "application/octet-stream",
  428: 
  429:           fileSize: Number(file.size) || 0,
  430: 
  431:           blob,
  432: 
  433:           createdAt: existingRecord?.createdAt || timestamp,
  434: 
  435:           updatedAt: timestamp,
  436:         };
  437: 
  438:         const putRequest = store.put(record);
  439: 
  440:         putRequest.onerror = () => {
  441:           reject(putRequest.error);
  442:         };
  443: 
  444:         putRequest.onsuccess = () => {
  445:           resolve(record);
  446:         };
  447:       };
  448:     });
  449: 
  450:     const [record] = await Promise.all([
  451:       preparedRecordPromise,
  452:       transactionToPromise(transaction),
  453:     ]);
  454: 
  455:     return createAssetMetadata(record);
  456:   } catch (error) {
  457:     if (error?.name === "ProjectAssetStorageError") {
  458:       throw error;
  459:     }
  460: 
  461:     const quotaExceeded = isStorageQuotaError(error);
  462: 
  463:     throw createProjectAssetStorageError({
  464:       code: quotaExceeded
  465:         ? "PROJECT_ASSET_QUOTA_EXCEEDED"
  466:         : "PROJECT_ASSET_WRITE_FAILED",
  467: 
  468:       message: quotaExceeded
  469:         ? "IndexedDB storage quota was exceeded."
  470:         : "Unable to save a Project asset.",
  471: 
  472:       publicMessage: quotaExceeded
  473:         ? "The browser does not have enough storage space for this Project file. Remove unused files or choose a smaller file."
  474:         : "The selected Project file could not be saved.",
  475: 
  476:       cause: error,
  477:     });
  478:   } finally {
  479:     database.close();
  480:   }
  481: }

  659:  * =========================================
  660:  */
  661: 
  662: export async function deleteProjectAssets(storageKeys = []) {
  663:   const normalizedStorageKeys = [
  664:     ...new Set(getArray(storageKeys).map(getText).filter(Boolean)),
  665:   ];
  666: 
  667:   if (normalizedStorageKeys.length === 0) {
  668:     return 0;
  669:   }

  682:       store.delete(normalizedStorageKey);

  692:       message: "Unable to delete Project assets.",
  693: 
  694:       publicMessage: "One or more Project files could not be removed.",
  695: 
  696:       cause: error,
  697:     });
  698:   } finally {
  699:     database.close();
  700:   }
  701: }

  703: /*
  704:  * =========================================
  705:  * Delete All Assets for Project
  706:  * =========================================
  707:  *
  708:  * A cursor is used instead of getAll(). This
  709:  * prevents every large Blob from being loaded
  710:  * into application memory before deletion.
  711:  */
  712: 
  713: export async function deleteProjectAssetsForProject(projectId) {
  714:   const normalizedProjectId = getText(projectId);
  715: 
  716:   if (!normalizedProjectId) {
  717:     return 0;
  718:   }
  719: 
  720:   const database = await openProjectAssetDatabase();
  721: 
  722:   try {
  723:     const transaction = database.transaction(
  724:       PROJECT_ASSET_STORE_NAME,
  725:       "readwrite",
  726:     );
  727: 
  728:     const store = transaction.objectStore(PROJECT_ASSET_STORE_NAME);
  729: 
  730:     const index = store.index(PROJECT_ID_INDEX_NAME);
  731: 
  732:     if (typeof IDBKeyRange === "undefined") {
  733:       throw new Error("IDBKeyRange is unavailable.");
  734:     }
  735: 
  736:     const keyRange = IDBKeyRange.only(normalizedProjectId);
  737: 
  738:     /*
  739:      * This variable is updated while the cursor
  740:      * deletes each matching Project asset.
  741:      */
  742: 
  743:     let cursorDeletedCount = 0;
  744: 
  745:     const cursorPromise = new Promise((resolve, reject) => {
  746:       const cursorRequest = index.openCursor(keyRange);
  747: 
  748:       cursorRequest.onerror = () => {
  749:         reject(
  750:           cursorRequest.error || new Error("The Project asset cursor failed."),
  751:         );
  752:       };
  753: 
  754:       cursorRequest.onsuccess = () => {
  755:         const cursor = cursorRequest.result;
  756: 
  757:         /*
  758:          * No cursor means every matching asset
  759:          * has been processed.
  760:          */
  761: 
  762:         if (!cursor) {
  763:           resolve(cursorDeletedCount);
  764: 
  765:           return;
  766:         }
  767: 
  768:         const deleteRequest = cursor.delete();
  769: 
  770:         deleteRequest.onerror = () => {
  771:           reject(
  772:             deleteRequest.error ||
  773:               new Error("A Project asset could not be deleted."),
  774:           );
  775:         };
  776: 
  777:         deleteRequest.onsuccess = () => {
  778:           cursorDeletedCount += 1;
  779: 
  780:           cursor.continue();
  781:         };
  782:       };
  783:     });
  784: 
  785:     /*
  786:      * Use a different variable name for the
  787:      * resolved count to avoid redeclaration.
  788:      */
  789: 
  790:     const [completedDeletedCount] = await Promise.all([
  791:       cursorPromise,
  792:       transactionToPromise(transaction),
  793:     ]);
  794: 
  795:     return completedDeletedCount;
  796:   } catch (error) {
  797:     throw createProjectAssetStorageError({
  798:       code: "PROJECT_ASSET_PROJECT_DELETE_FAILED",
  799: 
  800:       message: "Unable to delete Project assets.",
  801: 
  802:       publicMessage: "One or more Project files could not be removed.",
  803: 
  804:       cause: error,
  805:     });
  806:   } finally {
  807:     database.close();
  808:   }
  809: }
