 145:     relative.startsWith(`..${path.sep}`) ||
 146:     path.isAbsolute(relative)
 147:   ) {
 148:     throw new Error("Certification document directory is outside the storage root.");
 149:   }
 150: }

 152: export async function deleteOwnedCertificationDocument(userId, certificationId, storageKey) {
 153:   const filePath = resolveOwnedCertificationDocumentPath(userId, certificationId, storageKey);
 154: 
 155:   try {
 156:     await assertSafeCertificationDocumentDirectory(userId, certificationId);
 157: 
 158:     const stats = await fs.lstat(filePath);
 159: 
 160:     if (stats.isSymbolicLink() || !stats.isFile()) {
 161:       throw new Error("Certification document target is not a regular file.");
 162:     }
 163: 
 164:     await fs.unlink(filePath);
 165: 
 166:     return {
 167:       deleted: true,
 168:       storageKey,
 169:     };
 170:   } catch (error) {
 171:     if (error?.code === "ENOENT") {
 172:       return {
 173:         deleted: false,
 174:         storageKey,
 175:       };
 176:     }
 177: 
 178:     throw error;
 179:   }
 180: }
