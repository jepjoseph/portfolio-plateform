  147: function openProjectAssetDatabase() {
  148:   return new Promise((resolve, reject) => {
  149:     const request = getIndexedDb().open(
  150:       PROJECT_ASSET_DATABASE_NAME,
  151:       PROJECT_ASSET_DATABASE_VERSION,
  152:     );
  153: 
  154:     request.onupgradeneeded = () => {
  155:       const database = request.result;
  156: 
  157:       let store;
  158: 
  159:       if (!database.objectStoreNames.contains(PROJECT_ASSET_STORE_NAME)) {
  160:         store = database.createObjectStore(PROJECT_ASSET_STORE_NAME, {
  161:           keyPath: "storageKey",
  162:         });
  163:       } else {
  164:         store = request.transaction.objectStore(PROJECT_ASSET_STORE_NAME);
  165:       }
  166: 
  167:       if (!store.indexNames.contains(PROJECT_ID_INDEX_NAME)) {
  168:         store.createIndex(PROJECT_ID_INDEX_NAME, "projectId", {
  169:           unique: false,
  170:         });
  171:       }
  172: 
  173:       if (!store.indexNames.contains(ASSET_KIND_INDEX_NAME)) {
  174:         store.createIndex(ASSET_KIND_INDEX_NAME, "kind", {
  175:           unique: false,
  176:         });
  177:       }
  178:     };
  179: 
  180:     request.onsuccess = () => {
  181:       const database = request.result;
  182: 
  183:       /*
  184:        * A tab holding an old database connection
  185:        * must close it when another tab upgrades
  186:        * the database.
  187:        */
  188: 
  189:       database.onversionchange = () => {
  190:         database.close();
  191:       };
  192: 
  193:       resolve(database);
  194:     };
  195: 
  196:     request.onerror = () => {
  197:       reject(
  198:         createProjectAssetStorageError({
  199:           code: "PROJECT_ASSET_DATABASE_OPEN_FAILED",
  200: 
  201:           message: "Unable to open Project asset storage.",
  202: 
  203:           publicMessage: "Project file storage could not be opened.",
  204: 
  205:           cause: request.error,
  206:         }),
  207:       );
  208:     };
  209: 
  210:     request.onblocked = () => {
  211:       reject(
  212:         createProjectAssetStorageError({
  213:           code: "PROJECT_ASSET_DATABASE_BLOCKED",
  214: 
  215:           message: "The Project asset database operation was blocked.",
  216: 
  217:           publicMessage: "Close other tabs using this portfolio and try again.",
  218:         }),
  219:       );
  220:     };
  221:   });

  325: 
  326: export async function writeProjectAsset({
  327:   file,
  328:   projectId,
  329:   assetId = "",
  330:   storageKey = "",
  331:   kind = "media",
  332: } = {}) {
  333:   const normalizedKind = normalizeAssetKind(kind);
  334: 
  335:   const normalizedProjectId = getText(projectId);
  336: 
  337:   if (!normalizedProjectId) {
  338:     throw createProjectAssetStorageError({
  339:       code: "PROJECT_ASSET_PROJECT_ID_REQUIRED",
  340: 
  341:       message: "A Project identifier is required before storing an asset.",
  342: 
  343:       publicMessage:
  344:         "The Project file cannot be saved because the Project could not be identified.",
  345:     });
  346:   }
  347: 
  348:   validateProjectAssetFile(file, normalizedKind);
  349: 
  350:   const resolvedAssetId = getText(assetId) || createProjectId("project-asset");
  351: 
  352:   const resolvedStorageKey =
  353:     getText(storageKey) ||
  354:     `project:${normalizedProjectId}:${normalizedKind}:${resolvedAssetId}`;
  355: 
  356:   const database = await openProjectAssetDatabase();
  357: 
  358:   const timestamp = new Date().toISOString();
  359: 
  360:   try {
  361:     const transaction = database.transaction(
  362:       PROJECT_ASSET_STORE_NAME,
  363:       "readwrite",
  364:     );
  365: 
  366:     const store = transaction.objectStore(PROJECT_ASSET_STORE_NAME);
  367: 
  368:     /*
  369:      * The read and write are handled within the
  370:      * same request callback so the transaction
  371:      * cannot become inactive between operations.
  372:      */
  373: 
  374:     const existingRequest = store.get(resolvedStorageKey);
  375: 
  376:     const preparedRecordPromise = new Promise((resolve, reject) => {
  377:       existingRequest.onerror = () => {
  378:         reject(existingRequest.error);
  379:       };
  380: 
  381:       existingRequest.onsuccess = () => {
  382:         const existingRecord = existingRequest.result;
  383: 
  384:         /*
  385:          * Never allow a storage key belonging
  386:          * to another Project to be overwritten.
  387:          */
  388: 
  389:         if (
  390:           existingRecord &&
  391:           existingRecord.projectId !== normalizedProjectId
  392:         ) {
  393:           transaction.abort();
  394: 
  395:           reject(
  396:             createProjectAssetStorageError({
  397:               code: "PROJECT_ASSET_STORAGE_KEY_CONFLICT",
  398: 
  399:               message:
  400:                 "The requested asset storage key belongs to another Project.",
  401: 
  402:               publicMessage:
  403:                 "The Project file could not be saved because its storage identifier conflicts with another file.",
  404:             }),
  405:           );
  406: 
  407:           return;
  408:         }
  409: 
  410:         const blob = file.slice(
  411:           0,
  412:           file.size,
  413:           file.type || "application/octet-stream",
  414:         );
  415: 
  416:         const record = {
  417:           storageKey: resolvedStorageKey,
  418: 
  419:           projectId: normalizedProjectId,
  420: 
  421:           assetId: resolvedAssetId,
  422: 
  423:           kind: normalizedKind,
  424: 
  425:           fileName: getText(file.name),
  426: 
  427:           mimeType: getText(file.type) || "application/octet-stream",
  428: 
  429:           fileSize: Number(file.size) || 0,
  430: 
  431:           blob,
  432: 
  433:           createdAt: existingRecord?.createdAt || timestamp,
  434: 
  435:           updatedAt: timestamp,
  436:         };
  437: 
  438:         const putRequest = store.put(record);
  439: 
  440:         putRequest.onerror = () => {
  441:           reject(putRequest.error);
  442:         };
  443: 
  444:         putRequest.onsuccess = () => {
  445:           resolve(record);
  446:         };
  447:       };
  448:     });
  449: 
  450:     const [record] = await Promise.all([
  451:       preparedRecordPromise,
  452:       transactionToPromise(transaction),
  453:     ]);
  454: 
  455:     return createAssetMetadata(record);
  456:   } catch (error) {
  457:     if (error?.name === "ProjectAssetStorageError") {
  458:       throw error;
  459:     }
  460: 
  461:     const quotaExceeded = isStorageQuotaError(error);
  462: 
  463:     throw createProjectAssetStorageError({
  464:       code: quotaExceeded
  465:         ? "PROJECT_ASSET_QUOTA_EXCEEDED"
  466:         : "PROJECT_ASSET_WRITE_FAILED",
  467: 
  468:       message: quotaExceeded
  469:         ? "IndexedDB storage quota was exceeded."
  470:         : "Unable to save a Project asset.",
  471: 
  472:       publicMessage: quotaExceeded
  473:         ? "The browser does not have enough storage space for this Project file. Remove unused files or choose a smaller file."
  474:         : "The selected Project file could not be saved.",
  475: 
  476:       cause: error,
  477:     });
  478:   } finally {
  479:     database.close();
  480:   }
  481: }

  661: 
  662: export async function deleteProjectAssets(storageKeys = []) {
  663:   const normalizedStorageKeys = [
  664:     ...new Set(getArray(storageKeys).map(getText).filter(Boolean)),
  665:   ];
  666: 
  667:   if (normalizedStorageKeys.length === 0) {
  668:     return 0;
  669:   }
  670: 
  671:   const database = await openProjectAssetDatabase();
  672: 
  673:   try {
  674:     const transaction = database.transaction(
  675:       PROJECT_ASSET_STORE_NAME,
  676:       "readwrite",
  677:     );
  678: 
  679:     const store = transaction.objectStore(PROJECT_ASSET_STORE_NAME);
  680: 
  681:     normalizedStorageKeys.forEach((normalizedStorageKey) => {
  682:       store.delete(normalizedStorageKey);
  683:     });
  684: 
  685:     await transactionToPromise(transaction);
  686: 
  687:     return normalizedStorageKeys.length;
  688:   } catch (error) {
  689:     throw createProjectAssetStorageError({
  690:       code: "PROJECT_ASSET_DELETE_FAILED",
  691: 
  692:       message: "Unable to delete Project assets.",
  693: 
  694:       publicMessage: "One or more Project files could not be removed.",
  695: 
  696:       cause: error,
  697:     });
  698:   } finally {
  699:     database.close();
  700:   }
  701: }

  703: /*
  704:  * =========================================
  705:  * Delete All Assets for Project
  706:  * =========================================
  707:  *
  708:  * A cursor is used instead of getAll(). This
  709:  * prevents every large Blob from being loaded
  710:  * into application memory before deletion.
  711:  */
  712: 
  713: export async function deleteProjectAssetsForProject(projectId) {
  714:   const normalizedProjectId = getText(projectId);
  715: 
  716:   if (!normalizedProjectId) {
  717:     return 0;
  718:   }
  719: 
  720:   const database = await openProjectAssetDatabase();
  721: 
  722:   try {
  723:     const transaction = database.transaction(
  724:       PROJECT_ASSET_STORE_NAME,
  725:       "readwrite",
  726:     );
  727: 
  728:     const store = transaction.objectStore(PROJECT_ASSET_STORE_NAME);
  729: 
  730:     const index = store.index(PROJECT_ID_INDEX_NAME);
  731: 
  732:     if (typeof IDBKeyRange === "undefined") {
  733:       throw new Error("IDBKeyRange is unavailable.");
  734:     }
  735: 
  736:     const keyRange = IDBKeyRange.only(normalizedProjectId);
  737: 
  738:     /*
  739:      * This variable is updated while the cursor
  740:      * deletes each matching Project asset.
  741:      */
  742: 
  743:     let cursorDeletedCount = 0;
  744: 
  745:     const cursorPromise = new Promise((resolve, reject) => {
  746:       const cursorRequest = index.openCursor(keyRange);
  747: 
  748:       cursorRequest.onerror = () => {
  749:         reject(
  750:           cursorRequest.error || new Error("The Project asset cursor failed."),
  751:         );
  752:       };
  753: 
  754:       cursorRequest.onsuccess = () => {
  755:         const cursor = cursorRequest.result;
  756: 
  757:         /*
  758:          * No cursor means every matching asset
  759:          * has been processed.
  760:          */
  761: 
  762:         if (!cursor) {
  763:           resolve(cursorDeletedCount);
  764: 
  765:           return;
  766:         }
  767: 
  768:         const deleteRequest = cursor.delete();
  769: 
  770:         deleteRequest.onerror = () => {
  771:           reject(

  186:   const uploadCollection = getArray(uploads);
  187: 
  188:   const metadata = [];
  189: 
  190:   const createdStorageKeys = [];
  191: 
  192:   try {
  193:     for (let index = 0; index < uploadCollection.length; index += 1) {
  194:       const upload = uploadCollection[index];
  195: 
  196:       if (!(upload?.file instanceof Blob)) {
  197:         throw createProjectServiceError(
  198:           "PROJECT_UPLOAD_FILE_INVALID",
  199:           `Project ${kind} upload does not contain a valid Blob.`,
  200:           `One of the selected Project ${
  201:             kind === "media" ? "media files" : "documents"
  202:           } is invalid.`,
  203:         );
  204:       }
  205: 
  206:       const storedAsset = await writeProjectAsset({
  207:         file: upload.file,
  208: 
  209:         projectId,
  210: 
  211:         assetId: upload.assetId || upload.id,
  212: 
  213:         storageKey: upload.storageKey,
  214: 
  215:         kind,
  216:       });
  217: 
  218:       createdStorageKeys.push(storedAsset.storageKey);
  219: 
  220:       const metadataValues = {
  221:         ...upload,
  222: 
  223:         /*
  224:          * File objects must never be included
  225:          * in localStorage metadata.
  226:          */
  227: 
  228:         file: undefined,
  229: 
  230:         id: storedAsset.assetId,
  231: 
  232:         fileName: storedAsset.fileName,
  233: 
  234:         mimeType: storedAsset.mimeType,
  235: 
  236:         fileSize: storedAsset.fileSize,
  237: 
  238:         storageProvider: "indexed-db",
  239: 
  240:         storageKey: storedAsset.storageKey,
  241: 
  242:         uploadedAt: storedAsset.updatedAt,
  243:       };
  244: 
  245:       metadata.push(
  246:         kind === "media"
  247:           ? createProjectMedia(metadataValues, index)
  248:           : createProjectDocument(metadataValues, index),
  249:       );
  250:     }
  251: 
  252:     return {
  253:       metadata,
  254:       createdStorageKeys,
  255:     };
  256:   } catch (error) {
  257:     /*
  258:      * If one upload fails, remove every asset
  259:      * already created during this collection.
  260:      */
  261: 
  262:     await deleteProjectAssets(createdStorageKeys).catch(() => {});
  263: 
  264:     throw error;
  265:   }
  266: }

  268: /*
  269:  * =========================================
  270:  * Prepare Project Assets
  271:  * =========================================
  272:  */
  273: 
  274: async function prepareProjectAssets(
  275:   project,
  276:   { mediaUploads = [], documentUploads = [] } = {},
  277: ) {
  278:   const storedMedia = await storeProjectUploads(
  279:     project.id,
  280:     mediaUploads,
  281:     "media",
  282:   );
  283: 
  284:   try {
  285:     const storedDocuments = await storeProjectUploads(
  286:       project.id,
  287:       documentUploads,
  288:       "document",
  289:     );
  290: 
  291:     /*
  292:      * A staged upload selected in the Media
  293:      * Editor must take priority. Before upload,
  294:      * its ID cannot be resolved by normalizeProject
  295:      * because it is not yet part of project.media.
  296:      */
  297: 
  298:     const featuredMediaResult = prepareFeaturedProjectMedia(
  299:       getArray(project.media),
  300:       storedMedia.metadata,
  301:       getText(project.presentation?.featuredMediaId),
  302:     );
  303:     const preparedProject = normalizeProject({
  304:       ...project,
  305: 
  306:       media: featuredMediaResult.media,
  307: 
  308:       supportingDocuments: [
  309:         ...getArray(project.supportingDocuments),
  310: 
  311:         ...storedDocuments.metadata,
  312:       ],
  313: 
  314:       presentation: {
  315:         ...project.presentation,
  316: 
  317:         featuredMediaId: featuredMediaResult.featuredMediaId,
  318:       },
  319:     });
  320: 
  321:     return {
  322:       project: preparedProject,
  323: 
  324:       createdStorageKeys: [
  325:         ...storedMedia.createdStorageKeys,
  326:         ...storedDocuments.createdStorageKeys,
  327:       ],
  328:     };
  329:   } catch (error) {
  330:     /*
  331:      * Document storage failed after media was
  332:      * saved, so the new media must be rolled
  333:      * back as well.
  334:      */
  335: 
  336:     await deleteProjectAssets(storedMedia.createdStorageKeys).catch(() => {});
  337: 
  338:     throw error;
  339:   }
  340: }

  342: /*
  343:  * =========================================
  344:  * Safe Removed-Asset Cleanup
  345:  * =========================================
  346:  */
  347: 
  348: async function removeUnreferencedAssets(project, requestedStorageKeys) {
  349:   const referencedStorageKeys = new Set([
  350:     ...getArray(project.media)
  351:       .map((mediaItem) => getText(mediaItem.storageKey))
  352:       .filter(Boolean),
  353: 
  354:     ...getArray(project.supportingDocuments)
  355:       .map((documentRecord) => getText(documentRecord.storageKey))
  356:       .filter(Boolean),
  357:   ]);
  358: 
  359:   const removableStorageKeys = [
  360:     ...new Set(getArray(requestedStorageKeys).map(getText).filter(Boolean)),
  361:   ].filter((storageKey) => !referencedStorageKeys.has(storageKey));
  362: 
  363:   if (removableStorageKeys.length === 0) {
  364:     return 0;
  365:   }
  366: 
  367:   return deleteProjectAssets(removableStorageKeys);
  368: }

  430: /*
  431:  * =========================================
  432:  * Create Project
  433:  * =========================================
  434:  */
  435: 
  436: export async function createProject(
  437:   projectData,
  438:   {
  439:     collections = {},
  440:     mediaUploads = [],
  441:     documentUploads = [],
  442:     removedAssetStorageKeys = [],
  443:   } = {},
  444: ) {
  445:   const records = readStoredProjects();
  446: 
  447:   let project = createProjectModel(projectData);
  448: 
  449:   ensureUniqueProject(records, project);
  450: 
  451:   const preparedAssets = await prepareProjectAssets(project, {
  452:     mediaUploads,
  453:     documentUploads,
  454:   });
  455: 
  456:   project = preparedAssets.project;
  457: 
  458:   try {
  459:     const validation = validateForSave(project, collections);
  460: 
  461:     /*
  462:      * Store metadata only after every staged
  463:      * asset was written successfully.
  464:      */
  465: 
  466:     replaceStoredProjects([...records, project]);
  467: 
  468:     /*
  469:      * This collection is normally empty while
  470:      * creating, but accepting it keeps the
  471:      * service contract consistent.
  472:      */
  473: 
  474:     await removeUnreferencedAssets(project, removedAssetStorageKeys).catch(
  475:       () => {},
  476:     );
  477: 
  478:     return {
  479:       project,
  480:       warnings: validation.warnings,
  481:     };
  482:   } catch (error) {
  483:     /*
  484:      * Metadata was not saved, so every new
  485:      * IndexedDB asset must be removed.
  486:      */
  487: 
  488:     await deleteProjectAssets(preparedAssets.createdStorageKeys).catch(
  489:       () => {},
  490:     );
  491: 
  492:     throw error;
  493:   }
  494: }

  496: /*
  497:  * =========================================
  498:  * Update Project
  499:  * =========================================
  500:  */
  501: 
  502: export async function updateProject(
  503:   projectId,
  504:   updates,
  505:   {
  506:     collections = {},
  507:     mediaUploads = [],
  508:     documentUploads = [],
  509:     removedAssetStorageKeys = [],
  510:   } = {},
  511: ) {
  512:   const normalizedProjectId = requireProjectId(projectId);
  513: 
  514:   const records = readStoredProjects();
  515: 
  516:   const projectIndex = findProjectIndex(records, normalizedProjectId);
  517: 
  518:   if (projectIndex < 0) {
  519:     throw createProjectServiceError(
  520:       "PROJECT_NOT_FOUND",
  521:       `Project not found: ${normalizedProjectId}`,
  522:       "The selected Project could not be found.",
  523:       {
  524:         projectId: normalizedProjectId,
  525:       },
  526:     );
  527:   }
  528: 
  529:   let project = updateProjectModel(records[projectIndex], updates);
  530: 
  531:   ensureUniqueProject(records, project, project.id);
  532: 
  533:   const preparedAssets = await prepareProjectAssets(project, {
  534:     mediaUploads,
  535:     documentUploads,
  536:   });
  537: 
  538:   project = preparedAssets.project;
  539: 
  540:   try {
  541:     const validation = validateForSave(project, collections);
  542: 
  543:     const nextRecords = [...records];
  544: 
  545:     nextRecords[projectIndex] = project;
  546: 
  547:     /*
  548:      * Save updated metadata before deleting
  549:      * old assets. If this write fails, existing
  550:      * files remain recoverable.
  551:      */
  552: 
  553:     replaceStoredProjects(nextRecords);
  554: 
  555:     /*
  556:      * Asset deletion occurs only after the new
  557:      * Project record has been saved.
  558:      */
  559: 
  560:     await removeUnreferencedAssets(project, removedAssetStorageKeys).catch(
  561:       () => {},
  562:     );
  563: 
  564:     return {
  565:       project,
  566:       warnings: validation.warnings,
  567:     };
  568:   } catch (error) {
  569:     /*
  570:      * Preserve all pre-existing files and remove
  571:      * only assets created during this attempt.
  572:      */
  573: 
  574:     await deleteProjectAssets(preparedAssets.createdStorageKeys).catch(
  575:       () => {},
  576:     );
  577: 
  578:     throw error;
  579:   }
  580: }

  583:  * =========================================
  584:  * Archive Project
  585:  * =========================================
  586:  */
  587: 
  588: export async function archiveProject(projectId) {
  589:   const normalizedProjectId = requireProjectId(projectId);
  590: 
  591:   const records = readStoredProjects();
  592: 
  593:   const projectIndex = findProjectIndex(records, normalizedProjectId);
  594: 
  595:   if (projectIndex < 0) {
  596:     throw createProjectServiceError(
  597:       "PROJECT_NOT_FOUND",
  598:       `Project not found: ${normalizedProjectId}`,
  599:       "The selected Project could not be found.",
  600:       {
  601:         projectId: normalizedProjectId,
  602:       },
  603:     );
  604:   }
  605: 
  606:   const project = records[projectIndex];
  607: 
  608:   if (project.recordStatus === "archived") {
  609:     return project;
  610:   }
  611: 
  612:   const archivedProject = normalizeProject({
  613:     ...project,
  614: 
  615:     recordStatus: "archived",
  616: 
  617:     updatedAt: new Date().toISOString(),
  618:   });
  619: 
  620:   const nextRecords = [...records];
  621: 
  622:   nextRecords[projectIndex] = archivedProject;
  623: 
  624:   replaceStoredProjects(nextRecords);
  625: 
  626:   return archivedProject;
  627: }

  635: export async function restoreProject(projectId) {
  636:   const normalizedProjectId = requireProjectId(projectId);
  637: 
  638:   const records = readStoredProjects();
  639: 
  640:   const projectIndex = findProjectIndex(records, normalizedProjectId);

  648:         projectId: normalizedProjectId,
  649:       },

  656:     return project;

  663: 
  664:     updatedAt: new Date().toISOString(),
  665:   });
  666: 
  667:   const nextRecords = [...records];
  668: 
  669:   nextRecords[projectIndex] = restoredProject;
  670: 
  671:   replaceStoredProjects(nextRecords);
  672: 
  673:   return restoredProject;

  676: /*
  677:  * =========================================
  678:  * Delete Project Permanently
  679:  * =========================================
  680:  */
  681: 
  682: export async function deleteProject(projectId) {
  683:   const normalizedProjectId = requireProjectId(projectId);
  684: 
  685:   const records = readStoredProjects();
  686: 
  687:   const projectIndex = findProjectIndex(records, normalizedProjectId);
  688: 
  689:   if (projectIndex < 0) {
  690:     throw createProjectServiceError(
  691:       "PROJECT_NOT_FOUND",
  692:       `Project not found: ${normalizedProjectId}`,
  693:       "The selected Project could not be found.",
  694:       {
  695:         projectId: normalizedProjectId,
  696:       },
  697:     );
  698:   }
  699: 
  700:   const deletedProject = records[projectIndex];
  701: 
  702:   const nextRecords = records.filter(
  703:     (project) => project.id !== normalizedProjectId,
  704:   );
  705: 
  706:   /*
  707:    * Delete metadata first. If localStorage
  708:    * fails, the Project and all assets remain.
  709:    */
  710: 
  711:   replaceStoredProjects(nextRecords);
  712: 
  713:   /*
  714:    * Once metadata is removed, delete every
  715:    * IndexedDB asset owned by this Project.
  716:    *
  717:    * Cleanup is best effort because the Project
  718:    * metadata has already been deleted.
  719:    */
  720: 
  721:   let assetCleanupError = null;
  722: 
  723:   try {
  724:     await deleteProjectAssetsForProject(normalizedProjectId);
  725:   } catch (error) {
  726:     assetCleanupError = error;
  727:   }
  728: 
  729:   return {
  730:     ...deletedProject,
  731: 
  732:     assetCleanupWarning: assetCleanupError
  733:       ? "The Project was deleted, but one or more locally stored files could not be cleaned up."
  734:       : "",
  735:   };
  736: }
