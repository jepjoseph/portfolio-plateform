 672:  * Effective Credential State

 676: export function getEffectiveCredentialState(certification, today = new Date()) {
 677:   const normalized = normalizeCertification(certification);
 678: 
 679:   const todayMonth = today.toISOString().slice(0, 7);
 680: 
 681:   if (
 682:     normalized.credential.state === "planned" ||
 683:     (normalized.dates.issueDate && normalized.dates.issueDate > todayMonth)
 684:   ) {
 685:     return "planned";
 686:   }
 687: 
 688:   if (
 689:     !normalized.dates.doesNotExpire &&
 690:     normalized.dates.expirationDate &&
 691:     normalized.dates.expirationDate < todayMonth
 692:   ) {
 693:     return "expired";
 694:   }
 695: 
 696:   return normalized.credential.state;
 697: }

  52:   publicMessage,

 123: function findMatchingCertification(
 124:   certifications,
 125:   candidate,
 126:   excludedCertificationId = "",
 127: ) {
 128:   const candidateName = normalizeComparisonText(candidate.name);
 129: 
 130:   const candidateIssuer = normalizeComparisonText(
 131:     candidate.issuingOrganization?.name,
 132:   );
 133: 
 134:   if (!candidateName || !candidateIssuer) {
 135:     return null;
 136:   }
 137: 
 138:   return (
 139:     certifications.find((certification) => {
 140:       if (certification.id === excludedCertificationId) {
 141:         return false;
 142:       }
 143: 
 144:       const certificationName = normalizeComparisonText(certification.name);
 145: 
 146:       const certificationIssuer = normalizeComparisonText(
 147:         certification.issuingOrganization?.name,
 148:       );
 149: 
 150:       return (
 151:         certificationName === candidateName &&
 152:         certificationIssuer === candidateIssuer
 153:       );
 154:     }) || null
 155:   );
 156: }

 158: function getDocumentStorageKeys(certification) {
 159:   return getArray(certification?.supportingDocuments)
 160:     .map((document) => getText(document?.storageKey))

 164: function getRemovedDocumentStorageKeys(
 165:   currentCertification,
 166:   nextCertification,
 167: ) {
 168:   const nextStorageKeys = new Set(getDocumentStorageKeys(nextCertification));
 169: 
 170:   return getDocumentStorageKeys(currentCertification).filter(
 171:     (storageKey) => !nextStorageKeys.has(storageKey),
 172:   );

 200:  * Document Cleanup
 201:  * =========================================
 202:  */
 203: 
 204: async function rollbackStoredDocuments(storageKeys = []) {
 205:   await Promise.allSettled(
 206:     storageKeys.map((storageKey) => deleteCertificationDocument(storageKey)),
 207:   );
 208: }
 209: 
 210: async function removeUnusedDocuments(storageKeys = []) {
 211:   if (storageKeys.length === 0) {
 212:     return;
 213:   }
 214: 
 215:   try {
 216:     await deleteCertificationDocuments(storageKeys);
 217:   } catch {
 218:     /*
 219:      * Certification metadata has already been saved.
 220:      *
 221:      * A failed binary cleanup should not make the
 222:      * interface report that the Certification save
 223:      * itself failed.
 224:      *
 225:      * Orphaned IndexedDB files can be removed later
 226:      * through a maintenance operation.
 227:      */
 228:   }
 229: }

 231: /*
 232:  * =========================================
 233:  * Store Uploaded Documents
 234:  * =========================================
 235:  */
 236: 
 237: async function storeDocumentUploads(certificationId, uploads = []) {
 238:   const documents = [];
 239:   const storageKeys = [];
 240: 
 241:   try {
 242:     for (const [index, upload] of getArray(uploads).entries()) {
 243:       if (!upload?.file) {
 244:         throw createCertificationServiceError(
 245:           "CERTIFICATION_DOCUMENT_FILE_REQUIRED",
 246:           `Document upload ${index} does not contain a file.`,
 247:           "Select a file for every certificate document.",
 248:         );
 249:       }
 250: 
 251:       const metadata = upload.metadata || {};
 252: 
 253:       const stored = await writeCertificationDocument({
 254:         file: upload.file,
 255: 
 256:         certificationId,
 257: 
 258:         documentId: upload.documentId || metadata.id,
 259: 
 260:         storageKey: upload.storageKey || metadata.storageKey,
 261:       });
 262: 
 263:       storageKeys.push(stored.storageKey);
 264: 
 265:       const document = createCertificationDocument(
 266:         {
 267:           ...metadata,
 268: 
 269:           id: stored.documentId,
 270: 
 271:           documentType:
 272:             upload.documentType || metadata.documentType || "certificate",
 273: 
 274:           name: upload.name || metadata.name || stored.fileName,
 275: 
 276:           description: upload.description || metadata.description || "",
 277: 
 278:           fileName: stored.fileName,
 279:           extension: stored.extension,
 280:           mimeType: stored.mimeType,
 281:           size: stored.size,
 282:           storageKey: stored.storageKey,
 283:           previewMode: stored.previewMode,
 284: 
 285:           isPrimary: upload.isPrimary ?? metadata.isPrimary ?? false,
 286: 
 287:           createdAt: stored.createdAt,
 288:           updatedAt: stored.updatedAt,
 289:           uploadedAt: stored.updatedAt,
 290:         },
 291:         index,
 292:       );
 293: 
 294:       documents.push(document);
 295:     }
 296: 
 297:     return {
 298:       documents,
 299:       storageKeys,
 300:     };
 301:   } catch (error) {
 302:     await rollbackStoredDocuments(storageKeys);
 303: 
 304:     throw error;
 305:   }
 306: }

 326: export function getCertificationById(
 327:   certificationId,
 328:   { includeArchived = true } = {},
 329: ) {

 331: 
 332:   if (!normalizedId) {

 337:     readStoredCertifications().find((record) => record.id === normalizedId) ||
 338:     null;
 339:

 341:     return null;
 342:   }
 343: 
 344:   return certification;
 345: }

 365: /*
 366:  * =========================================
 367:  * Create Certification
 368:  * =========================================
 369:  */
 370: 
 371: export async function createCertification(
 372:   certificationData,
 373:   {
 374:     documentUploads = [],
 375:     skills = null,
 376:     trainingRecords = null,
 377:     educationRecords = null,
 378:   } = {},
 379: ) {
 380:   const certifications = readStoredCertifications();
 381: 
 382:   let certification = createCertificationModel(certificationData);
 383: 
 384:   const stored = await storeDocumentUploads(certification.id, documentUploads);
 385: 
 386:   if (stored.documents.length > 0) {
 387:     certification = normalizeCertification({
 388:       ...certification,
 389: 
 390:       supportingDocuments: [
 391:         ...certification.supportingDocuments,
 392:         ...stored.documents,
 393:       ],
 394:     });
 395:   }
 396: 
 397:   try {
 398:     const validation = validateForSave(certification, {
 399:       skills,
 400:       trainingRecords,
 401:       educationRecords,
 402:     });
 403: 
 404:     const duplicate = findMatchingCertification(
 405:       certifications,
 406:       validation.certification,
 407:     );
 408: 
 409:     if (duplicate) {
 410:       throw createDuplicateError(duplicate);
 411:     }
 412: 
 413:     const transaction = saveCertificationWithTrainingSync({
 414:       certificationRecords: [...certifications, validation.certification],
 415: 
 416:       changedCertificationIds: [validation.certification.id],
 417:     });
 418: 
 419:     const savedCertification =
 420:       transaction.certificationRecords.find(
 421:         (record) => record.id === validation.certification.id,
 422:       ) || validation.certification;
 423: 
 424:     return {
 425:       certification: savedCertification,
 426:       warnings: validation.warnings,
 427:     };
 428:   } catch (error) {
 429:     await rollbackStoredDocuments(stored.storageKeys);
 430: 
 431:     throw error;
 432:   }
 433: }

 435: /*
 436:  * =========================================
 437:  * Update Certification
 438:  * =========================================
 439:  */
 440: 
 441: export async function updateCertification(
 442:   certificationId,
 443:   certificationData,
 444:   {
 445:     documentUploads = [],
 446:     skills = null,
 447:     trainingRecords = null,
 448:     educationRecords = null,
 449:   } = {},
 450: ) {
 451:   const normalizedId = getText(certificationId);
 452: 
 453:   const certifications = readStoredCertifications();
 454: 
 455:   const certificationIndex = findCertificationIndex(
 456:     certifications,
 457:     normalizedId,
 458:   );
 459: 
 460:   if (certificationIndex < 0) {
 461:     throw createNotFoundError(normalizedId);
 462:   }
 463: 
 464:   const currentCertification = certifications[certificationIndex];
 465: 
 466:   let nextCertification = updateCertificationModel(
 467:     currentCertification,
 468:     certificationData,
 469:   );
 470: 
 471:   const stored = await storeDocumentUploads(normalizedId, documentUploads);
 472: 
 473:   if (stored.documents.length > 0) {
 474:     nextCertification = normalizeCertification({
 475:       ...nextCertification,
 476: 
 477:       supportingDocuments: [
 478:         ...nextCertification.supportingDocuments,
 479:         ...stored.documents,
 480:       ],
 481:     });
 482:   }
 483: 
 484:   try {
 485:     const validation = validateForSave(nextCertification, {
 486:       skills,
 487:       trainingRecords,
 488:       educationRecords,
 489:     });
 490: 
 491:     const duplicate = findMatchingCertification(
 492:       certifications,
 493:       validation.certification,
 494:       normalizedId,
 495:     );
 496: 
 497:     if (duplicate) {
 498:       throw createDuplicateError(duplicate);
 499:     }
 500: 
 501:     const removedStorageKeys = getRemovedDocumentStorageKeys(
 502:       currentCertification,
 503:       validation.certification,
 504:     );
 505: 
 506:     const nextCertifications = [...certifications];
 507: 
 508:     nextCertifications[certificationIndex] = validation.certification;
 509: 
 510:     const transaction = saveCertificationWithTrainingSync({
 511:       certificationRecords: nextCertifications,
 512: 
 513:       changedCertificationIds: [validation.certification.id],
 514:     });
 515: 
 516:     await removeUnusedDocuments(removedStorageKeys);
 517: 
 518:     const savedCertification =
 519:       transaction.certificationRecords.find(
 520:         (record) => record.id === validation.certification.id,
 521:       ) || validation.certification;
 522: 
 523:     return {
 524:       certification: savedCertification,
 525:       warnings: validation.warnings,
 526:     };
 527:   } catch (error) {
 528:     await rollbackStoredDocuments(stored.storageKeys);
 529: 
 530:     throw error;
 531:   }
 532: }

 540: export async function archiveCertification(certificationId) {
 541:   const certification = getCertificationById(certificationId);
 542: 
 543:   if (!certification) {
 544:     throw createNotFoundError(getText(certificationId));
 545:   }
 546: 
 547:   if (certification.status === "archived") {
 548:     return certification;
 549:   }
 550: 
 551:   const certifications = readStoredCertifications();
 552: 
 553:   const certificationIndex = findCertificationIndex(
 554:     certifications,
 555:     certification.id,
 556:   );
 557: 
 558:   const archivedCertification = normalizeCertification({
 559:     ...certification,
 560: 
 561:     status: "archived",
 562: 
 563:     updatedAt: new Date().toISOString(),
 564:   });
 565: 
 566:   certifications[certificationIndex] = archivedCertification;
 567: 
 568:   const transaction = saveCertificationWithTrainingSync({
 569:     certificationRecords: certifications,
 570: 
 571:     changedCertificationIds: [archivedCertification.id],
 572:   });
 573: 
 574:   return (
 575:     transaction.certificationRecords.find(
 576:       (record) => record.id === archivedCertification.id,
 577:     ) || archivedCertification
 578:   );
 579: }

 582:  * =========================================
 583:  * Restore Certification
 584:  * =========================================
 585:  */
 586: 
 587: export async function restoreCertification(certificationId) {
 588:   const certification = getCertificationById(certificationId);
 589: 
 590:   if (!certification) {
 591:     throw createNotFoundError(getText(certificationId));
 592:   }
 593: 
 594:   if (certification.status !== "archived") {
 595:     return certification;
 596:   }
 597: 
 598:   const certifications = readStoredCertifications();
 599: 
 600:   const duplicate = findMatchingCertification(
 601:     certifications,
 602:     certification,
 603:     certification.id,
 604:   );
 605: 
 606:   if (duplicate && duplicate.status !== "archived") {
 607:     throw createDuplicateError(duplicate);
 608:   }
 609: 
 610:   const certificationIndex = findCertificationIndex(
 611:     certifications,
 612:     certification.id,
 613:   );
 614: 
 615:   const restoredCertification = normalizeCertification({
 616:     ...certification,
 617: 
 618:     status: "active",
 619: 
 620:     updatedAt: new Date().toISOString(),
 621:   });
 622: 
 623:   certifications[certificationIndex] = restoredCertification;
 624: 
 625:   const transaction = saveCertificationWithTrainingSync({
 626:     certificationRecords: certifications,
 627: 
 628:     changedCertificationIds: [restoredCertification.id],
 629:   });
 630: 
 631:   return (
 632:     transaction.certificationRecords.find(
 633:       (record) => record.id === restoredCertification.id,
 634:     ) || restoredCertification
 635:   );
 636: }

 640:  * Delete Certification
 641:  * =========================================
 642:  */
 643: 
 644: export async function deleteCertification(certificationId) {
 645:   const normalizedId = getText(certificationId);
 646: 
 647:   const certifications = readStoredCertifications();
 648: 
 649:   const certificationIndex = findCertificationIndex(
 650:     certifications,
 651:     normalizedId,
 652:   );
 653: 
 654:   if (certificationIndex < 0) {
 655:     throw createNotFoundError(normalizedId);
 656:   }
 657: 
 658:   const certification = certifications[certificationIndex];
 659: 
 660:   const nextCertifications = certifications.filter(
 661:     (record) => record.id !== normalizedId,
 662:   );
 663: 
 664:   /*
 665:    * Remove the metadata first.
 666:    *
 667:    * If IndexedDB cleanup fails, the deleted
 668:    * Certification does not reappear.
 669:    */
 670: 
 671:   saveCertificationWithTrainingSync({
 672:     certificationRecords: nextCertifications,
 673: 
 674:     deletedCertificationIds: [normalizedId],
 675:   });
 676: 
 677:   try {
 678:     await deleteCertificationDocumentsForCertification(normalizedId);
 679:   } catch {
 680:     await removeUnusedDocuments(getDocumentStorageKeys(certification));
 681:   }
 682: 
 683:   return certification;
 684: }

 829: 
 830: export async function replaceCertificationDocumentFile(
 831:   certificationId,
 832:   documentId,
 833:   file,
 834:   validationCollections = {},
 835: ) {
 836:   const certification = getCertificationById(certificationId);
 837: 
 838:   if (!certification) {
 839:     throw createNotFoundError(getText(certificationId));
 840:   }
 841: 
 842:   const normalizedDocumentId = getText(documentId);
 843: 
 844:   const currentDocument = certification.supportingDocuments.find(
 845:     (document) => document.id === normalizedDocumentId,
 846:   );
 847: 
 848:   if (!currentDocument) {
 849:     throw createCertificationServiceError(
 850:       "CERTIFICATION_DOCUMENT_NOT_FOUND",
 851:       `Certification document not found: ${normalizedDocumentId}`,
 852:       "The selected certificate document could not be found.",
 853:     );
 854:   }
 855: 
 856:   /*
 857:    * Preserve the existing document so it can be
 858:    * restored if metadata validation fails.
 859:    */
