 215: function prepareRequestBody({ body, headers }) {
 216:   if (body === undefined || body === null) {
 217:     return undefined;
 218:   }
 219: 
 220:   if (shouldSerializeAsJson(body)) {
 221:     if (!headers.has("Content-Type")) {
 222:       headers.set("Content-Type", "application/json");
 223:     }
 224: 
 225:     return JSON.stringify(body);
 226:   }
 227: 
 228:   /*
 229:    * Never set Content-Type manually for
 230:    * FormData. The browser must add its
 231:    * multipart boundary.
 232:    */
 233: 
 234:   return body;
 235: }
 236: 
 237: /*
 238:  * =========================================
 239:  * Response Parsing
 240:  * =========================================
 241:  */
 242: 
 243: async function parseResponseBody(response) {
 244:   if (response.status === 204 || response.status === 205) {
 245:     return null;
 246:   }
 247: 
 248:   const text = await response.text();
 249: 
 250:   if (!text) {
 251:     return null;
 252:   }
 253: 
 254:   const contentType = response.headers.get("content-type") || "";
 255: 
 256:   if (contentType.includes("application/json")) {
 257:     try {
 258:       return JSON.parse(text);
 259:     } catch {
 260:       return {
 261:         message: "The server returned an invalid JSON response.",
 262:       };
 263:     }
 264:   }
 265: 
 266:   /*
 267:    * Some development proxies may omit the
 268:    * JSON content type. Attempt JSON parsing
 269:    * before treating the body as plain text.
 270:    */
 271: 
 272:   try {
 273:     return JSON.parse(text);
 274:   } catch {
 275:     return {
 276:       message: text,
 277:     };
 278:   }
 279: }

 282:  * =========================================
 283:  * API Error Creation
 284:  * =========================================
 285:  */
 286: 
 287: function createResponseError({ response, data, method, url }) {
 288:   const error = new ApiError({
 289:     message: data?.message || `The server returned HTTP ${response.status}.`,
 290: 
 291:     status: response.status,
 292: 
 293:     code: data?.code || "API_REQUEST_FAILED",
 294: 
 295:     fieldErrors: data?.fieldErrors,
 296: 
 297:     errors: data?.errors,
 298: 
 299:     details: data?.details,
 300: 
 301:     method,
 302: 
 303:     url,
 304:   });
 305: 
 306:   notifyInvalidAuthenticationSession(error);
 307: 
 308:   return error;
 309: }

 311: /*
 312:  * =========================================
 313:  * Timeout and Abort Signal
 314:  * =========================================
 315:  */
 316: 
 317: function createRequestAbortController({ signal, timeoutMs }) {
 318:   const controller = new AbortController();
 319: 
 320:   let wasTimedOut = false;
 321: 
 322:   const handleExternalAbort = () => {
 323:     controller.abort(signal?.reason);
 324:   };
 325: 
 326:   if (signal) {
 327:     if (signal.aborted) {
 328:       handleExternalAbort();
 329:     } else {
 330:       signal.addEventListener("abort", handleExternalAbort, {
 331:         once: true,
 332:       });
 333:     }
 334:   }
 335: 
 336:   const timeoutId = globalThis.setTimeout(() => {
 337:     wasTimedOut = true;
 338: 
 339:     controller.abort();
 340:   }, timeoutMs);
 341: 
 342:   return {
 343:     signal: controller.signal,
 344: 
 345:     wasTimedOut() {
 346:       return wasTimedOut;
 347:     },
 348: 
 349:     cleanup() {
 350:       globalThis.clearTimeout(timeoutId);
 351: 
 352:       signal?.removeEventListener?.("abort", handleExternalAbort);
 353:     },
 354:   };
 355: }

 357: /*
 358:  * =========================================
 359:  * API Request
 360:  * =========================================
 361:  */
 362: 
 363: export async function apiRequest(
 364:   endpoint,
 365:   {
 366:     method = "GET",
 367:     body,
 368:     query,
 369:     headers: customHeaders,
 370:     signal,
 371:     timeoutMs = DEFAULT_TIMEOUT_MS,
 372:     cache = "no-store",
 373:   } = {},
 374: ) {
 375:   const normalizedMethod = String(method).trim().toUpperCase() || "GET";
 376: 
 377:   const normalizedEndpoint = normalizeEndpoint(endpoint);
 378: 
 379:   const requestUrl = appendQueryParameters(
 380:     `${API_BASE_URL}${normalizedEndpoint}`,
 381:     query,
 382:   );
 383: 
 384:   const headers = new Headers(customHeaders || {});
 385: 
 386:   if (!headers.has("Accept")) {
 387:     headers.set("Accept", "application/json");
 388:   }
 389: 
 390:   const requestBody = prepareRequestBody({
 391:     body,
 392: 
 393:     headers,
 394:   });
 395: 
 396:   const abortState = createRequestAbortController({
 397:     signal,
 398: 
 399:     timeoutMs: Number(timeoutMs) > 0 ? Number(timeoutMs) : DEFAULT_TIMEOUT_MS,
 400:   });
 401: 
 402:   try {
 403:     const response = await fetch(requestUrl, {
 404:       method: normalizedMethod,
 405: 
 406:       /*
 407:        * Required for HttpOnly session
 408:        * cookies on cross-origin requests.
 409:        */
 410: 
 411:       credentials: "include",
 412: 
 413:       headers,
 414: 
 415:       body: ["GET", "HEAD"].includes(normalizedMethod)
 416:         ? undefined
 417:         : requestBody,
 418: 
 419:       signal: abortState.signal,
 420: 
 421:       cache,
 422:     });
 423: 
 424:     const data = await parseResponseBody(response);
 425: 
 426:     if (!response.ok) {
 427:       throw createResponseError({
 428:         response,
 429: 
 430:         data,
 431: 
 432:         method: normalizedMethod,
 433: 
 434:         url: requestUrl,
 435:       });
 436:     }
 437: 
 438:     return data;
 439:   } catch (error) {
 440:     if (error instanceof ApiError) {
 441:       throw error;
 442:     }
 443: 
 444:     if (error?.name === "AbortError") {
 445:       throw new ApiError({
 446:         message: abortState.wasTimedOut()
 447:           ? "The server took too long to respond."
 448:           : "The request was canceled.",
 449: 
 450:         status: 0,
 451: 
 452:         code: abortState.wasTimedOut()
 453:           ? "API_REQUEST_TIMEOUT"
 454:           : "API_REQUEST_ABORTED",
 455: 
 456:         method: normalizedMethod,
 457: 
 458:         url: requestUrl,
 459: 
 460:         cause: error,
 461:       });
 462:     }
 463: 
 464:     throw new ApiError({
 465:       message:
 466:         "Unable to connect to the server. Check your connection and try again.",
 467: 
 468:       status: 0,
 469: 
 470:       code: "API_NETWORK_ERROR",
 471: 
 472:       method: normalizedMethod,
 473: 
 474:       url: requestUrl,
 475: 
 476:       cause: error,
 477:     });
 478:   } finally {
 479:     abortState.cleanup();
 480:   }
 481: }
