  112:   assertValidExperience(experienceData);
  113: 
  114:   const experiences = readStoredExperiences();
  115: 
  116:   const experience = createExperienceModel(experienceData);
  117: 
  118:   const duplicateId = experiences.some((item) => item.id === experience.id);
  119: 
  120:   if (duplicateId) {
  121:     throw createServiceError({
  122:       message: `Experience ID "${experience.id}" already exists.`,
  123:       publicMessage:
  124:         "This experience could not be created because its identifier already exists.",
  125:       code: "EXPERIENCE_ID_CONFLICT",
  126:       status: 409,
  127:     });
  128:   }
  129: 
  130:   replaceStoredExperiences([experience, ...experiences]);
  131: 
  132:   return resolveAsync(experience);
  133: }

  135: /*
  136:  * =========================================
  137:  * Update Experience
  138:  * =========================================
  139:  */
  140: 
  141: export async function updateExperience(experienceId, updates) {
  142:   if (!experienceId) {
  143:     throw createServiceError({
  144:       message: "An experience ID is required to update an experience.",
  145:       publicMessage: "The experience could not be identified.",
  146:       code: "EXPERIENCE_ID_REQUIRED",
  147:       status: 400,
  148:     });
  149:   }
  150: 
  151:   const experiences = readStoredExperiences();
  152: 
  153:   const experienceIndex = experiences.findIndex(
  154:     (experience) => experience.id === experienceId,
  155:   );
  156: 
  157:   if (experienceIndex === -1) {
  158:     throw createNotFoundError(experienceId);
  159:   }
  160: 
  161:   const currentExperience = experiences[experienceIndex];
  162: 
  163:   /*
  164:    * Merge the record before validating it because
  165:    * updates may include only part of the model.
  166:    */
  167: 
  168:   const mergedExperience = {
  169:     ...currentExperience,
  170:     ...updates,
  171: 
  172:     position: {
  173:       ...currentExperience.position,
  174:       ...updates?.position,
  175:     },
  176: 
  177:     organization: {
  178:       ...currentExperience.organization,
  179:       ...updates?.organization,
  180:     },
  181: 
  182:     location: {
  183:       ...currentExperience.location,
  184:       ...updates?.location,
  185:     },
  186: 
  187:     dates: {
  188:       ...currentExperience.dates,
  189:       ...updates?.dates,
  190:     },
  191: 
  192:     leadership: {
  193:       ...currentExperience.leadership,
  194:       ...updates?.leadership,
  195:     },
  196: 
  197:     privateInformation: {
  198:       ...currentExperience.privateInformation,
  199:       ...updates?.privateInformation,
  200:     },
  201: 
  202:     visibility: {
  203:       ...currentExperience.visibility,
  204:       ...updates?.visibility,
  205:     },
  206:   };
  207: 
  208:   assertValidExperience(mergedExperience);
  209: 
  210:   const updatedExperience = updateExperienceModel(currentExperience, updates);
  211: 
  212:   const nextExperiences = [...experiences];
  213: 
  214:   nextExperiences[experienceIndex] = updatedExperience;
  215: 
  216:   replaceStoredExperiences(nextExperiences);
  217: 
  218:   return resolveAsync(updatedExperience);
  219: }

  251: export async function deleteExperience(experienceId) {
  252:   if (!experienceId) {
  253:     throw createServiceError({
  254:       message: "An experience ID is required to delete an experience.",
  255:       publicMessage: "The experience could not be identified.",
  256:       code: "EXPERIENCE_ID_REQUIRED",
  257:       status: 400,
  258:     });
  259:   }
  260: 
  261:   const experiences = readStoredExperiences();
  262: 
  263:   const experienceToDelete = experiences.find(
  264:     (experience) => experience.id === experienceId,
  265:   );
  266: 
  267:   if (!experienceToDelete) {
  268:     throw createNotFoundError(experienceId);
  269:   }
  270: 
  271:   const remainingExperiences = experiences.filter(
  272:     (experience) => experience.id !== experienceId,
  273:   );
  274: 
  275:   replaceStoredExperiences(remainingExperiences);
  276: 
  277:   return resolveAsync({
  278:     id: experienceId,
  279:     deleted: true,
  280:   });
  281: }
