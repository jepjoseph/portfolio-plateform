import {
  getAuthenticatedCertifications,
  getAuthenticatedCertification,
  createAuthenticatedCertification,
  updateAuthenticatedCertification,
} from "../services/Service/certificationService.js";

function certificationRequestError(message) {
  const error = new Error(message);
  error.name = "CertificationRequestError";
  error.code = "CERTIFICATION_REQUEST_INVALID";
  error.status = 400;
  error.publicMessage = message;
  return error;
}

function requireFields(value, allowedFields) {
  if (
    value === null ||
    typeof value !== "object" ||
    Array.isArray(value)
  ) {
    throw certificationRequestError(
      "Request body must be a JSON object.",
    );
  }

  const unexpected = Object.keys(value).filter(
    (field) => !allowedFields.includes(field),
  );

  if (unexpected.length > 0) {
    throw certificationRequestError(
      `Unexpected request fields: ${unexpected.join(", ")}.`,
    );
  }

  return value;
}

function sendSuccess(response, status, message, data) {
  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",
    Pragma: "no-cache",
    Expires: "0",
    "X-Content-Type-Options": "nosniff",
  });

  return response.status(status).json({
    status: "success",
    message,
    data,
  });
}

export async function listMyCertifications(request, response, next) {
  try {
    const certifications = await getAuthenticatedCertifications({
      userId: request.auth.userId,
    });

    return sendSuccess(
      response,
      200,
      "Certifications retrieved successfully.",
      { certifications },
    );
  } catch (error) {
    return next(error);
  }
}

}

import { Router } from "express";

import { requireAuthentication } from "../middleware/authenticationMiddleware.js";

import {
  listMyCertifications,
  getMyCertification,
  createMyCertification,
  updateMyCertification,
} from "../controllers/certificationController.js";

import {
  uploadMyCertificationDocument,
  listMyCertificationDocuments,
  getMyCertificationDocument,
  deleteMyCertificationDocument,
} from "../controllers/certificationDocumentController.js";

import {
  requireCertificationOwnership,
} from "../middleware/certification/certificationOwnershipMiddleware.js";

import {
  uploadCertificationDocument,
} from "../middleware/certification/certificationDocumentUploadMiddleware.js";

import {
  validateCertificationDocument,
} from "../middleware/certification/certificationDocumentValidationMiddleware.js";

const router = Router();

// All Certification endpoints require a valid authenticated session.
router.use(requireAuthentication);

// Certification CRUD.
router.get("/", listMyCertifications);

router.post("/", createMyCertification);

router.get("/:certificationId", getMyCertification);

router.patch("/:certificationId", updateMyCertification);

// List documents belonging to an owned Certification.
router.get(
  "/:certificationId/documents",
  requireCertificationOwnership,
  listMyCertificationDocuments,
);

// Verify ownership before writing uploaded files.
router.post(
  "/:certificationId/documents",
  requireCertificationOwnership,
  uploadCertificationDocument,
  validateCertificationDocument,
  uploadMyCertificationDocument,
);

// Download only documents belonging to an owned Certification.
router.get(
  "/:certificationId/documents/:documentId",
  requireCertificationOwnership,
  getMyCertificationDocument,
);

// Delete only documents belonging to an owned Certification.
router.delete(
  "/:certificationId/documents/:documentId",
  requireCertificationOwnership,
  deleteMyCertificationDocument,
);

// Certification deletion is intentionally not exposed.
// It requires coordinated SQL and private-file cleanup.

export default router;
