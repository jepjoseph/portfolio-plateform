  284: function normalizeDocuments(value) {
  285:   const usedIds = new Set();
  286: 
  287:   const documents = getArray(value).reduce((result, document, index) => {
  288:     if (!document || typeof document !== "object") {
  289:       return result;
  290:     }
  291: 
  292:     const normalized = createCertificationDocument(document, index);
  293: 
  294:     if (usedIds.has(normalized.id)) {
  295:       return result;
  296:     }
  297: 
  298:     usedIds.add(normalized.id);
  299: 
  300:     result.push({
  301:       ...normalized,
  302:       order: result.length,
  303:     });
  304: 
  305:     return result;
  306:   }, []);
  307: 
  308:   /*
  309:    * Only one active document can be the primary
  310:    * document used for the Certification item preview.
  311:    */
  312: 
  313:   const requestedPrimaryIndex = documents.findIndex(
  314:     (document) => document.isPrimary && document.status !== "archived",
  315:   );
  316: 
  317:   const primaryIndex =
  318:     requestedPrimaryIndex >= 0
  319:       ? requestedPrimaryIndex
  320:       : documents.findIndex((document) => document.status !== "archived");
  321: 
  322:   return documents.map((document, index) => ({
  323:     ...document,
  324:     isPrimary: index === primaryIndex,
  325:   }));
  326: }

  409: export function normalizeCertification(value = {}) {
  410:   const source = value && typeof value === "object" ? value : {};
  411: 
  412:   const empty = createEmptyCertification();

  416:   /*
  417:    * Support both the new object structure and

  421:   const issuer =
  422:     source.issuingOrganization && typeof source.issuingOrganization === "object"
  423:       ? source.issuingOrganization

  425:           name: source.issuingOrganization || source.issuer,

  429:     source.dates && typeof source.dates === "object" ? source.dates : {};

  672:  * Effective Credential State
  673:  * =========================================
  674:  */
  675: 
  676: export function getEffectiveCredentialState(certification, today = new Date()) {
  677:   const normalized = normalizeCertification(certification);
  678: 
  679:   const todayMonth = today.toISOString().slice(0, 7);
  680: 
  681:   if (
  682:     normalized.credential.state === "planned" ||
  683:     (normalized.dates.issueDate && normalized.dates.issueDate > todayMonth)
  684:   ) {
  685:     return "planned";
  686:   }
  687: 
  688:   if (
  689:     !normalized.dates.doesNotExpire &&
  690:     normalized.dates.expirationDate &&
  691:     normalized.dates.expirationDate < todayMonth
  692:   ) {
  693:     return "expired";
  694:   }
  695: 
  696:   return normalized.credential.state;
  697: }
