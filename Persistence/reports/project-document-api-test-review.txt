/*
 * =========================================
 * Allowed Frontend Origins
 * =========================================
 */

const allowedOrigins = environment.FRONTEND_URL.split(",")
  .map((origin) => origin.trim())
  .filter(Boolean);

/*
 * =========================================
 * CORS
 * =========================================
 */

app.use(
  cors({
    origin(origin, callback) {
      /*
       * Postman, curl and server-to-server
       * requests may not include Origin.
       */

      if (!origin) {
        return callback(null, true);
      }

      if (allowedOrigins.includes(origin)) {
        return callback(null, true);
      }

      const error = new Error(`Origin not allowed by CORS: ${origin}`);

      error.name = "CorsOriginError";

      error.code = "CORS_ORIGIN_DENIED";

      error.status = 403;

      error.publicMessage = "This website is not allowed to access the API.";

      return callback(error);
    },

    credentials: true,

    methods: ["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],

    allowedHeaders: ["Content-Type", "Authorization"],
  }),
);

/*
 * =========================================
 * Public Profile Pictures
 * =========================================

import { getAuthenticatedProject } from "../../services/Service/projectService.js";

/*
 * =========================================
 * Project Ownership Middleware
 * =========================================
 *
 * Verifies that the requested Project belongs
 * to the authenticated user.
 *
 * This middleware must run after authentication
 * and before Project media upload processing.
 */

export async function requireProjectOwnership(request, response, next) {
  try {
    const project = await getAuthenticatedProject({
      userId: request.auth?.userId,
      projectId: request.params.projectId,
    });

    request.authorizedProject = project;

    return next();
  } catch (error) {
    return next(error);
  }
}
