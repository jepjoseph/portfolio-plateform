  178: function validateCertificationRules(value, context) {
  179:   const { dates, credential, supportingDocuments } = value;
  180: 
  181:   function issue(path, message) {
  182:     context.addIssue({
  183:       code: "custom",
  184:       path,
  185:       message,
  186:     });
  187:   }
  188: 
  189:   if (credential.state !== "planned" && !dates.issueDate) {
  190:     issue(["dates", "issueDate"], "Issue date is required.");
  191:   }
  192: 
  193:   if (dates.issueDate && dates.expirationDate &&
  194:       dates.expirationDate < dates.issueDate) {
  195:     issue(["dates", "expirationDate"],
  196:       "Expiration date cannot precede issue date.");
  197:   }
  198: 
  199:   if (dates.issueDate && dates.lastRenewedDate &&
  200:       dates.lastRenewedDate < dates.issueDate) {
  201:     issue(["dates", "lastRenewedDate"],
  202:       "Renewal date cannot precede issue date.");
  203:   }
  204: 
  205:   if (dates.lastRenewedDate && dates.nextRenewalDate &&
  206:       dates.nextRenewalDate < dates.lastRenewedDate) {
  207:     issue(["dates", "nextRenewalDate"],
  208:       "Next renewal cannot precede last renewal.");
  209:   }
  210: 
  211:   if (dates.doesNotExpire &&
  212:       (dates.expirationDate || dates.nextRenewalDate)) {
  213:     issue(["dates", "doesNotExpire"],
  214:       "Non-expiring certifications cannot have expiration or next-renewal dates.");
  215:   }
  216: 
  217:   const active = supportingDocuments.filter(
  218:     (document) => document.status !== "archived",
  219:   );
  220: 
  221:   const primary = active.filter(
  222:     (document) => document.isPrimary,
  223:   );
  224: 
  225:   if (active.length && primary.length !== 1) {
  226:     issue(["supportingDocuments"],
  227:       "Exactly one active document must be primary.");
  228:   }
  229: 
  230:   const seenDocuments = new Set();
  231: 
  232:   supportingDocuments.forEach((document, index) => {
  233:     if (seenDocuments.has(document.id)) {
  234:       issue(["supportingDocuments", index, "id"],
  235:         "Duplicate document ID.");
  236:     }
  237:     seenDocuments.add(document.id);
  238: 
  239:     if (document.status !== "archived" && !document.storageKey) {
  240:       issue(["supportingDocuments", index, "storageKey"],
  241:         "Active documents require a storage key.");
  242:     }
  243: 
  244:     const extension = document.fileName.toLowerCase().slice(
  245:       document.fileName.lastIndexOf("."),
  246:     );
  247: 
  248:     if (extension !== document.extension) {
  249:       issue(["supportingDocuments", index, "extension"],
  250:         "Document extension does not match the filename.");
  251:     }
  252:   });
  253: 
  254:   for (const [records, field, key] of [
  255:     [value.skillRelationships, "skillRelationships", "skillId"],
  256:     [value.relatedRecords.trainingRelationships,
  257:       "relatedRecords.trainingRelationships", "trainingId"],
  258:   ]) {
  259:     const seen = new Set();
  260: 
  261:     records.forEach((record, index) => {
  262:       if (seen.has(record[key])) {
  263:         issue([field, index, key], "Duplicate relationship.");
  264:       }
  265:       seen.add(record[key]);
  266:     });
  267:   }
  268: 
  269:   const educationIds = value.relatedRecords.educationIds;
  270:   if (new Set(educationIds).size !== educationIds.length) {
  271:     issue(["relatedRecords", "educationIds"],
  272:       "Duplicate Education relationship.");
  273:   }
  274: 
  275:   if (new Date(value.updatedAt) < new Date(value.createdAt)) {
  276:     issue(["updatedAt"],
  277:       "Updated timestamp cannot precede creation.");
  278:   }
  279: }
  280: 
  281: export const completeCertificationSchema =
  282:   certificationBaseSchema.superRefine(validateCertificationRules);

  285:   completeCertificationSchema;

  293:       "Provide at least one Certification field to update.",
  294:     );
