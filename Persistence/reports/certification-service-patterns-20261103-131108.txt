 203: export async function updateAuthenticatedSkill({
 204:   userId,
 205:   skillId,
 206:   updates,
 207:   rowVersion,
 208: }) {
 209:   const authenticatedUserId = assertUserId(userId);
 210:   const normalizedSkillId = assertSkillId(skillId);
 211:   const expectedRowVersion = parseRowVersion(rowVersion);
 212: 
 213:   const validatedUpdates = parseSchema(updateSkillSchema, updates);
 214: 
 215:   const existing = await findSkillById(
 216:     authenticatedUserId,
 217:     normalizedSkillId,
 218:   );
 219: 
 220:   if (!existing) {
 221:     throw notFound();
 222:   }
 223: 
 224:   const merged = parseSchema(completeSkillSchema, {
 225:     ...existing,
 226:     ...validatedUpdates,
 227: 
 228:     proficiency: {
 229:       ...existing.proficiency,
 230:       ...validatedUpdates.proficiency,
 231:     },
 232: 
 233:     language: {
 234:       ...existing.language,
 235:       ...validatedUpdates.language,
 236:     },
 237:   });
 238: 
 239:   const candidate = toStoredSkill(existing.id, merged);
 240: 
 241:   const existingSkills = await listSkillsByUser(authenticatedUserId);
 242: 
 243:   assertNoConflicts(candidate, existingSkills, existing.id);
 244: 
 245:   try {
 246:     const updated = await updateSkill(
 247:       authenticatedUserId,
 248:       candidate,
 249:       expectedRowVersion,
 250:     );
 251: 
 252:     if (!updated) {
 253:       throw versionConflict();
 254:     }
 255: 
 256:     return updated;
 257:   } catch (error) {
 258:     handleDatabaseConflict(error);
 259:   }
 260: }

 299: export async function importAuthenticatedSkills({
 300:   userId,
 301:   skills,
 302: }) {
 303:   if (
 304:     typeof userId !== "string" ||
 305:     userId.trim().length === 0
 306:   ) {
 307:     const error = new Error("Authentication is required.");
 308:     error.status = 401;
 309:     error.code = "AUTHENTICATION_REQUIRED";
 310:     error.publicMessage = error.message;
 311:     throw error;
 312:   }
 313: 
 314:   const { importSkillsSchema } = await import(
 315:     "../../schemas/skillSchemas.js"
 316:   );
 317: 
 318:   const { importSkillsBatch } = await import(
 319:     "../../repositories/skillRepository.js"
 320:   );
 321: 
 322:   const result = importSkillsSchema.safeParse({ skills });
 323: 
 324:   if (!result.success) {
 325:     const error = new Error("Invalid Skills import payload.");
 326: 
 327:     error.status = 400;
 328:     error.code = "SKILL_IMPORT_INVALID";
 329:     error.publicMessage = error.message;
 330:     error.errors = result.error.issues.map((issue) => ({
 331:       path: issue.path.join("."),
 332:       message: issue.message,
 333:     }));
 334: 
 335:     throw error;
 336:   }
 337: 
 338:   return importSkillsBatch(userId, result.data.skills);
 339: }
