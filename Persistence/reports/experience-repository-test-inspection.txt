import assert from "node:assert/strict";

/*

 * Mock Database State
 * =========================================
 */

let databaseState;

function resetDatabaseState() {
  databaseState = {
    queryQueue: [],

    executedQueries: [],

    transactions: [],
  };
}

resetDatabaseState();

/*
 * =========================================
 * Queue SQL Result
 * =========================================
 */

function queueQuery(expectedPattern, result) {
  databaseState.queryQueue.push({
    expectedPattern,

    result,
  });
}

function assertQueryQueueEmpty() {
  assert.equal(
    databaseState.queryQueue.length,
    0,
    "Not every queued SQL operation was executed.",
  );
}

/*
 * =========================================
 * Mock SQL Request
 * =========================================
 */

class MockRequest {
  constructor(parent = null) {
    this.parent = parent;
    this.inputs = {};
  }

  input(name, type, value) {
    this.inputs[name] = {
      type,
      value,
    };

    return this;
  }

  async query(queryText) {
    const queued = databaseState.queryQueue.shift();

    assert.ok(queued, `Unexpected SQL query:\n${queryText}`);

    assert.match(queryText, queued.expectedPattern);

    databaseState.executedQueries.push({
      queryText,

      inputs: this.inputs,

      parent: this.parent,
    });

    if (queued.result instanceof Error) {
      throw queued.result;
    }

    if (typeof queued.result === "function") {
      return queued.result({
        queryText,

        inputs: this.inputs,

        parent: this.parent,
      });
    }

    return queued.result;
  }
}

/*
 * =========================================
 * Mock SQL Transaction
 * =========================================
 */

class MockTransaction {
  constructor(pool) {
    this.pool = pool;

    this.beginCalls = [];
    this.commitCalls = 0;
    this.rollbackCalls = 0;

    databaseState.transactions.push(this);
  }

  async begin(isolationLevel) {
    this.beginCalls.push(isolationLevel);
  }

  async commit() {
    this.commitCalls += 1;
  }

  async rollback() {
    this.rollbackCalls += 1;
  }
}
