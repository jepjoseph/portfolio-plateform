function setPrivateHeaders(response) {
  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",
    Pragma: "no-cache",
    Expires: "0",
    "X-Content-Type-Options": "nosniff",

}

export async function getMyProjectDocument(request, response, next) {
  let fileHandle;

  try {
    const userId = request.auth.userId;
    const projectId = request.params.projectId;
    const documentId = request.params.documentId;

    const document = await getAuthenticatedProjectDocument({
      userId,
      projectId,
      documentId,
    });

    const filePath = resolveOwnedProjectDocumentPath(
      userId,
      projectId,
      document.storageKey,
    );

    await assertSafeProjectDocumentDirectory(userId, projectId);

    const pathStats = await fsPromises.lstat(filePath);

    if (pathStats.isSymbolicLink() || !pathStats.isFile()) {
      throw createNotFoundError();
    }

    const flags =
      fs.constants.O_RDONLY | (fs.constants.O_NOFOLLOW ?? 0);

    fileHandle = await fsPromises.open(filePath, flags);

    const stats = await fileHandle.stat();

    if (!stats.isFile() || stats.size !== document.fileSize) {
      throw createNotFoundError();
    }

    // Prevent CR/LF and header parameter injection.
    const safeName = String(document.originalFileName || "document")
      .replace(/[\r\n"\\]/g, "_")
      .replace(/[\x00-\x1f\x7f]/g, "_")
      .slice(0, 200);

    const encodedName = encodeURIComponent(
      document.originalFileName || "document",
    );

    setPrivateHeaders(response);

    response.set({
      "Content-Type": document.mimeType,
      "Content-Length": String(stats.size),
      "Content-Disposition":
        `attachment; filename="${safeName}"; filename*=UTF-8''${encodedName}`,
      "Accept-Ranges": "none",
    });

    const stream = fileHandle.createReadStream({
      autoClose: true,
    });

    fileHandle = null;

    stream.on("error", (error) => {
      if (!response.headersSent) {
        next(error);
      } else {
        response.destroy(error);
      }
    });

    stream.pipe(response);
  } catch (error) {
    if (fileHandle) {
      await fileHandle.close().catch(() => {});
    }

    if (error.code === "ENOENT") {
      return next(createNotFoundError());
    }

    return next(error);
  }
}

 * Register Uploaded Project Document
 * =========================================

export async function registerProjectDocument({
  userId,
  projectId,
  uploadedFile,
}) {
  if (!userId || !projectId) {
    throw createDocumentServiceError(
      "Project document registration is not authorized.",
      "PROJECT_DOCUMENT_REGISTRATION_UNAUTHORIZED",
      403,
    );
  }

  if (!uploadedFile) {
    throw createDocumentServiceError(
      "Validated Project document is required.",
      "PROJECT_DOCUMENT_FILE_REQUIRED",
    );
  }

  const {
    filename,
    path: uploadedPath,
    originalname,
    verifiedMimeType,
    verifiedSize,
  } = uploadedFile;

  if (
    !filename ||
    !uploadedPath ||
    !originalname ||
    !verifiedMimeType ||
    !Number.isSafeInteger(verifiedSize) ||
    verifiedSize <= 0
  ) {
    throw createDocumentServiceError(
      "Project document validation information is incomplete.",
      "PROJECT_DOCUMENT_VALIDATION_INCOMPLETE",
    );
  }

  const storageKey = createProjectDocumentStorageKey(userId, projectId, filename);

  const expectedPath = resolveOwnedProjectDocumentPath(
    userId,
    projectId,
    storageKey,
  );

  if (path.resolve(uploadedPath) !== expectedPath) {
    throw createDocumentServiceError(
      "Project document file location is invalid.",
      "PROJECT_DOCUMENT_INVALID_FILE_LOCATION",
      403,
    );
  }

  await assertSafeProjectDocumentDirectory(userId, projectId);

  /*
   * The file has already been stored and verified
   * by the upload and validation middleware.
   *
   * Now register its metadata in Azure SQL.
   */

  try {
    return await createProjectDocument(userId, projectId, {
      storageKey,
      originalFileName: originalname,
      mimeType: verifiedMimeType,
      fileSize: verifiedSize,
    });
  } catch (databaseError) {
    /*
     * The insert failed or its result is uncertain.
     *
     * Do not delete the uploaded file:
     * SQL Server might have committed the insert
     * before the connection failed.
     */

    try {
      const existingDocument = await findProjectDocumentByStorageKey(
        userId,
        projectId,
        storageKey,
      );

      if (existingDocument) {
        // The metadata exists, so preserve its file.
        return existingDocument;
      }
    } catch (lookupError) {
      console.error(
        "Unable to reconcile Project document registration:",
        lookupError,
      );
    }

    /*
     * A missing record at lookup time does not prove
     * that the original insert cannot commit later.
     *
     * Preserve the file for later reconciliation.
     */

    const error = createDocumentServiceError(
      "Project document registration could not be confirmed.",
      "PROJECT_DOCUMENT_REGISTRATION_UNCONFIRMED",
      503,
    );

    error.cause = databaseError;

    throw error;
  }
}

    path: uploadedPath,
    originalname,
    verifiedMimeType,
    verifiedSize,

      "CERTIFICATION_DOCUMENT_TOO_LARGE",

    throw createDocumentError(

      403,

  await assertSafeCertificationDocumentDirectory(
    userId,
    certificationId,
  );

  const documentId = crypto.randomUUID();

  try {
    return await createCertificationDocument(userId, certificationId, {
      id: documentId,
      storageKey,
      originalFileName: originalname,
      mimeType: verifiedMimeType,
      fileSize: verifiedSize,

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import multer from "multer";

import {
  getCertificationDocumentDirectory,
  getCertificationDocumentStorageRoot,
  assertSafeCertificationDocumentDirectory,
} from "../../services/Storage/certificationDocumentStorageService.js";

export const MAX_CERTIFICATION_DOCUMENT_SIZE = 10 * 1024 * 1024;

export const ALLOWED_CERTIFICATION_DOCUMENT_TYPES = new Map([
  ["application/pdf", { extension: ".pdf" }],
  [
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    { extension: ".docx" },
  ],
  ["image/jpeg", { extension: ".jpg" }],
  ["image/png", { extension: ".png" }],
  ["image/webp", { extension: ".webp" }],
]);

function createUploadError(message, code, status = 400) {
  const error = new Error(message);

  error.name = "CertificationDocumentUploadError";
  error.code = code;
  error.status = status;
  error.publicMessage = message;

  return error;
}

function getAuthorizedUploadContext(request) {
  const userId = String(request.auth?.userId ?? "").trim();
  const certificationId = String(
    request.params?.certificationId ?? "",
  ).trim();

  const authorizedCertification = request.authorizedCertification;

  if (!userId || !certificationId || !authorizedCertification) {
    throw createUploadError(
      "Certification document upload is not authorized.",
      "CERTIFICATION_DOCUMENT_UPLOAD_UNAUTHORIZED",
      403,
    );
  }

  if (
    String(authorizedCertification.id ?? "").trim() !== certificationId
  ) {
    throw createUploadError(
      "Authorized Certification does not match the upload destination.",
      "CERTIFICATION_DOCUMENT_UPLOAD_CERTIFICATION_MISMATCH",
      403,
    );
  }

  return { userId, certificationId };
}

async function ensureSafeUploadDirectory(userId, certificationId) {
  const root = path.resolve(getCertificationDocumentStorageRoot());

  const destination = path.resolve(
    getCertificationDocumentDirectory(userId, certificationId),
  );

  const relative = path.relative(root, destination);

  if (
    !relative ||
    relative === ".." ||
    relative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(relative)
  ) {
    throw new Error("Certification upload destination is invalid.");
  }

  // The configured storage root is expected to be a trusted
  // administrator-controlled location outside the public web root.
  await fs.mkdir(root, { recursive: true });

  const directories = [
    root,
    path.dirname(destination),
    destination,
  ];

  for (const directory of directories) {
    try {
      await fs.mkdir(directory);
    } catch (error) {
      if (error?.code !== "EEXIST") {
        throw error;
      }
    }

    const stats = await fs.lstat(directory);

    if (stats.isSymbolicLink() || !stats.isDirectory()) {
      throw new Error(
        "Certification upload directory is not a regular directory.",
      );
    }
  }

  await assertSafeCertificationDocumentDirectory(
    userId,
    certificationId,
  );

  return destination;
}

const storage = multer.diskStorage({
  destination(request, file, callback) {
    let context;

    try {
      context = getAuthorizedUploadContext(request);
    } catch (error) {
      return callback(error);
    }

    ensureSafeUploadDirectory(
      context.userId,
      context.certificationId,
    )
      .then((directory) => callback(null, directory))
      .catch((error) => callback(error));
  },

  filename(request, file, callback) {
    try {
      getAuthorizedUploadContext(request);

      const configuration = ALLOWED_CERTIFICATION_DOCUMENT_TYPES.get(
        file.mimetype,
      );

      if (!configuration) {
        throw createUploadError(
          "Unsupported Certification document type.",
          "CERTIFICATION_DOCUMENT_TYPE_NOT_ALLOWED",
        );
      }

      return callback(
        null,
        `${crypto.randomUUID()}${configuration.extension}`,
      );
    } catch (error) {
      return callback(error);
    }
  },
});

function certificationDocumentFileFilter(request, file, callback) {
  try {
    getAuthorizedUploadContext(request);

    if (!ALLOWED_CERTIFICATION_DOCUMENT_TYPES.has(file.mimetype)) {
      throw createUploadError(
        "Unsupported Certification document type.",
        "CERTIFICATION_DOCUMENT_TYPE_NOT_ALLOWED",
      );
    }

    return callback(null, true);
  } catch (error) {
    return callback(error);
  }
}

const certificationDocumentUpload = multer({
  storage,
  limits: {
    fileSize: MAX_CERTIFICATION_DOCUMENT_SIZE,
    files: 1,
  },
  fileFilter: certificationDocumentFileFilter,
}).single("document");

export function uploadCertificationDocument(request, response, next) {
  // Reject requests lacking verified ownership before invoking Multer.
  try {
    getAuthorizedUploadContext(request);
  } catch (error) {
    return next(error);
  }

  return certificationDocumentUpload(
    request,
    response,
    (error) => {
      if (!error) {
        return next();
      }

      if (error instanceof multer.MulterError) {
        if (error.code === "LIMIT_FILE_SIZE") {
          return next(
            createUploadError(
              "Certification document exceeds the 10 MiB upload limit.",
              "CERTIFICATION_DOCUMENT_TOO_LARGE",
              413,
            ),
          );
        }

        if (
          error.code === "LIMIT_FILE_COUNT" ||
          error.code === "LIMIT_UNEXPECTED_FILE"
        ) {
          return next(
            createUploadError(
              "Upload exactly one file using the document field.",
              "CERTIFICATION_DOCUMENT_INVALID_FILE_COUNT",
            ),
          );
        }

        return next(
          createUploadError(
            "Certification document upload could not be processed.",
            "CERTIFICATION_DOCUMENT_UPLOAD_INVALID",
          ),
        );
      }

      return next(error);
    },
  );
}

import path from "node:path";

function validationError(message, code, status = 400) {
  const error = new Error(message);
  error.name = "CertificationDocumentValidationError";
  error.code = code;
  error.status = status;
  error.publicMessage = message;
  return error;
}

function getContext(request) {
  const userId = String(request.auth?.userId ?? "").trim();
  const certificationId = String(request.params?.certificationId ?? "").trim();
  const authorizedId = String(
    request.authorizedCertification?.id ?? "",
  ).trim();

  if (!userId || !certificationId || authorizedId !== certificationId) {
    throw validationError(
      "Certification document validation is not authorized.",
      "CERTIFICATION_DOCUMENT_VALIDATION_UNAUTHORIZED",
      403,
    );
  }

  return { userId, certificationId };
}

function getUploadedPath(request, context) {
  if (!request.file?.path || !request.file?.filename) {
    throw validationError(
      "Please select a Certification document.",
      "CERTIFICATION_DOCUMENT_FILE_REQUIRED",
    );
  }

  const directory = path.resolve(
    getCertificationDocumentDirectory(
      context.userId,
      context.certificationId,
    ),
  );

  const filePath = path.resolve(request.file.path);

  if (
    path.dirname(filePath) !== directory ||
    path.basename(filePath) !== request.file.filename
  ) {
    throw validationError(
      "Certification document location is invalid.",
      "CERTIFICATION_DOCUMENT_INVALID_FILE_LOCATION",
      403,
    );
  }

  return filePath;
}

function validateDocxPackage(filePath) {
  return new Promise((resolve) => {
    yauzl.open(
      filePath,
      {
        lazyEntries: true,
        autoClose: true,
        validateEntrySizes: true,
        strictFileNames: true,
      },
      (error, zip) => {
        if (error || !zip) {
          resolve(false);
          return;
        }

        const required = new Set([
          "[Content_Types].xml",
          "_rels/.rels",
          "word/document.xml",
        ]);

        let finished = false;
        let count = 0;

        const finish = (valid) => {
          if (finished) return;
          finished = true;
          zip.close();
          resolve(valid);
        };

        zip.on("error", () => finish(false));

        zip.on("entry", (entry) => {
          count++;

          if (count > 10000) {
            finish(false);
            return;
          }

          required.delete(entry.fileName);
          zip.readEntry();
        });

        zip.on("end", () => finish(required.size === 0));
        zip.readEntry();
      },
    );
  });
}

async function matchesContent(filePath, mimeType) {
  const detected = await fileTypeFromFile(filePath);

  if (!detected) return false;

  if (
    mimeType ===
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
  ) {
    if (
      detected.mime !== "application/zip" &&
      detected.mime !==
        "application/vnd.openxmlformats-officedocument.wordprocessingml.document"
    ) {
      return false;
    }
    return validateDocxPackage(filePath);
  }

  return detected.mime === mimeType;
}

async function removeRejectedFile(context, filePath) {
  if (!context || !filePath) return;

  const directory = path.resolve(
    getCertificationDocumentDirectory(
      context.userId,
      context.certificationId,
    ),
  );

  if (path.dirname(filePath) !== directory) return;

  await assertSafeCertificationDocumentDirectory(
    context.userId,
    context.certificationId,
  );

  try {
    const stats = await fs.lstat(filePath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw new Error("Rejected document is not a regular file.");
    }

    await fs.unlink(filePath);
  } catch (error) {
    if (error?.code !== "ENOENT") throw error;
  }
}

export async function validateCertificationDocument(
  request,
  response,
  next,
) {
  let context;
  let filePath;

  try {
    context = getContext(request);
    filePath = getUploadedPath(request, context);

    await assertSafeCertificationDocumentDirectory(
      context.userId,
      context.certificationId,
    );

    const stats = await fs.lstat(filePath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw validationError(
        "Uploaded Certification document is not a regular file.",
        "CERTIFICATION_DOCUMENT_INVALID_FILE",
      );
    }

    if (stats.size === 0) {
      throw validationError(
        "Uploaded Certification document is empty.",
        "CERTIFICATION_DOCUMENT_EMPTY_FILE",
      );
    }

    if (stats.size > MAX_CERTIFICATION_DOCUMENT_SIZE) {
      throw validationError(
        "Certification document exceeds the 10 MiB limit.",
        "CERTIFICATION_DOCUMENT_TOO_LARGE",
        413,
      );
    }

    const mimeType = request.file.mimetype;
    const configuration = ALLOWED_CERTIFICATION_DOCUMENT_TYPES.get(mimeType);

    if (!configuration) {
      throw validationError(
        "Unsupported Certification document type.",
        "CERTIFICATION_DOCUMENT_TYPE_NOT_ALLOWED",
      );
    }

    if (
      path.extname(request.file.filename).toLowerCase() !==
      configuration.extension
    ) {
      throw validationError(
        "Certification document extension does not match its type.",
        "CERTIFICATION_DOCUMENT_EXTENSION_MISMATCH",
      );
    }

    if (!(await matchesContent(filePath, mimeType))) {
      throw validationError(
        "Certification document contents do not match its declared type.",
        "CERTIFICATION_DOCUMENT_CONTENT_TYPE_MISMATCH",
      );
    }

    request.certificationDocumentFile = {
      ...request.file,
      verifiedMimeType: mimeType,
      verifiedExtension: configuration.extension,
      verifiedSize: stats.size,
    };

    return next();
  } catch (error) {
    if (context && filePath) {
      try {
        await removeRejectedFile(context, filePath);
      } catch (cleanupError) {
        console.error(
          "Failed to clean up rejected Certification document:",
          cleanupError,
        );

        return next(
          validationError(
            "Certification document cleanup failed.",
            "CERTIFICATION_DOCUMENT_CLEANUP_FAILED",
            500,
          ),
        );
      }
    }

    return next(error);
  }
}

  });

/*
 * =========================================
 * Global Error Handler
 * =========================================
 */

export function globalErrorHandler(error, request, response, next) {
  /*
   * Express delegates to its default handler
   * if a response has already started.
   */

  if (response.headersSent) {
    return next(error);
  }

  const isMalformedJson =
    error?.type === "entity.parse.failed" && error instanceof SyntaxError;

  const status = isMalformedJson ? 400 : normalizeStatus(error?.status);

  const code = isMalformedJson
    ? "INVALID_JSON"
    : normalizePublicCode(error, status);

  const message = isMalformedJson
    ? "The request body contains invalid JSON."
    : getPublicMessage(error, status);

  /*
   * Log complete technical information only
   * on the server.
   */

  console.error("Unhandled API error:", {
    method: request.method,

    url: request.originalUrl,

    name: error?.name,

    code: error?.code,

    message: error?.message,

    status,

    provider: error?.provider,

    stack: process.env.NODE_ENV === "development" ? error?.stack : undefined,
  });

  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",

    Pragma: "no-cache",

    Expires: "0",
  });

  const fieldErrors = normalizeFieldErrors(error?.fieldErrors);

  const validationErrors = normalizeValidationErrors(
    error?.errors || error?.validationErrors,
  );

  return response.status(status).json({
    status: "error",

    code,

    message,

    ...(Object.keys(fieldErrors).length > 0
      ? {
          fieldErrors,
        }
      : {}),

    ...(validationErrors.length > 0
      ? {
          errors: validationErrors,
        }
      : {}),

    /*
     * Development details are useful while
     * building the API but are excluded from
     * production responses.
     */

    ...(process.env.NODE_ENV === "development"
      ? {
          debug: {
            name: error?.name || "Error",

            technicalMessage: error?.message || "Unknown error",

            provider: error?.provider || null,
          },
        }
      : {}),
  });
}

/*
 * =========================================
 * Status Normalization
 * =========================================
 */

function normalizeStatus(status) {
  const numericStatus = Number(status);

  if (
    Number.isInteger(numericStatus) &&
    numericStatus >= 400 &&
    numericStatus <= 599
  ) {
    return numericStatus;
  }

  return 500;
}

/*
 * =========================================
 * Public Error Code
 * =========================================
 *
 * Internal database and provider codes should
 * not automatically be returned to clients.
 * Application errors are public only when
 * they also provide a publicMessage.
 */

function normalizePublicCode(error, status) {
  const code = typeof error?.code === "string" ? error.code.trim() : "";

  const isSafeCode = /^[A-Z][A-Z0-9_]{1,99}$/.test(code);

  if (error?.publicMessage && isSafeCode) {
    return code;
  }

  if (status === 400) {
    return "BAD_REQUEST";
  }

  if (status === 401) {
    return "AUTHENTICATION_REQUIRED";
  }

  if (status === 403) {
    return "ACCESS_FORBIDDEN";
  }

  if (status === 404) {
    return "RESOURCE_NOT_FOUND";
  }

  if (status === 409) {
    return "RESOURCE_CONFLICT";
  }

  if (status === 429) {
    return "RATE_LIMIT_EXCEEDED";
  }

  if (status === 503) {
    return "SERVICE_UNAVAILABLE";
  }

  return "INTERNAL_SERVER_ERROR";
}

/*
 * =========================================
 * Public Error Message
 * =========================================
 */

function getPublicMessage(error, status) {
  if (typeof error?.publicMessage === "string" && error.publicMessage.trim()) {
    return error.publicMessage.trim();
  }

  if (status === 400) {
    return "The request could not be processed.";
  }

  if (status === 401) {
    return "Authentication is required.";
  }

  if (status === 403) {
    return "You do not have permission to perform this operation.";
  }

  if (status === 404) {
    return "The requested resource was not found.";
  }

  if (status === 409) {
    return "The request conflicts with the current resource state.";
  }

  if (status === 429) {
    return "Too many requests were submitted. Please try again later.";
  }

  if (status === 503) {
    return "The service is temporarily unavailable.";
  }

  return "An unexpected server error occurred.";
}

/*
 * =========================================
 * Field Error Normalization
 * =========================================
 */

function normalizeFieldErrors(fieldErrors) {
  if (
    !fieldErrors ||
    typeof fieldErrors !== "object" ||
    Array.isArray(fieldErrors)
  ) {
    return {};
  }

  const normalized = {};

  for (const [fieldName, messages] of Object.entries(fieldErrors)) {
    const safeFieldName = String(fieldName).slice(0, 100);

    const messageValues = Array.isArray(messages) ? messages : [messages];

    const safeMessages = messageValues
      .filter((message) => typeof message === "string")
      .map((message) => message.trim().slice(0, 500))
      .filter(Boolean);

    if (safeMessages.length > 0) {
      normalized[safeFieldName] = safeMessages;
    }
  }

  return normalized;
}

/*
 * =========================================
 * Validation Error Normalization
 * =========================================
 */

function normalizeValidationErrors(errors) {
  if (!Array.isArray(errors)) {
    return [];
  }

  return errors
    .slice(0, 50)
    .map((validationError) => {
      if (typeof validationError === "string") {
        return {
          field: "",

          code: "VALIDATION_ERROR",

          message: validationError.trim().slice(0, 500),
        };
      }

      if (!validationError || typeof validationError !== "object") {
        return null;
      }

      const message =
        typeof validationError.message === "string"
          ? validationError.message.trim().slice(0, 500)
          : "";

      if (!message) {
        return null;
      }

      return {
        field:
          typeof validationError.field === "string"
            ? validationError.field.trim().slice(0, 100)
            : "",

        code:
          typeof validationError.code === "string"
            ? validationError.code.trim().slice(0, 100)
            : "VALIDATION_ERROR",

        message,
      };
    })
    .filter(Boolean);
}
