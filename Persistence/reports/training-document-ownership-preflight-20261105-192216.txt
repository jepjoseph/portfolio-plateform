 128:  * =========================================

 130:  * Resolves the HttpOnly refresh-token cookie,
 131:  * validates the database session and places
 132:  * public-safe identity information on:
 133:  *
 134:  * request.auth
 135:  *
 136:  * This middleware validates the session but
 137:  * does not rotate its refresh token.
 138:  */

 140: export async function requireAuthentication(request, response, next) {
 141:   /*
 142:    * Protected responses must not be cached
 143:    * or shared between authenticated users.
 144:    */
 145: 
 146:   response.set({
 147:     "Cache-Control": "no-store, no-cache, must-revalidate, private",
 148: 
 149:     Pragma: "no-cache",
 150: 
 151:     Expires: "0",
 152:   });
 153: 
 154:   response.vary("Cookie");
 155: 
 156:   try {
 157:     const refreshToken = getRefreshTokenFromRequest(request);
 158: 
 159:     if (!refreshToken) {
 160:       throw createAuthenticationError({
 161:         message: "The request does not contain a refresh-token cookie.",
 162: 
 163:         code: "AUTHENTICATION_REQUIRED",
 164: 
 165:         status: 401,
 166: 
 167:         publicMessage: "You must sign in to access this resource.",
 168:       });
 169:     }
 170: 
 171:     const authentication = await getAuthenticatedSession({
 172:       refreshToken,
 173:     });
 174: 
 175:     request.auth = createRequestAuthentication(authentication);
 176: 
 177:     return next();
 178:   } catch (error) {
 179:     if (shouldClearSessionCookie(error)) {
 180:       safelyClearRefreshTokenCookie(response);
 181:     }
 182: 
 183:     return next(error);
 184:   }
