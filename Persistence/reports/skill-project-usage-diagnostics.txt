   36: /*
   37:  * =========================================
   38:  * Date Parsing
   39:  * =========================================
   40:  */
   41: 
   42: export function parseProjectDate(value) {
   43:   const normalizedValue = getText(value);
   44: 
   45:   if (!normalizedValue) {
   46:     return null;
   47:   }
   48: 
   49:   const match = normalizedValue.match(
   50:     /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
   51:   );
   52: 
   53:   if (!match) {
   54:     return null;
   55:   }
   56: 
   57:   const year = Number(match[1]);
   58:   const month = Number(match[2]);
   59:   const day = match[3] ? Number(match[3]) : 1;
   60: 
   61:   const parsedDate = new Date(Date.UTC(year, month - 1, day));
   62: 
   63:   if (
   64:     match[3] &&
   65:     (parsedDate.getUTCFullYear() !== year ||
   66:       parsedDate.getUTCMonth() !== month - 1 ||
   67:       parsedDate.getUTCDate() !== day)
   68:   ) {
   69:     return null;
   70:   }
   71: 
   72:   return {
   73:     value: normalizedValue,
   74:     date: parsedDate,
   75:     timestamp: parsedDate.getTime(),
   76:     year,
   77:     month,
   78:     day,
   79:     hasDay: Boolean(match[3]),
   80:   };
   81: }

  188:     return new Intl.DateTimeFormat(locale, formatOptions).format(

  191:   } catch {

  262:     switch (sortOption) {
  263:       case "name-descending":
  264:         return compareText(secondSkill?.name, firstSkill?.name);
  265: 
  266:       case "category-ascending":
  267:         return (
  268:           compareText(
  269:             getSkillCategoryDisplay(firstSkill),
  270:             getSkillCategoryDisplay(secondSkill),
  271:           ) || compareText(firstSkill?.name, secondSkill?.name)
  272:         );
  273: 
  274:       case "type-ascending":
  275:         return (
  276:           compareText(
  277:             getSkillTypeDisplay(firstSkill),
  278:             getSkillTypeDisplay(secondSkill),
  279:           ) || compareText(firstSkill?.name, secondSkill?.name)
  280:         );
  281: 
  282:       case "proficiency-descending": {
  283:         const proficiencyOrder = {
  284:           expert: 4,
  285:           advanced: 3,
  286:           intermediate: 2,
  287:           foundational: 1,
  288:           "": 0,
  289:         };
  290: 
  291:         const firstLevel =
  292:           proficiencyOrder[firstSkill?.proficiency?.level || ""] || 0;
  293: 
  294:         const secondLevel =
  295:           proficiencyOrder[secondSkill?.proficiency?.level || ""] || 0;
  296: 
  297:         return (
  298:           secondLevel - firstLevel ||
  299:           compareText(firstSkill?.name, secondSkill?.name)
  300:         );
  301:       }
  302: 
  303:       case "years-descending":
  304:         return (
  305:           getNumericValue(secondSkill?.proficiency?.yearsOfExperience) -
  306:             getNumericValue(firstSkill?.proficiency?.yearsOfExperience) ||
  307:           compareText(firstSkill?.name, secondSkill?.name)
  308:         );
  309: 
  310:       case "recently-created":
  311:         return getCreatedTime(secondSkill) - getCreatedTime(firstSkill);
  312: 
  313:       case "recently-updated":
  314:         return getUpdatedTime(secondSkill) - getUpdatedTime(firstSkill);
  315: 
  316:       case "name-ascending":
  317:       default:
  318:         return compareText(firstSkill?.name, secondSkill?.name);
  319:     }
  320:   });
  321: 
  322:   return sortedSkills;
  323: }
