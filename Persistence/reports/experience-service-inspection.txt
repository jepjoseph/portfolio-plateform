import {
  listProjectsByUser,
  findProjectById,
  createProject,
  updateProject,
  deleteProject,
} from "../../repositories/projectRepository.js";

import {
  parseProjectInput,
  parseProjectRowVersion,
} from "../../schemas/projectSchemas.js";

const SERVER_OWNED_FIELDS = [
  "userId",
  "user_id",
  "createdAt",
  "updatedAt",
  "rowVersion",
];

function createProjectError({ message, code, status, publicMessage }) {
  const error = new Error(message);
  error.name = "ProjectServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;
  return error;
}

function assertUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw createProjectError({
      message: "An authenticated user ID is required.",
      code: "PROJECT_AUTHENTICATION_REQUIRED",
      status: 401,
      publicMessage: "You must sign in to access your Projects.",
    });
  }

  return userId.trim();
}

function assertProjectId(projectId) {
  if (
    typeof projectId !== "string" ||
    !projectId.trim() ||
    projectId.length > 200
  ) {
    throw createProjectError({
      message: "A valid Project ID is required.",
      code: "PROJECT_ID_INVALID",
      status: 400,
      publicMessage: "The Project identifier is invalid.",
    });
  }

  return projectId.trim();
}

function projectNotFound() {
  return createProjectError({
    message: "Project not found for authenticated user.",
    code: "PROJECT_NOT_FOUND",
    status: 404,
    publicMessage: "The requested Project could not be found.",
  });
}

function projectVersionConflict() {
  return createProjectError({
    message: "Project version conflict.",
    code: "PROJECT_VERSION_CONFLICT",
    status: 409,
    publicMessage:
      "This Project was changed elsewhere. Reload it before saving.",
  });
}

function projectIdMismatch() {
  return createProjectError({
    message: "The Project body ID does not match the requested Project ID.",
    code: "PROJECT_ID_MISMATCH",
    status: 400,
    publicMessage: "The Project identifiers do not match.",
  });
}

function withoutServerFields(project) {
  const result = { ...project };

  for (const field of SERVER_OWNED_FIELDS) {
    delete result[field];
  }

  return result;
}

export async function getAuthenticatedProjects({ userId }) {
  return listProjectsByUser(assertUserId(userId));
}

export async function getAuthenticatedProject({ userId, projectId }) {
  const project = await findProjectById(
    assertUserId(userId),
    assertProjectId(projectId),
  );

  if (!project) {
    throw projectNotFound();
  }

  return project;
}

export async function createAuthenticatedProject({ userId, project }) {
  const authenticatedUserId = assertUserId(userId);

  // Reject invalid data before any database operation.
  // Client-supplied server-owned fields remain forbidden.
  const validatedProject = parseProjectInput(project);

  try {
    return await createProject(authenticatedUserId, validatedProject);
  } catch (error) {
    if (error.number === 2627 || error.number === 2601) {
      throw createProjectError({
        message: "Project ID already exists.",
        code: "PROJECT_ALREADY_EXISTS",
        status: 409,
        publicMessage: "A Project with this identifier already exists.",
      });
    }

    throw error;
  }
}

export async function updateAuthenticatedProject({
  userId,
  projectId,
  project,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);

  // Validate the supplied version and full Project body.
  const expectedRowVersion = parseProjectRowVersion(rowVersion);
  const validatedProject = parseProjectInput(project);

  if (validatedProject.id !== normalizedProjectId) {
    throw projectIdMismatch();
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function setAuthenticatedProjectArchiveStatus({
  userId,
  projectId,
  rowVersion,
  archived,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  if (typeof archived !== "boolean") {
    throw createProjectError({
      message: "Archive status must be a boolean.",
      code: "PROJECT_ARCHIVE_STATUS_INVALID",
      status: 400,
      publicMessage: "A valid archive status is required.",
    });
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  // Database-managed metadata is not editable Project content.
  const editableProject = withoutServerFields(existingProject);

  const validatedProject = parseProjectInput({
    ...editableProject,
    recordStatus: archived ? "archived" : "active",
  });

  // SQL Server must match the caller's row version.
  // Do not substitute the version from the latest database read.
  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function deleteAuthenticatedProject({
  userId,
  projectId,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const deleted = await deleteProject(
    authenticatedUserId,
    normalizedProjectId,
    expectedRowVersion,
  );

  if (!deleted) {
    throw projectVersionConflict();
  }

  return { deleted: true };
}

const CATEGORIES = new Set([
  "employment", "internship", "freelance", "self-employment",
  "volunteer", "leadership", "research", "teaching",
  "military", "other",
]);

const EMPLOYMENT_TYPES = new Set([
  "full-time", "part-time", "contract", "temporary",
  "internship", "apprenticeship", "freelance",
  "self-employed", "volunteer", "seasonal", "other",
]);

const WORK_ARRANGEMENTS = new Set([
  "on-site", "remote", "hybrid", "field-based",
  "travel-based", "other",
]);

const STATUSES = new Set(["active", "archived"]);
const PROFICIENCIES = new Set([
  "", "foundational", "intermediate", "advanced", "expert",
]);

const LIMITS = {
  "position.title": 200,
  "organization.name": 200,
  "organization.website": 500,
  "organization.industry": 150,
  "location.city": 120,
  "location.stateRegion": 120,
  "location.country": 120,
  "location.displayValue": 300,
  "overview": 3000,
  "leadership.description": 1500,
  "privateInformation.reasonForLeaving": 1000,
  "privateInformation.notes": 3000,
};

const COLLECTION_LIMITS = {
  responsibilities: 30,
  achievements: 30,
  skills: 50,
  technologies: 50,
  relatedProjectIds: 30,
};

const SERVER_FIELDS = [
  "userId", "user_id", "createdAt", "updatedAt",
  "rowVersion", "recordStatus", "record_status",
];

const ID_PATTERN =
  /^experience-(?:[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}|\d{13}-[a-z0-9]{8})$/i;

const DATE_PATTERN =
  /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/;

function isObject(value) {
  return value !== null &&
    typeof value === "object" &&
    !Array.isArray(value);
}

function createValidationError(errors) {
  const error = new Error("Experience validation failed.");
  error.name = "ExperienceValidationError";
  error.code = "EXPERIENCE_VALIDATION_FAILED";
  error.status = 400;
  error.publicMessage = "Please correct the Experience information.";
  error.errors = errors;
  error.fieldErrors = {};

  for (const item of errors) {
    if (!error.fieldErrors[item.field]) {
      error.fieldErrors[item.field] = item.message;
    }
  }

  return error;
}

function addError(errors, field, code, message) {
  errors.push({ field, code, message });
}

function text(value) {
  return typeof value === "string" ? value.trim() : "";
}

function checkText(errors, field, value, max, required = false) {
  if (value !== undefined && typeof value !== "string") {
    addError(errors, field, "invalid_type", "Must be text.");
    return;
  }

  if (required && !text(value)) {
    addError(errors, field, "required", "This field is required.");
  }

  if (typeof value === "string" && value.trim().length > max) {
    addError(errors, field, "too_long", `Maximum ${max} characters.`);
  }
}

function checkObject(errors, source, key) {
  if (!isObject(source[key])) {
    addError(errors, key, "invalid_type", "Must be an object.");
    return {};
  }

  return source[key];
}

function checkBoolean(errors, field, value) {
  if (value !== undefined && typeof value !== "boolean") {
    addError(errors, field, "invalid_type", "Must be a boolean.");
  }
}

function checkOption(errors, field, value, allowed) {
  if (!allowed.has(value)) {
    addError(errors, field, "invalid_option", "Invalid selection.");
  }
}

function validDate(value) {
  const match = typeof value === "string"
    ? value.match(DATE_PATTERN)
    : null;

  if (!match) return false;
  if (!match[3]) return true;

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);
  const date = new Date(Date.UTC(year, month - 1, day));

  return date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day;
}

function comparableDate(value, endOfMonth = false) {
  const [year, month, suppliedDay] = value.split("-").map(Number);
  const day = suppliedDay ||
    (endOfMonth ? new Date(Date.UTC(year, month, 0)).getUTCDate() : 1);

  return Date.UTC(year, month - 1, day);
}

function checkDates(errors, dates) {
  const start = dates.startDate;
  const end = dates.endDate;
  const current = dates.isCurrent;

  checkBoolean(errors, "dates.isCurrent", current);

  if (!validDate(start)) {
    addError(errors, "dates.startDate", "invalid_date",
      "A valid start date is required.");
  }

  if (current === true && text(end)) {
    addError(errors, "dates.endDate", "current_position_has_end_date",
      "A current position cannot have an end date.");
  }

  if (current !== true && !text(end)) {
    addError(errors, "dates.endDate", "required",
      "An end date is required for a previous position.");
  } else if (text(end) && !validDate(end)) {
    addError(errors, "dates.endDate", "invalid_date",
      "Invalid end date.");
  }

  if (validDate(start) && validDate(end) &&
      comparableDate(end, true) < comparableDate(start)) {
    addError(errors, "dates.endDate", "invalid_date_order",
      "End date cannot precede start date.");
  }
}

function checkCollection(errors, experience, name) {
  const values = experience[name];

  if (values === undefined) return;

  if (!Array.isArray(values)) {
    addError(errors, name, "invalid_type", "Must be an array.");
    return;
  }

  if (values.length > COLLECTION_LIMITS[name]) {
    addError(errors, name, "too_many_items",
      `Maximum ${COLLECTION_LIMITS[name]} items.`);
  }
}

function checkUnique(errors, values, field, identity) {
  const seen = new Set();

  values.forEach((item, index) => {
    const key = identity(item).trim().toLowerCase();
    if (!key) return;

    if (seen.has(key)) {
      addError(errors, `${field}.${index}`, "duplicate",
        "Duplicate entry.");
    }
    seen.add(key);
  });
}

function checkRelationships(errors, values, kind) {
  if (!Array.isArray(values)) return;

  values.forEach((value, index) => {
    const prefix = `${kind}.${index}`;

    if (!isObject(value)) {
      addError(errors, prefix, "invalid_type", "Must be an object.");
      return;
    }

    const skillId = text(value.skillId || value.profileSkillId);
    const snapshot = text(
      value.nameSnapshot || value.name || value.label || value.value
    );

    if (!skillId && !snapshot) {
      addError(errors, `${prefix}.skillId`, "required",
        "Select a saved skill or provide its name.");
    }

    if (skillId && !snapshot) {
      addError(errors, `${prefix}.nameSnapshot`, "required_snapshot",
        "A name snapshot is required.");
    }

    for (const field of ["id", "skillId"]) {
      if (value[field] !== undefined) {
        checkText(errors, `${prefix}.${field}`, value[field], 200);
      }
    }

    const levelField = kind === "skills" ? "level" : "proficiency";
    const level = value[levelField] ?? "";

    checkOption(errors, `${prefix}.${levelField}`, level, PROFICIENCIES);

    if (value.order !== undefined &&
        (!Number.isInteger(value.order) || value.order < 0)) {
      addError(errors, `${prefix}.order`, "invalid_order",
        "Order must be a nonnegative integer.");
    }

    const limits = kind === "skills"
      ? { nameSnapshot: 150, categorySnapshot: 100,
          typeSnapshot: 100, usageDescription: 500 }
      : { nameSnapshot: 150, name: 150,
          category: 100, usageDescription: 500 };

    for (const [field, max] of Object.entries(limits)) {
      checkText(errors, `${prefix}.${field}`, value[field], max);
    }
  });

  checkUnique(errors, values, kind, (value) => {
    if (!isObject(value)) return "";
    const id = text(value.skillId || value.profileSkillId);
    const name = text(value.nameSnapshot || value.name);
    return id ? `id:${id}` : `name:${name}`;
  });
}

export function parseExperienceInput(value) {
  const errors = [];

  if (!isObject(value)) {
    throw createValidationError([{
      field: "experience",
      code: "invalid_type",
      message: "Experience must be an object.",
    }]);
  }

  for (const field of SERVER_FIELDS) {
    if (Object.hasOwn(value, field)) {
      addError(errors, field, "server_owned",
        "This field is managed by the server.");
    }
  }

  if (typeof value.id !== "string" ||
      value.id.length > 200 || !ID_PATTERN.test(value.id)) {
    addError(errors, "id", "invalid_identifier",
      "A valid Experience ID is required.");
  }

  checkOption(errors, "category", value.category, CATEGORIES);
  checkOption(errors, "status", value.status, STATUSES);

  const position = checkObject(errors, value, "position");
  const organization = checkObject(errors, value, "organization");
  const dates = checkObject(errors, value, "dates");

  checkOption(errors, "position.employmentType",
    position.employmentType, EMPLOYMENT_TYPES);
  checkOption(errors, "position.workArrangement",
    position.workArrangement, WORK_ARRANGEMENTS);

  for (const [field, max] of Object.entries(LIMITS)) {
    const [section, property] = field.split(".");
    const fieldValue = property ? value[section]?.[property] : value[field];

    checkText(errors, field, fieldValue, max,
      field === "position.title" || field === "organization.name");
  }

  if (value.organization?.website) {
    const website = text(value.organization.website);

    try {
      const normalized = /^https?:\/\//i.test(website)
        ? website : `https://${website}`;
      const url = new URL(normalized);

      if (!["http:", "https:"].includes(url.protocol) || !url.hostname) {
        throw new Error("Invalid URL");
      }
    } catch {
      addError(errors, "organization.website", "invalid_url",
        "A valid website is required.");
    }
  }

  checkDates(errors, dates);

  for (const name of Object.keys(COLLECTION_LIMITS)) {
    checkCollection(errors, value, name);
  }

  for (const kind of ["responsibilities", "achievements"]) {
    const values = value[kind];
    if (!Array.isArray(values)) continue;

    values.forEach((item, index) => {
      const itemText = typeof item === "string"
        ? item : isObject(item) ? item.text : "";

      checkText(errors, `${kind}.${index}.text`, itemText,
        kind === "responsibilities" ? 1500 : 1500, true);

      if (kind === "achievements" && isObject(item)) {
        checkText(errors, `${kind}.${index}.metric`, item.metric, 300);
      }
    });

    checkUnique(errors, values, kind, (item) =>
      typeof item === "string" ? item : text(item?.text));
  }

  checkRelationships(errors, value.skills, "skills");
  checkRelationships(errors, value.technologies, "technologies");

  if (Array.isArray(value.relatedProjectIds)) {
    value.relatedProjectIds.forEach((id, index) => {
      checkText(errors, `relatedProjectIds.${index}`, id, 200, true);
    });

    checkUnique(errors, value.relatedProjectIds,
      "relatedProjectIds", (id) => text(id));
  }

  for (const name of ["location", "leadership",
    "privateInformation", "visibility", "overviewMeta"]) {
    if (value[name] !== undefined && !isObject(value[name])) {
      addError(errors, name, "invalid_type", "Must be an object.");
    }
  }

  const leadership = isObject(value.leadership) ? value.leadership : {};
  checkBoolean(errors, "leadership.hasLeadershipResponsibilities",
    leadership.hasLeadershipResponsibilities);

  const people = leadership.peopleManaged;
  if (people !== undefined && people !== null && people !== "") {
    const number = Number(people);

    if (!Number.isInteger(number) || number < 0 || number > 100000) {
      addError(errors, "leadership.peopleManaged", "invalid_number",
        "People managed must be between 0 and 100000.");
    }
  }

  if (isObject(value.visibility)) {
    for (const [key, enabled] of Object.entries(value.visibility)) {
      checkBoolean(errors, `visibility.${key}`, enabled);
    }
  }

  const hasOverview = Boolean(text(value.overview));
  const hasResponsibility = Array.isArray(value.responsibilities) &&
    value.responsibilities.some((item) =>
      Boolean(text(typeof item === "string" ? item : item?.text)));
  const hasAchievement = Array.isArray(value.achievements) &&
    value.achievements.some((item) =>
      Boolean(text(typeof item === "string" ? item : item?.text)));

  if (!hasOverview && !hasResponsibility && !hasAchievement) {
    addError(errors, "overview", "insufficient_content",
      "Provide an overview, responsibility, or achievement.");
  }

  if (errors.length) {
    throw createValidationError(errors);
  }

  return structuredClone(value);
}

export function parseExperienceRowVersion(value) {
  if (typeof value !== "string" ||
      !/^[A-Za-z0-9+/]{10}[AEIMQUYcgkosw048]=$/.test(value) ||
      Buffer.from(value, "base64").length !== 8) {
    throw createValidationError([{
      field: "rowVersion",
      code: "EXPERIENCE_VERSION_INVALID",
      message: "A valid Experience version is required.",
    }]);
  }

  return value;
}

function isPlainObject(value) {
  return value !== null && typeof value === "object" && !Array.isArray(value);

function isValidProjectDate(value) {
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

export function parseProjectInput(value) {
  const errors = [];

  if (!isPlainObject(value)) {
    throw createValidationError([
      {
        field: "project",
        code: "PROJECT_OBJECT_REQUIRED",
        message: "Project data must be an object.",
      },
    ]);
  }

  for (const field of SERVER_OWNED_FIELDS) {
    if (Object.hasOwn(value, field)) {
      addError(
        errors,
        field,
        "PROJECT_SERVER_FIELD_FORBIDDEN",
        `${field} cannot be supplied by the client.`,
      );
    }
  }

  if (typeof value.id !== "string" || !PROJECT_ID_PATTERN.test(value.id)) {
    addError(
      errors,
      "id",
      "PROJECT_ID_INVALID",
      "A valid Project identifier is required.",
    );
  }

  if (
    typeof value.title !== "string" ||
    !value.title.trim() ||
    value.title.trim().length > 200
  ) {
    addError(
      errors,
      "title",
      "PROJECT_TITLE_INVALID",
      "Project title must contain 1 to 200 characters.",
    );
  }

  if (
    typeof value.role !== "string" ||
    !value.role.trim() ||
    value.role.trim().length > 160
  ) {
    addError(
      errors,
      "role",
      "PROJECT_ROLE_INVALID",
      "Project role must contain 1 to 160 characters.",
    );
  }

  if (!CATEGORIES.has(value.category)) {
    addError(
      errors,
      "category",
      "PROJECT_CATEGORY_INVALID",
      "Select a valid Project category.",
    );
  }

  if (!LIFECYCLE_STATUSES.has(value.lifecycleStatus)) {
    addError(
      errors,
      "lifecycleStatus",
      "PROJECT_LIFECYCLE_INVALID",
      "Select a valid lifecycle status.",
    );
  }

  if (!RECORD_STATUSES.has(value.recordStatus)) {
    addError(
      errors,
      "recordStatus",
      "PROJECT_RECORD_STATUS_INVALID",
      "Select a valid record status.",
    );
  }

  if (!isPlainObject(value.dates)) {
    addError(
      errors,
      "dates",
      "PROJECT_DATES_INVALID",
      "Project dates must be an object.",
    );
  } else {
    if (!isValidProjectDate(value.dates.startDate)) {
      addError(
        errors,
        "dates.startDate",
        "PROJECT_START_DATE_INVALID",
        "A valid Project start date is required.",
      );
    }

    if (value.dates.endDate && !isValidProjectDate(value.dates.endDate)) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_INVALID",
        "Enter a valid Project end date.",
      );
    }

    if (value.dates.isCurrent === true && value.dates.endDate) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_CURRENT_END_DATE_CONFLICT",
        "An ongoing Project cannot have an end date.",
      );
    }

    if (
      value.dates.isCurrent === true &&
      !["in-progress", "maintained"].includes(value.lifecycleStatus)
    ) {
      addError(
        errors,
        "lifecycleStatus",
        "PROJECT_CURRENT_LIFECYCLE_CONFLICT",
        "An ongoing Project must be In Progress or Maintained.",
      );
    }

    if (
      ["completed", "cancelled"].includes(value.lifecycleStatus) &&
      value.dates.isCurrent !== true &&
      !value.dates.endDate
    ) {
      addError(
        errors,
        "dates.endDate",
        "PROJECT_END_DATE_REQUIRED",
        "A completion or cancellation date is required.",
      );
    }

    if (
      isValidProjectDate(value.dates.startDate) &&
      isValidProjectDate(value.dates.endDate)
    ) {
      const start = new Date(
        value.dates.startDate.length === 7
          ? `${value.dates.startDate}-01`
          : value.dates.startDate,
      );

      const end = new Date(
        value.dates.endDate.length === 7
          ? `${value.dates.endDate}-01`
          : value.dates.endDate,
      );

      if (end < start) {
        addError(
          errors,
          "dates.endDate",
          "PROJECT_DATE_ORDER_INVALID",
          "End date cannot be earlier than start date.",
        );
      }
    }
  }

  if (
    !isPlainObject(value.presentation) ||
    typeof value.presentation.isFeatured !== "boolean"
  ) {
    addError(
      errors,
      "presentation.isFeatured",
      "PROJECT_FEATURED_INVALID",
      "Project featured status must be a boolean.",
    );
  }

  const urlFields = [
    ["organization.website", value.organization?.website],
    ...[
      "liveUrl",
      "repositoryUrl",
      "documentationUrl",
      "caseStudyUrl",
      "videoUrl",
    ].map((name) => [`links.${name}`, value.links?.[name]]),
  ];

  for (const [field, urlValue] of urlFields) {
    if (urlValue === undefined || urlValue === "") continue;

    let valid = false;

    if (typeof urlValue === "string") {
      try {
        const parsed = new URL(urlValue);
        valid = ["http:", "https:"].includes(parsed.protocol);
      } catch {
        valid = false;
      }
    }

    if (!valid) {
      addError(
        errors,
        field,
        "PROJECT_URL_INVALID",
        "Enter a valid HTTP or HTTPS URL.",
      );
    }
  }

  const collectionLimits = [
    ["problem.objectives", value.problem?.objectives, 25],
    ["problem.constraints", value.problem?.constraints, 25],
    ["solution.features", value.solution?.features, 40],
    ["solution.contributions", value.solution?.contributions, 40],
    ["results.metrics", value.results?.metrics, 25],
    ["skillRelationships", value.skillRelationships, 60],
    ["technologies", value.technologies, 60],
    ["awards", value.awards, 20],
    ["media", value.media, 30],
    ["supportingDocuments", value.supportingDocuments, 30],
    ...[
      "experienceRelationships",
      "educationRelationships",
      "trainingRelationships",
      "certificationRelationships",
    ].map((name) => [
      `relatedRecords.${name}`,
      value.relatedRecords?.[name],
      50,
    ]),
  ];

  for (const [field, items, maximum] of collectionLimits) {
    if (items === undefined) continue;

    if (!Array.isArray(items) || items.length > maximum) {
      addError(
        errors,
        field,
        "PROJECT_COLLECTION_INVALID",
        `This collection must contain at most ${maximum} items.`,
      );
    }
  }
  errors.push(...validateNestedProject(value));

  if (errors.length > 0) {
    throw createValidationError(errors);
  }

  return structuredClone(value);
}

export function parseProjectRowVersion(value) {
  if (typeof value !== "string" || !/^[A-Za-z0-9+/]{11}=$/.test(value)) {
    throw createValidationError([
      {
        field: "rowVersion",
        code: "PROJECT_VERSION_INVALID",
        message: "A valid Project version is required.",
      },
    ]);
  }

  return value;
}

import { generateExperienceOverviewWithGroq } from "../providers/groqExperienceOverviewProvider.js";

const EXPERIENCE_OVERVIEW_PROVIDERS = {
  groq: generateExperienceOverviewWithGroq,
};

function getAIProviderName() {

    .trim()
    .toLowerCase();
}

export async function generateExperienceOverview(context = {}) {
  const providerName = getAIProviderName();

  const generateWithProvider = EXPERIENCE_OVERVIEW_PROVIDERS[providerName];

  if (!generateWithProvider) {
    const error = new Error(
      `The "${providerName}" provider does not currently support experience-overview generation.`,
    );

    error.status = 503;

    error.publicMessage =
      'Experience Overview generation currently requires AI_PROVIDER="groq".';

    throw error;
  }

  return generateWithProvider(context);
}
