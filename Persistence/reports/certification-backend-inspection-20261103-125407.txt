const environment = getEnvironment();

const app = express();

/*
 * =========================================
 * Express Information Header
 * =========================================
 */

app.disable("x-powered-by");

/*
 * =========================================
 * Security Headers
 * =========================================
 */

app.use(helmet());

/*
 * =========================================
 * Allowed Frontend Origins
 * =========================================
 */

const allowedOrigins = environment.FRONTEND_URL.split(",")
  .map((origin) => origin.trim())
  .filter(Boolean);

/*
 * =========================================
 * CORS
 * =========================================
 */

app.use(
  cors({
    origin(origin, callback) {
      /*
       * Postman, curl and server-to-server
       * requests may not include Origin.
       */

      if (!origin) {
        return callback(null, true);
      }

      if (allowedOrigins.includes(origin)) {
        return callback(null, true);
      }

      const error = new Error(`Origin not allowed by CORS: ${origin}`);

      error.name = "CorsOriginError";

      error.code = "CORS_ORIGIN_DENIED";

      error.status = 403;

      error.publicMessage = "This website is not allowed to access the API.";

      return callback(error);
    },

    credentials: true,

    methods: ["GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"],

    allowedHeaders: ["Content-Type", "Authorization"],
  }),
);

/*
 * =========================================
 * Public Profile Pictures
 * =========================================
 */

if (environment.PROFILE_PICTURE_STORAGE_PATH) {
  app.use(
    "/uploads/profile-pictures",
    (request, response, next) => {
      /*
       * Profile pictures are intentionally public assets.
       * Allow the portfolio frontend to display them even
       * though the frontend and API use different origins.
       */

      response.setHeader("Cross-Origin-Resource-Policy", "cross-origin");

      next();
    },
    express.static(environment.PROFILE_PICTURE_STORAGE_PATH, {
      fallthrough: true,
      index: false,
      dotfiles: "deny",
    }),
  );
}

/*

  }),

app.get("/api/health/database", async (req, res) => {
  const health = await checkDatabaseHealth();

  const isHealthy = health.status === "ok";

  let message = "Database connection is unavailable.";

  if (isHealthy) {
    message = "Database connection is healthy.";
  } else if (health.status === "not-configured") {
    message = "Database service is not configured.";
  }

  return res.status(isHealthy ? 200 : 503).json({
    status: health.status,

    service: "portfolio-platform-database",

    configured: health.configured,

    responseTimeMs: health.responseTimeMs,

    message,

    timestamp: new Date().toISOString(),
  });
});

function setPrivateHeaders(response) {
  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",
    Pragma: "no-cache",
    Expires: "0",
  });
}

function projectRequestError(message) {
  const error = new Error(message);
  error.name = "ProjectRequestError";
  error.code = "PROJECT_REQUEST_INVALID";
  error.status = 400;
  error.publicMessage = message;
  return error;
}

function requireObject(value, name) {
  if (value === null || typeof value !== "object" || Array.isArray(value)) {
    throw projectRequestError(`${name} must be a JSON object.`);
  }

  return value;
}

function requireFields(body, allowedFields) {
  const input = requireObject(body, "Request body");

  const unexpected = Object.keys(input).filter(
    (field) => !allowedFields.includes(field),
  );

  if (unexpected.length > 0) {
    throw projectRequestError(
      `Unexpected request fields: ${unexpected.join(", ")}.`,
    );
  }

  return input;
}

import {
  listProjectsByUser,
  findProjectById,
  createProject,
  updateProject,
  deleteProject,
} from "../../repositories/projectRepository.js";

import {
  parseProjectInput,
  parseProjectRowVersion,
} from "../../schemas/projectSchemas.js";

const SERVER_OWNED_FIELDS = [
  "userId",
  "user_id",
  "createdAt",
  "updatedAt",
  "rowVersion",
];

function createProjectError({ message, code, status, publicMessage }) {
  const error = new Error(message);
  error.name = "ProjectServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = publicMessage;
  return error;
}

function assertUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw createProjectError({
      message: "An authenticated user ID is required.",
      code: "PROJECT_AUTHENTICATION_REQUIRED",
      status: 401,
      publicMessage: "You must sign in to access your Projects.",
    });
  }

  return userId.trim();
}

function assertProjectId(projectId) {
  if (
    typeof projectId !== "string" ||
    !projectId.trim() ||
    projectId.length > 200
  ) {
    throw createProjectError({
      message: "A valid Project ID is required.",
      code: "PROJECT_ID_INVALID",
      status: 400,
      publicMessage: "The Project identifier is invalid.",
    });
  }

  return projectId.trim();
}

function projectNotFound() {
  return createProjectError({
    message: "Project not found for authenticated user.",
    code: "PROJECT_NOT_FOUND",
    status: 404,
    publicMessage: "The requested Project could not be found.",
  });
}

function projectVersionConflict() {
  return createProjectError({
    message: "Project version conflict.",
    code: "PROJECT_VERSION_CONFLICT",
    status: 409,
    publicMessage:
      "This Project was changed elsewhere. Reload it before saving.",
  });
}

function projectIdMismatch() {
  return createProjectError({
    message: "The Project body ID does not match the requested Project ID.",
    code: "PROJECT_ID_MISMATCH",
    status: 400,
    publicMessage: "The Project identifiers do not match.",
  });
}

function withoutServerFields(project) {
  const result = { ...project };

  for (const field of SERVER_OWNED_FIELDS) {
    delete result[field];
  }

  return result;
}

export async function getAuthenticatedProjects({ userId }) {
  return listProjectsByUser(assertUserId(userId));
}

export async function getAuthenticatedProject({ userId, projectId }) {
  const project = await findProjectById(
    assertUserId(userId),
    assertProjectId(projectId),
  );

  if (!project) {
    throw projectNotFound();
  }

  return project;
}

export async function createAuthenticatedProject({ userId, project }) {
  const authenticatedUserId = assertUserId(userId);

  // Reject invalid data before any database operation.
  // Client-supplied server-owned fields remain forbidden.
  const validatedProject = parseProjectInput(project);

  try {
    return await createProject(authenticatedUserId, validatedProject);
  } catch (error) {
    if (error.number === 2627 || error.number === 2601) {
      throw createProjectError({
        message: "Project ID already exists.",
        code: "PROJECT_ALREADY_EXISTS",
        status: 409,
        publicMessage: "A Project with this identifier already exists.",
      });
    }

    throw error;
  }
}

export async function updateAuthenticatedProject({
  userId,
  projectId,
  project,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);

  // Validate the supplied version and full Project body.
  const expectedRowVersion = parseProjectRowVersion(rowVersion);
  const validatedProject = parseProjectInput(project);

  if (validatedProject.id !== normalizedProjectId) {
    throw projectIdMismatch();
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function setAuthenticatedProjectArchiveStatus({
  userId,
  projectId,
  rowVersion,
  archived,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  if (typeof archived !== "boolean") {
    throw createProjectError({
      message: "Archive status must be a boolean.",
      code: "PROJECT_ARCHIVE_STATUS_INVALID",
      status: 400,
      publicMessage: "A valid archive status is required.",
    });
  }

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  // Database-managed metadata is not editable Project content.
  const editableProject = withoutServerFields(existingProject);

  const validatedProject = parseProjectInput({
    ...editableProject,
    recordStatus: archived ? "archived" : "active",
  });

  // SQL Server must match the caller's row version.
  // Do not substitute the version from the latest database read.
  const updatedProject = await updateProject(
    authenticatedUserId,
    validatedProject,
    expectedRowVersion,
  );

  if (!updatedProject) {
    throw projectVersionConflict();
  }

  return updatedProject;
}

export async function deleteAuthenticatedProject({
  userId,
  projectId,
  rowVersion,
}) {
  const authenticatedUserId = assertUserId(userId);
  const normalizedProjectId = assertProjectId(projectId);
  const expectedRowVersion = parseProjectRowVersion(rowVersion);

  const existingProject = await findProjectById(
    authenticatedUserId,
    normalizedProjectId,
  );

  if (!existingProject) {
    throw projectNotFound();
  }

  const deleted = await deleteProject(
    authenticatedUserId,
    normalizedProjectId,
    expectedRowVersion,
  );

  if (!deleted) {
    throw projectVersionConflict();
  }

  return { deleted: true };
}

import { getDatabasePool, sql } from "../config/database.js";

const PROJECT_COLUMNS = `
  id,
  user_id,
  title,
  category,
  lifecycle_status,
  record_status,
  is_featured,
  project_data,
  created_at,
  updated_at,
  row_version
`;

function mapProjectRow(row) {
  if (!row) {
    return null;
  }

  const projectData =
    typeof row.project_data === "string"
      ? JSON.parse(row.project_data)
      : row.project_data;

  return {
    ...projectData,
    id: row.id,
    title: row.title,
    category: row.category,
    lifecycleStatus: row.lifecycle_status,
    recordStatus: row.record_status,
    presentation: {
      ...(projectData.presentation || {}),
      isFeatured: Boolean(row.is_featured),
    },
    createdAt: row.created_at,
    updatedAt: row.updated_at,
    rowVersion: Buffer.from(row.row_version).toString("base64"),
  };
}

function bindProjectParameters(request, userId, project) {
  return request
    .input("id", sql.NVarChar(200), project.id)
    .input("userId", sql.UniqueIdentifier, userId)
    .input("title", sql.NVarChar(500), project.title)
    .input("category", sql.NVarChar(100), project.category)
    .input("lifecycleStatus", sql.NVarChar(50), project.lifecycleStatus)
    .input("recordStatus", sql.NVarChar(20), project.recordStatus)
    .input("isFeatured", sql.Bit, Boolean(project.presentation?.isFeatured))
    .input("projectData", sql.NVarChar(sql.MAX), JSON.stringify(project));
}

export async function listProjectsByUser(userId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
      ORDER BY updated_at DESC, id ASC;
    `);

  return result.recordset.map(mapProjectRow);
}

export async function findProjectById(userId, projectId) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId).query(`
      SELECT ${PROJECT_COLUMNS}
      FROM dbo.projects
      WHERE user_id = @userId
        AND id = @id;
    `);

  return mapProjectRow(result.recordset[0]);
}

export async function createProject(userId, project) {
  const pool = await getDatabasePool();

  await bindProjectParameters(pool.request(), userId, project).query(`
    INSERT INTO dbo.projects (
      id,
      user_id,
      title,
      category,
      lifecycle_status,
      record_status,
      is_featured,
      project_data
    )
    VALUES (
      @id,
      @userId,
      @title,
      @category,
      @lifecycleStatus,
      @recordStatus,
      @isFeatured,
      @projectData
    );
  `);

  return findProjectById(userId, project.id);
}

export async function updateProject(userId, project, expectedRowVersion) {
  const pool = await getDatabasePool();

  const request = bindProjectParameters(pool.request(), userId, project).input(
    "expectedRowVersion",
    sql.Binary(8),
    Buffer.from(expectedRowVersion, "base64"),
  );

  const result = await request.query(`
    UPDATE dbo.projects
    SET
      title = @title,
      category = @category,
      lifecycle_status = @lifecycleStatus,
      record_status = @recordStatus,
      is_featured = @isFeatured,
      project_data = @projectData,
      updated_at = SYSUTCDATETIME()
    WHERE id = @id
      AND user_id = @userId
      AND row_version = @expectedRowVersion;
  `);

  if (result.rowsAffected[0] === 0) {
    return null;
  }

  return findProjectById(userId, project.id);
}

export async function deleteProject(userId, projectId, expectedRowVersion) {
  const pool = await getDatabasePool();

  const result = await pool
    .request()
    .input("userId", sql.UniqueIdentifier, userId)
    .input("id", sql.NVarChar(200), projectId)
    .input(
      "expectedRowVersion",
      sql.Binary(8),
      Buffer.from(expectedRowVersion, "base64"),
    ).query(`
      DELETE FROM dbo.projects
      WHERE id = @id
        AND user_id = @userId
        AND row_version = @expectedRowVersion;
    `);

  return result.rowsAffected[0] > 0;
}

function createValidationError(errors) {
  const error = new Error("Project validation failed.");

  error.name = "ProjectValidationError";
  error.code = "PROJECT_VALIDATION_FAILED";
  error.status = 400;
  error.publicMessage = "Please correct the Project information.";
  error.errors = errors;

  error.fieldErrors = {};

  for (const item of errors) {
    if (!error.fieldErrors[item.field]) {
      error.fieldErrors[item.field] = item.message;
    }
  }

  return error;
}

function addError(errors, field, code, message) {

function isPlainObject(value) {

function isValidProjectDate(value) {
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/,
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = match[3] ? Number(match[3]) : 1;

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

  }

  return structuredClone(value);

function checkItems(errors, source, path, callback) {

    if (!object(item)) {

      return;

    callback(item, base);
  });

function createAuthenticationError({
  message,

  status = 401,
  publicMessage = "You must sign in to access this resource.",
}) {
  const error = new Error(message);

  error.name = "AuthenticationError";

  error.code = code;

  error.status = status;

  error.publicMessage = publicMessage;

  return error;

function shouldClearSessionCookie(error) {
  const invalidSessionCodes = new Set([
    "SESSION_REFRESH_TOKEN_INVALID",
    "SESSION_NOT_FOUND",
    "SESSION_REFRESH_CONFLICT",
  ]);

  return invalidSessionCodes.has(error?.code);
}

    );

}

 * Normalize Roles
 * =========================================
 */

function normalizeRoles(roles) {
  if (!Array.isArray(roles)) {
    return [];
  }

  return [
    ...new Set(
      roles
        .map((role) =>
          typeof role === "string" ? role.trim().toLowerCase() : "",
        )
        .filter(Boolean),
    ),
  ];
}

 * =========================================
 * Create Request Authentication
 * =========================================
 */

function createRequestAuthentication({ user, roles, session }) {
  if (!user?.id || !session?.id) {
    throw createAuthenticationError({
      message:
        "The resolved session does not contain a valid user and session.",

      code: "AUTHENTICATION_CONTEXT_INVALID",

      status: 500,

      publicMessage: "Your session could not be verified.",
    });
  }

  const normalizedRoles = Object.freeze(normalizeRoles(roles));

  return Object.freeze({
    isAuthenticated: true,

    userId: user.id,

    sessionId: session.id,

    user,

    roles: normalizedRoles,

    session,
  });
}

/*
 * =========================================
 * Require Authentication
 * =========================================
 *
 * Resolves the HttpOnly refresh-token cookie,
 * validates the database session and places
 * public-safe identity information on:
 *
 * request.auth
 *
 * This middleware validates the session but
 * does not rotate its refresh token.
 */

export async function requireAuthentication(request, response, next) {
  /*
   * Protected responses must not be cached
   * or shared between authenticated users.
   */

  response.set({
    "Cache-Control": "no-store, no-cache, must-revalidate, private",

    Pragma: "no-cache",

    Expires: "0",
  });

  response.vary("Cookie");

  try {
    const refreshToken = getRefreshTokenFromRequest(request);

    if (!refreshToken) {
      throw createAuthenticationError({
        message: "The request does not contain a refresh-token cookie.",

        code: "AUTHENTICATION_REQUIRED",

        status: 401,

        publicMessage: "You must sign in to access this resource.",
      });
    }

    const authentication = await getAuthenticatedSession({
      refreshToken,
    });

    request.auth = createRequestAuthentication(authentication);

    return next();
  } catch (error) {
    if (shouldClearSessionCookie(error)) {
      safelyClearRefreshTokenCookie(response);
    }

    return next(error);
  }
}
