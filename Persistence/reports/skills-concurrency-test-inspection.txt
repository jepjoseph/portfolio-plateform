const ROW_VERSION = "AAAAAAAAAAE=";

const SKILL = {
  id: "skill-test-123",

  status: "active",
  rowVersion: ROW_VERSION,

function mockResponse(data, status = 200) {
  return new Response(
    JSON.stringify(data),
    {
      status,
      headers: {
        "Content-Type": "application/json",
      },
    },
  );
}

async function withMockFetch(handler, callback) {
  const originalFetch = globalThis.fetch;
  const requests = [];

  globalThis.fetch = async (url, options = {}) => {
    const request = {
      url: String(url),
      method: options.method,
      credentials: options.credentials,
      headers: options.headers,
      body: options.body,
    };

    requests.push(request);

    return handler(request);
  };

  try {
    await callback(requests);
  } finally {
    globalThis.fetch = originalFetch;
  }
}

function assertEndpoint(request, method, path) {
  assert.equal(request.method, method);
  assert.equal(new URL(request.url).pathname, `/api${path}`);
  assert.equal(request.credentials, "include");
}

function parseBody(request) {
  return JSON.parse(request.body);
}

import {
  apiDelete,
  apiGet,
  apiPatch,
  apiPost,
} from "../apiClient.js";

const SKILLS_ENDPOINT = "/skills";

function skillPath(skillId) {
  if (
    typeof skillId !== "string" ||
    !skillId.trim() ||
    skillId.trim().length > 200
  ) {
    throw new TypeError("A valid Skill ID is required.");
  }

  return `${SKILLS_ENDPOINT}/${encodeURIComponent(skillId.trim())}`;
}

function requireRowVersion(rowVersion) {
  if (typeof rowVersion !== "string" || !rowVersion.trim()) {
    throw new TypeError("A Skill rowVersion is required.");
  }

  return rowVersion.trim();
}

function requireSkillData(skillData) {
  if (
    !skillData ||
    typeof skillData !== "object" ||
    Array.isArray(skillData)
  ) {
    throw new TypeError("Skill data must be an object.");
  }

  return skillData;
}

function requireUpdates(updates) {
  if (
    !updates ||
    typeof updates !== "object" ||
    Array.isArray(updates)
  ) {
    throw new TypeError("Skill updates must be an object.");
  }

  return updates;
}

export async function fetchSkills(options = {}) {
  const response = await apiGet(SKILLS_ENDPOINT, options);
  return response.data.skills;
}

export async function fetchSkillById(skillId, options = {}) {
  const response = await apiGet(skillPath(skillId), options);
  return response.data.skill;
}

export async function createSkillApi(skillData, options = {}) {
  const response = await apiPost(
    SKILLS_ENDPOINT,
    requireSkillData(skillData),
    options,
  );

  return response.data.skill;
}

export async function updateSkillApi(
  skillId,
  updates,
  rowVersion,
  options = {},
) {
  const response = await apiPatch(
    skillPath(skillId),
    {
      updates: requireUpdates(updates),
      rowVersion: requireRowVersion(rowVersion),
    },
    options,
  );

  return response.data.skill;
}

export async function setSkillArchivedApi(
  skillId,
  archived,
  rowVersion,
  options = {},
) {
  if (typeof archived !== "boolean") {
    throw new TypeError("archived must be a boolean.");
  }

  const response = await apiPatch(
    `${skillPath(skillId)}/archive`,
    {
      archived,
      rowVersion: requireRowVersion(rowVersion),
    },
    options,
  );

  return response.data.skill;
}

export function archiveSkillApi(
  skillId,
  rowVersion,
  options = {},
) {
  return setSkillArchivedApi(
    skillId,
    true,
    rowVersion,
    options,
  );
}

export function restoreSkillApi(
  skillId,
  rowVersion,
  options = {},
) {
  return setSkillArchivedApi(
    skillId,
    false,
    rowVersion,
    options,
  );
}

export async function deleteSkillApi(
  skillId,
  rowVersion,
  options = {},
) {
  const response = await apiDelete(skillPath(skillId), {
    ...options,
    body: {
      rowVersion: requireRowVersion(rowVersion),
    },
  });

  return response.data;
}

import {
  fetchSkills,
  fetchSkillById,
  createSkillApi,
  updateSkillApi,
  archiveSkillApi,
  restoreSkillApi,
} from "./skillApiService.js";

import {
  createEmptySkill,
  findMatchingSkill,
  normalizeSkillNameForComparison,
} from "../../models/skillModel.js";

import {
  validateSkill,
  validateSkillNameAvailability,
} from "./skillValidation.js";

function makeError(message, code, status = 400) {
  const error = new Error(message);
  error.code = code;
  error.status = status;
  error.publicMessage = message;
  return error;
}

function requireVersion(skill) {
  if (!skill?.rowVersion) {
    throw makeError(
      "The skill must be refreshed before it can be changed.",
      "SKILL_ROW_VERSION_REQUIRED",
      409,
    );
  }

  return skill.rowVersion;
}

function assertSkillCollection(skills) {
  if (!Array.isArray(skills)) {
    throw makeError(
      "The server returned an invalid Skill Library.",
      "SKILL_RESPONSE_INVALID",
      502,
    );
  }

  const ids = new Set();

  for (const skill of skills) {
    if (
      !skill ||
      typeof skill.id !== "string" ||
      typeof skill.name !== "string" ||
      typeof skill.rowVersion !== "string" ||
      !skill.rowVersion.trim() ||
      ids.has(skill.id)
    ) {
      throw makeError(
        "The server returned an invalid or duplicate Skill.",
        "SKILL_RESPONSE_INVALID",
        502,
      );
    }

    ids.add(skill.id);
  }

  return skills;
}

export async function getSkills({
  includeArchived = false,
  category = "all",
  type = "all",
  query = "",
} = {}) {
  const skills = assertSkillCollection(
    await fetchSkills({
      query: { includeArchived: true },
    }),
  );

  const search = normalizeSkillNameForComparison(query);

  return skills.filter((skill) => {
    if (!includeArchived && skill.status === "archived") return false;
    if (category !== "all" && skill.category !== category) return false;
    if (type !== "all" && skill.type !== type) return false;

    if (!search) return true;

    return [
      skill.name,
      skill.description,
      skill.category,
      skill.type,
      ...(skill.aliases || []),
    ].some((value) =>
      normalizeSkillNameForComparison(value).includes(search),
    );
  });
}

export async function getSkillById(skillId) {
  return fetchSkillById(skillId);
}

export async function createSkill(skillData) {
  return createSkillApi(skillData);
}

export async function updateSkill(skillId, updates = {}, rowVersion) {
  return updateSkillApi(

    requireVersion({ rowVersion }),
  );
}

export async function archiveSkill(skillId, rowVersion) {

}

export async function restoreSkill(skillId, rowVersion) {
  return restoreSkillApi(skillId, requireVersion({ rowVersion }));
}

export async function resolveSkillMatch(name) {
  const skills = await getSkills({ includeArchived: true });
  const matchingSkill = findMatchingSkill(skills, name);

  if (!matchingSkill) {
    return {
      matched: false,
      matchedBy: "",
      skill: null,
    };
  }

  const requested = normalizeSkillNameForComparison(name);
  const primary = normalizeSkillNameForComparison(matchingSkill.name);

  return {
    matched: true,
    matchedBy: requested === primary ? "name" : "alias",
    skill: matchingSkill,
  };
}

export async function findOrCreateSkill(
  skillData,
  { restoreArchived = true } = {},
) {
  const candidate =
    typeof skillData === "string"
      ? { name: skillData }
      : skillData && typeof skillData === "object"
        ? skillData
        : {};

  const name = String(candidate.name || "").trim();

  if (!name) {
    throw makeError("Enter a skill name.", "SKILL_NAME_REQUIRED");
  }

  const match = await resolveSkillMatch(name);

  if (match.matched) {
    const restored =
      restoreArchived && match.skill.status === "archived";

    const skill = restored
      ? await restoreSkill(match.skill.id, match.skill.rowVersion)
      : match.skill;

    return {
      skill,
      created: false,
      restored,
      matched: true,
      matchedBy: match.matchedBy,
    };
  }

  const skill = await createSkill({
    ...candidate,
    name,
    source: candidate.source || "manual",
    sourceContext: candidate.sourceContext || "experience",
  });

  return {
    skill,
    created: true,
    restored: false,
    matched: false,
    matchedBy: "",
  };
}

export async function validateSkillDraft(
  skillData,
  { excludeSkillId = "" } = {},
) {
  const skills = await getSkills({ includeArchived: true });

  const defaults = createEmptySkill();

  const candidate = {
    ...defaults,
    ...skillData,
    proficiency: {
      ...defaults.proficiency,
      ...skillData?.proficiency,
    },
    language: {
      ...defaults.language,
      ...skillData?.language,
    },
  };

  return validateSkill(candidate, {
    existingSkills: skills,
    excludeSkillId,
  });
}

export async function checkSkillNameAvailability(
  name,
  { excludeSkillId = "" } = {},
) {
  const skills = await getSkills({ includeArchived: true });

  return validateSkillNameAvailability(
    name,
    skills,
    excludeSkillId,
  );
}
