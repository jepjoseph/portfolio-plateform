  55: function normalizeNullableNumber(value) {
  56:   if (value === "" || value === null || value === undefined) {
  57:     return null;
  58:   }
  59: 
  60:   const numberValue = Number(value);
  61: 
  62:   if (!Number.isFinite(numberValue)) {
  63:     return null;
  64:   }
  65: 
  66:   return Math.max(0, Math.trunc(numberValue));
  67: }

  69: function normalizeDate(value) {
  70:   if (typeof value !== "string") {
  71:     return "";
  72:   }
  73: 
  74:   const trimmedValue = value.trim();
  75: 
  76:   if (!trimmedValue) {
  77:     return "";
  78:   }
  79: 
  80:   /*
  81:    * The form uses YYYY-MM values, but this also
  82:    * permits YYYY-MM-DD records imported later.
  83:    */
  84: 
  85:   if (
  86:     !/^\d{4}-(0[1-9]|1[0-2])(?:-(0[1-9]|[12]\d|3[01]))?$/.test(trimmedValue)
  87:   ) {
  88:     return "";
  89:   }
  90: 
  91:   return trimmedValue;
  92: }
