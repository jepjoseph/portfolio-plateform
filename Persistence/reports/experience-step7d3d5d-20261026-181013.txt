   41:     "SESSION_REFRESH_TOKEN_INVALID",
   42:     "SESSION_NOT_FOUND",

   73:   if (!Array.isArray(roles)) {
   74:     return [];
   75:   }
   76: 
   77:   return [
   78:     ...new Set(
   79:       roles
   80:         .map((role) =>

   82:         )
   83:         .filter(Boolean),

   88: /*
   89:  * =========================================
   90:  * Create Request Authentication
   91:  * =========================================
   92:  */
   93: 
   94: function createRequestAuthentication({ user, roles, session }) {
   95:   if (!user?.id || !session?.id) {
   96:     throw createAuthenticationError({
   97:       message:
   98:         "The resolved session does not contain a valid user and session.",
   99: 
  100:       code: "AUTHENTICATION_CONTEXT_INVALID",
  101: 
  102:       status: 500,
  103: 
  104:       publicMessage: "Your session could not be verified.",
  105:     });
  106:   }
  107: 
  108:   const normalizedRoles = Object.freeze(normalizeRoles(roles));
  109: 
  110:   return Object.freeze({
  111:     isAuthenticated: true,
  112: 
  113:     userId: user.id,
  114: 
  115:     sessionId: session.id,
  116: 
  117:     user,
  118: 
  119:     roles: normalizedRoles,
  120: 
  121:     session,
  122:   });
  123: }
  124: 
  125: /*
  126:  * =========================================
  127:  * Require Authentication
  128:  * =========================================
  129:  *
  130:  * Resolves the HttpOnly refresh-token cookie,
  131:  * validates the database session and places
  132:  * public-safe identity information on:
  133:  *
  134:  * request.auth
  135:  *
  136:  * This middleware validates the session but
  137:  * does not rotate its refresh token.
  138:  */
  139: 
  140: export async function requireAuthentication(request, response, next) {
  141:   /*
  142:    * Protected responses must not be cached
  143:    * or shared between authenticated users.
  144:    */
  145: 
  146:   response.set({
  147:     "Cache-Control": "no-store, no-cache, must-revalidate, private",
  148: 
  149:     Pragma: "no-cache",
  150: 
  151:     Expires: "0",
  152:   });
  153: 
  154:   response.vary("Cookie");
  155: 
  156:   try {
  157:     const refreshToken = getRefreshTokenFromRequest(request);
  158: 
  159:     if (!refreshToken) {
  160:       throw createAuthenticationError({
  161:         message: "The request does not contain a refresh-token cookie.",
  162: 
  163:         code: "AUTHENTICATION_REQUIRED",
  164: 
  165:         status: 401,
  166: 
  167:         publicMessage: "You must sign in to access this resource.",
  168:       });
  169:     }
  170: 
  171:     const authentication = await getAuthenticatedSession({
  172:       refreshToken,
  173:     });
  174: 
  175:     request.auth = createRequestAuthentication(authentication);
  176: 
  177:     return next();
  178:   } catch (error) {
  179:     if (shouldClearSessionCookie(error)) {
  180:       safelyClearRefreshTokenCookie(response);
  181:     }
  182: 
  183:     return next(error);
  184:   }
  185: }
