 356:       .filter(Boolean),
 357:   ]);
 358: 
 359:   const removableStorageKeys = [
 360:     ...new Set(getArray(requestedStorageKeys).map(getText).filter(Boolean)),
 361:   ].filter((storageKey) => !referencedStorageKeys.has(storageKey));
 362: 
 363:   if (removableStorageKeys.length === 0) {
 364:     return 0;
 365:   }
 366: 
 367:   return deleteProjectAssets(removableStorageKeys);
 368: }

 430: /*
 431:  * =========================================
 432:  * Create Project
 433:  * =========================================
 434:  */
 435: 
 436: export async function createProject(
 437:   projectData,
 438:   {
 439:     collections = {},
 440:     mediaUploads = [],
 441:     documentUploads = [],
 442:     removedAssetStorageKeys = [],
 443:   } = {},
 444: ) {
 445:   const records = readStoredProjects();
 446: 
 447:   let project = createProjectModel(projectData);
 448: 
 449:   ensureUniqueProject(records, project);
 450: 
 451:   const preparedAssets = await prepareProjectAssets(project, {
 452:     mediaUploads,
 453:     documentUploads,
 454:   });
 455: 
 456:   project = preparedAssets.project;
 457: 
 458:   try {
 459:     const validation = validateForSave(project, collections);
 460: 
 461:     /*
 462:      * Store metadata only after every staged
 463:      * asset was written successfully.
 464:      */
 465: 
 466:     replaceStoredProjects([...records, project]);
 467: 
 468:     /*
 469:      * This collection is normally empty while
 470:      * creating, but accepting it keeps the
 471:      * service contract consistent.
 472:      */
 473: 
 474:     await removeUnreferencedAssets(project, removedAssetStorageKeys).catch(
 475:       () => {},
 476:     );
 477: 
 478:     return {
 479:       project,
 480:       warnings: validation.warnings,
 481:     };
 482:   } catch (error) {
 483:     /*
 484:      * Metadata was not saved, so every new
 485:      * IndexedDB asset must be removed.
 486:      */
 487: 
 488:     await deleteProjectAssets(preparedAssets.createdStorageKeys).catch(
 489:       () => {},
 490:     );
 491: 
 492:     throw error;
 493:   }
 494: }

 496: /*
 497:  * =========================================
 498:  * Update Project
 499:  * =========================================
 500:  */
 501: 
 502: export async function updateProject(
 503:   projectId,
 504:   updates,
 505:   {
 506:     collections = {},
 507:     mediaUploads = [],
 508:     documentUploads = [],
 509:     removedAssetStorageKeys = [],
 510:   } = {},
 511: ) {
 512:   const normalizedProjectId = requireProjectId(projectId);
 513: 
 514:   const records = readStoredProjects();
 515: 
 516:   const projectIndex = findProjectIndex(records, normalizedProjectId);
 517: 
 518:   if (projectIndex < 0) {
 519:     throw createProjectServiceError(
 520:       "PROJECT_NOT_FOUND",
 521:       `Project not found: ${normalizedProjectId}`,
 522:       "The selected Project could not be found.",
 523:       {
 524:         projectId: normalizedProjectId,
 525:       },
 526:     );
 527:   }
 528: 
 529:   let project = updateProjectModel(records[projectIndex], updates);
 530: 
 531:   ensureUniqueProject(records, project, project.id);
 532: 
 533:   const preparedAssets = await prepareProjectAssets(project, {
 534:     mediaUploads,
 535:     documentUploads,
 536:   });
 537: 
 538:   project = preparedAssets.project;
 539: 
 540:   try {
 541:     const validation = validateForSave(project, collections);
 542: 
 543:     const nextRecords = [...records];
 544: 
 545:     nextRecords[projectIndex] = project;
 546: 
 547:     /*
 548:      * Save updated metadata before deleting
 549:      * old assets. If this write fails, existing
 550:      * files remain recoverable.
 551:      */
 552: 
 553:     replaceStoredProjects(nextRecords);
 554: 
 555:     /*
 556:      * Asset deletion occurs only after the new
 557:      * Project record has been saved.
 558:      */
 559: 
 560:     await removeUnreferencedAssets(project, removedAssetStorageKeys).catch(
 561:       () => {},
 562:     );
 563: 
 564:     return {
 565:       project,
 566:       warnings: validation.warnings,
 567:     };
 568:   } catch (error) {
 569:     /*
 570:      * Preserve all pre-existing files and remove
 571:      * only assets created during this attempt.
 572:      */
 573: 
 574:     await deleteProjectAssets(preparedAssets.createdStorageKeys).catch(
 575:       () => {},
 576:     );
 577: 
 578:     throw error;
 579:   }
 580: }

 588: export async function archiveProject(projectId) {
 589:   const normalizedProjectId = requireProjectId(projectId);
 590: 
 591:   const records = readStoredProjects();
 592: 
 593:   const projectIndex = findProjectIndex(records, normalizedProjectId);
 594: 
 595:   if (projectIndex < 0) {
 596:     throw createProjectServiceError(
 597:       "PROJECT_NOT_FOUND",
 598:       `Project not found: ${normalizedProjectId}`,
 599:       "The selected Project could not be found.",
 600:       {
 601:         projectId: normalizedProjectId,
 602:       },
 603:     );
 604:   }
 605: 
 606:   const project = records[projectIndex];
 607: 
 608:   if (project.recordStatus === "archived") {
 609:     return project;
 610:   }
 611: 
 612:   const archivedProject = normalizeProject({
 613:     ...project,
 614: 
 615:     recordStatus: "archived",
 616: 
 617:     updatedAt: new Date().toISOString(),
 618:   });
 619: 
 620:   const nextRecords = [...records];
 621: 
 622:   nextRecords[projectIndex] = archivedProject;
 623: 
 624:   replaceStoredProjects(nextRecords);
 625: 
 626:   return archivedProject;
 627: }

 676: /*
 677:  * =========================================
 678:  * Delete Project Permanently
 679:  * =========================================
 680:  */
 681: 
 682: export async function deleteProject(projectId) {
 683:   const normalizedProjectId = requireProjectId(projectId);
 684: 
 685:   const records = readStoredProjects();
 686: 
 687:   const projectIndex = findProjectIndex(records, normalizedProjectId);
 688: 
 689:   if (projectIndex < 0) {
 690:     throw createProjectServiceError(
 691:       "PROJECT_NOT_FOUND",
 692:       `Project not found: ${normalizedProjectId}`,
 693:       "The selected Project could not be found.",
 694:       {
 695:         projectId: normalizedProjectId,
 696:       },
 697:     );
 698:   }
 699: 
 700:   const deletedProject = records[projectIndex];
 701: 
 702:   const nextRecords = records.filter(
 703:     (project) => project.id !== normalizedProjectId,
 704:   );
 705: 
 706:   /*
 707:    * Delete metadata first. If localStorage
 708:    * fails, the Project and all assets remain.
 709:    */
 710: 
 711:   replaceStoredProjects(nextRecords);
 712: 
 713:   /*
 714:    * Once metadata is removed, delete every
 715:    * IndexedDB asset owned by this Project.
 716:    *
 717:    * Cleanup is best effort because the Project
 718:    * metadata has already been deleted.
 719:    */
 720: 
 721:   let assetCleanupError = null;
 722: 
 723:   try {
 724:     await deleteProjectAssetsForProject(normalizedProjectId);
 725:   } catch (error) {
 726:     assetCleanupError = error;
 727:   }
 728: 
 729:   return {
 730:     ...deletedProject,
 731: 
 732:     assetCleanupWarning: assetCleanupError
 733:       ? "The Project was deleted, but one or more locally stored files could not be cleaned up."
 734:       : "",
 735:   };
 736: }
