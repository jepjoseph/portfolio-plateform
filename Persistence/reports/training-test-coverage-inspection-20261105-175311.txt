function trainingError(message, code, status, errors) {
  const error = new Error(message);
  error.name = "TrainingServiceError";
  error.code = code;
  error.status = status;
  error.publicMessage = message;

  if (errors) error.errors = errors;

  return error;
}

function requireUserId(userId) {
  if (typeof userId !== "string" || !userId.trim()) {
    throw trainingError(
      "Authentication is required.",
      "TRAINING_AUTH_REQUIRED",
      401,
    );
  }

  return userId.trim();
}

function requireTrainingId(trainingId) {
  if (
    typeof trainingId !== "string" ||
    !trainingId.trim() ||
    trainingId.length > 200
  ) {
    throw trainingError(
      "A valid Training identifier is required.",
      "TRAINING_ID_INVALID",
      400,
    );
  }

  return trainingId.trim();
}

function requireObject(value, field) {
  if (
    value === null ||
    typeof value !== "object" ||
    Array.isArray(value)
  ) {
    throw trainingError(
      `${field} must be a JSON object.`,
      "TRAINING_REQUEST_INVALID",
      400,
    );
  }

  return value;
}

function validateTraining(training) {
  requireObject(training, "Training");

    throw trainingError(
      "Training validation failed.",
      "TRAINING_VALIDATION_FAILED",

      formatTrainingSchemaErrors(result.error),
    );
  }

  return result.data;
}

function validateTrainingUpdates(updates) {
  const result = updateTrainingSchema.safeParse(updates);

  if (!result.success) {
    throw trainingError(
      "Training validation failed.",
      "TRAINING_VALIDATION_FAILED",
      400,
      formatTrainingSchemaErrors(result.error),
    );
  }

  return result.data;
}

function mergeTrainingUpdates(existing, changes) {
  const candidate = {
    ...existing,
    ...changes,
  };

  if (changes.provider) {
    candidate.provider = {
      ...existing.provider,
      ...changes.provider,
    };
  }

  if (changes.delivery) {
    candidate.delivery = {
      ...existing.delivery,
      ...changes.delivery,
      location: {
        ...existing.delivery.location,
        ...(changes.delivery.location || {}),
      },
    };
  }

  for (const field of [
    "dates",
    "completion",
    "visibility",
    "privateInformation",
  ]) {
    if (changes[field]) {
      candidate[field] = {
        ...existing[field],
        ...changes[field],
      };
    }
  }

  return candidate;
}

function requireValidRowVersion(value) {
  if (
    typeof value !== "string" ||
    !/^[A-Za-z0-9+/]{11}=$/.test(value)
  ) {
    throw trainingError(
      "A valid Training version is required.",
      "TRAINING_VERSION_INVALID",
      400,
    );
  }

  const bytes = Buffer.from(value, "base64");

  if (
    bytes.length !== 8 ||
    bytes.toString("base64") !== value
  ) {
    throw trainingError(
      "A valid Training version is required.",
      "TRAINING_VERSION_INVALID",
      400,
    );
  }

  return value;
}

function requireNoNewDocuments(documents) {
  if (documents !== undefined) {
    if (!Array.isArray(documents)) {
      throw trainingError(
        "Supporting documents must be an array.",
        "TRAINING_DOCUMENTS_INVALID",
        400,
      );
    }

    if (documents.length > 0) {
      throw trainingError(
        "Training document uploads are not available yet.",
        "TRAINING_DOCUMENTS_NOT_READY",
        400,
      );
    }
  }
}

function requireNoRelationshipUpdates(updates) {

    "relatedCertificationIds",
    "relatedEducationIds",
    "relatedExperienceIds",
    "relatedProjectIds",

      );
    }
  }
}

    ...record,

    updatedAt: new Date(record.updatedAt).toISOString(),

}

export async function createAuthenticatedTraining({
  userId,
  training,
}) {
  const authenticatedUserId = requireUserId(userId);
  const input = requireObject(training, "Training");

  requireNoNewDocuments(input.supportingDocuments);

  const now = new Date().toISOString();

  // Remove backend-managed metadata from the client payload.
  const {
    rowVersion: _clientRowVersion,
    userId: _clientUserId,
    user_id: _clientUserIdSql,
    ...clientModel
  } = input;

  const candidate = validateTraining({
    ...clientModel,
    id: randomUUID(),
    createdAt: now,
    updatedAt: now,
  });

  try {
    const created = await createTrainingRecord(
      authenticatedUserId,
      candidate,
    );

    return normalizeRecord(created);
  } catch (error) {
    const number =
      error.number ??
      error.originalError?.number ??
      error.cause?.number;

    if (number === 2601 || number === 2627) {
      throw trainingError(
        "A Training identifier already exists.",
        "TRAINING_ID_CONFLICT",
        409,
      );
    }

    throw error;
  }
}

export async function updateAuthenticatedTraining({
  userId,
  trainingId,
  updates,
  rowVersion,
}) {
  const authenticatedUserId = requireUserId(userId);
  const id = requireTrainingId(trainingId);
  const expectedVersion = requireValidRowVersion(rowVersion);
  const changes = requireObject(updates, "Training updates");

  if (!Object.keys(changes).length) {
    throw trainingError(
      "Provide at least one Training field to update.",
      "TRAINING_REQUEST_INVALID",
      400,
    );
  }

  const forbidden = [
    "id",
    "userId",
    "user_id",
    "createdAt",
    "updatedAt",
    "rowVersion",
    "modelVersion",
  ];

  for (const field of forbidden) {
    if (Object.hasOwn(changes, field)) {
      throw trainingError(
        `${field} cannot be updated.`,
        "TRAINING_SERVER_FIELD_FORBIDDEN",
        400,
      );
    }
  }

  requireNoRelationshipUpdates(changes);

  if (Object.hasOwn(changes, "supportingDocuments")) {
    throw trainingError(
      "Updating Training documents is not available yet.",
      "TRAINING_DOCUMENTS_NOT_READY",
      400,
    );
  }

  // Reject unsupported top-level fields before database access.
  const allowedFields = new Set([
    "title",
    "trainingType",
    "provider",
    "delivery",
    "dates",
    "completion",
    "description",
    "instructors",
    "topics",
    "learningOutcomes",
    "visibility",
    "privateInformation",
    "source",
    "sourceContext",
    "status",
  ]);

  for (const field of Object.keys(changes)) {
    if (!allowedFields.has(field)) {
      throw trainingError(
        "Training validation failed.",
        "TRAINING_VALIDATION_FAILED",
        400,
        [{
          field,
          code: "unrecognized_keys",
          message: "Unsupported Training field.",
        }],
      );
    }
  }

  const existing = await findTrainingById(
    authenticatedUserId,
    id,
  );

  if (!existing) throw notFound();

  if (existing.rowVersion !== expectedVersion) {
    throw versionConflict();
  }

  const {
    rowVersion: _storedVersion,
    ...storedModel
  } = normalizeRecord(existing);

  const merged = mergeTrainingUpdates(storedModel, changes);

  // Validate PATCH fields after merging nested structures.
  const validatedChanges = validateTrainingUpdates(
    Object.fromEntries(
      Object.keys(changes).map((field) => [
        field,
        merged[field],
      ]),
    ),
  );

  const candidate = validateTraining({
    ...mergeTrainingUpdates(storedModel, validatedChanges),
    id: storedModel.id,
    createdAt: storedModel.createdAt,
    updatedAt: new Date().toISOString(),
  });

  const updated = await updateTrainingRecord(
    authenticatedUserId,
    candidate,
    expectedVersion,
  );

  if (!updated) {
    const current = await findTrainingById(
      authenticatedUserId,
      id,
    );

    if (!current) throw notFound();

    throw versionConflict();
  }

  return normalizeRecord(updated);
}

function validateRules(value, context) {
  const issue = (path, message) => context.addIssue({
    code: "custom",
    path,
    message,
  });

  if (value.dates.isCurrent && value.dates.endDate) {
    issue(["dates", "endDate"], "Current training cannot have an end date.");
  }

  if (
    value.dates.endDate &&
    dateBounds(value.dates.endDate).end <
      dateBounds(value.dates.startDate).start
  ) {
    issue(["dates", "endDate"], "End date precedes start date.");
  }

  if (
    value.completion.status === "in-progress" &&
    !value.dates.isCurrent
  ) {
    issue(["dates", "isCurrent"], "In-progress training must be current.");
  }

  if (
    new Date(value.updatedAt).getTime() <
    new Date(value.createdAt).getTime()
  ) {
    issue(["updatedAt"], "Updated timestamp precedes creation.");
  }

  function unique(items, path, getKey) {
    const seen = new Set();

    items.forEach((item, index) => {
      const key = getKey(item);

      if (!key) return;

      const normalized = String(key).normalize("NFKC").toLowerCase();

      if (seen.has(normalized)) {
        issue([...path, index], "Duplicate item.");
      }

      seen.add(normalized);
    });
  }

  for (const field of [
    "instructors", "topics", "learningOutcomes",
    "skillRelationships", "certificationRelationships",
    "supportingDocuments",
  ]) {
    unique(value[field], [field], (item) => item.id);
  }

  unique(
    value.skillRelationships,
    ["skillRelationships"],
    (item) => item.skillId || item.nameSnapshot,
  );

  unique(
    value.certificationRelationships,
    ["certificationRelationships"],
    (item) => item.certificationId,
  );

  for (const field of [
    "relatedEducationIds", "relatedCertificationIds",
    "relatedExperienceIds", "relatedProjectIds",
  ]) {
    unique(value[field], [field], (item) => item);
  }

  const canonicalIds = value.certificationRelationships.map(
    (item) => item.certificationId,
  );

  if (
    JSON.stringify(canonicalIds) !==
    JSON.stringify(value.relatedCertificationIds)
  ) {
    issue(
      ["relatedCertificationIds"],
      "Certification IDs must match canonical relationships.",
    );
  }
}
