 312: export async function updateAuthenticatedTraining({
 313:   userId,
 314:   trainingId,
 315:   updates,
 316:   rowVersion,
 317: }) {
 318:   const authenticatedUserId = requireUserId(userId);
 319:   const id = requireTrainingId(trainingId);
 320:   const expectedVersion = requireValidRowVersion(rowVersion);
 321:   const changes = requireObject(updates, "Training updates");
 322: 
 323:   if (!Object.keys(changes).length) {
 324:     throw trainingError(
 325:       "Provide at least one Training field to update.",
 326:       "TRAINING_REQUEST_INVALID",
 327:       400,
 328:     );
 329:   }
 330: 
 331:   const forbidden = [
 332:     "id",
 333:     "userId",
 334:     "user_id",
 335:     "createdAt",
 336:     "updatedAt",
 337:     "rowVersion",
 338:     "modelVersion",
 339:   ];
 340: 
 341:   for (const field of forbidden) {
 342:     if (Object.hasOwn(changes, field)) {
 343:       throw trainingError(
 344:         `${field} cannot be updated.`,
 345:         "TRAINING_SERVER_FIELD_FORBIDDEN",
 346:         400,
 347:       );
 348:     }
 349:   }
 350: 
 351:   requireNoRelationshipUpdates(changes);
 352: 
 353:   if (Object.hasOwn(changes, "supportingDocuments")) {
 354:     throw trainingError(
 355:       "Updating Training documents is not available yet.",
 356:       "TRAINING_DOCUMENTS_NOT_READY",
 357:       400,
 358:     );
 359:   }
 360: 
 361:   // Reject unsupported top-level fields before database access.
 362:   const allowedFields = new Set([
 363:     "title",
 364:     "trainingType",
 365:     "provider",
 366:     "delivery",
 367:     "dates",
 368:     "completion",
 369:     "description",
 370:     "instructors",
 371:     "topics",
 372:     "learningOutcomes",
 373:     "visibility",
 374:     "privateInformation",
 375:     "source",
 376:     "sourceContext",
 377:     "status",
 378:   ]);
 379: 
 380:   for (const field of Object.keys(changes)) {
 381:     if (!allowedFields.has(field)) {
 382:       throw trainingError(
 383:         "Training validation failed.",
 384:         "TRAINING_VALIDATION_FAILED",
 385:         400,
 386:         [{
 387:           field,
 388:           code: "unrecognized_keys",
 389:           message: "Unsupported Training field.",
 390:         }],
 391:       );
 392:     }
 393:   }
 394: 
 395:   const existing = await findTrainingById(
 396:     authenticatedUserId,
 397:     id,
 398:   );
 399: 
 400:   if (!existing) throw notFound();
 401: 
 402:   if (existing.rowVersion !== expectedVersion) {
 403:     throw versionConflict();
 404:   }
 405: 
 406:   const {
 407:     rowVersion: _storedVersion,
 408:     ...storedModel
 409:   } = normalizeRecord(existing);
 410: 
 411:   const merged = mergeTrainingUpdates(storedModel, changes);
 412: 
 413:   // Validate PATCH fields after merging nested structures.
 414:   const validatedChanges = validateTrainingUpdates(
 415:     Object.fromEntries(
 416:       Object.keys(changes).map((field) => [
 417:         field,
 418:         merged[field],
 419:       ]),
 420:     ),
 421:   );
 422: 
 423:   const candidate = validateTraining({
 424:     ...mergeTrainingUpdates(storedModel, validatedChanges),
 425:     id: storedModel.id,
 426:     createdAt: storedModel.createdAt,
 427:     updatedAt: new Date().toISOString(),
 428:   });
 429: 
 430:   const updated = await updateTrainingRecord(
 431:     authenticatedUserId,
 432:     candidate,
 433:     expectedVersion,
 434:   );
 435: 
 436:   if (!updated) {
 437:     const current = await findTrainingById(
 438:       authenticatedUserId,
 439:       id,
 440:     );
 441: 
 442:     if (!current) throw notFound();
 443: 
 444:     throw versionConflict();
 445:   }
 446: 
 447:   return normalizeRecord(updated);
 448: }
