function setPrivateHeaders(response) {

function projectRequestError(message) {
  const error = new Error(message);
  error.name = "ProjectRequestError";
  error.code = "PROJECT_REQUEST_INVALID";
  error.status = 400;
  error.publicMessage = message;
  return error;
}

function requireObject(value, name) {
  if (value === null || typeof value !== "object" || Array.isArray(value)) {
    throw projectRequestError(`${name} must be a JSON object.`);
  }

  return value;
}

function requireFields(body, allowedFields) {
  const input = requireObject(body, "Request body");

  const unexpected = Object.keys(input).filter(
    (field) => !allowedFields.includes(field),
  );

  if (unexpected.length > 0) {
    throw projectRequestError(
      `Unexpected request fields: ${unexpected.join(", ")}.`,
    );
  }

  return input;
}

async function ensureSafeUploadDirectory(userId, projectId) {
  const root = getProjectMediaStorageRoot();
  const destination = getProjectMediaDirectory(userId, projectId);

  await fs.mkdir(root, { recursive: true });

  const relative = path.relative(root, destination);

  if (
    !relative ||
    relative === ".." ||
    relative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(relative)
  ) {
    throw new Error("Project media upload destination is invalid.");
  }

  const directories = [
    root,
    path.dirname(destination),
    destination,
  ];

  for (const directory of directories) {
    try {
      await fs.mkdir(directory);
    } catch (error) {
      if (error?.code !== "EEXIST") {
        throw error;
      }
    }

    const stats = await fs.lstat(directory);

    if (stats.isSymbolicLink() || !stats.isDirectory()) {
      throw new Error(
        "Project media upload directory is not a regular directory.",
      );
    }
  }

  const canonicalRoot = await fs.realpath(root);
  const canonicalDestination = await fs.realpath(destination);
  const canonicalRelative = path.relative(
    canonicalRoot,
    canonicalDestination,
  );

  if (
    !canonicalRelative ||
    canonicalRelative === ".." ||
    canonicalRelative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(canonicalRelative)
  ) {
    throw new Error("Project media upload destination escapes storage root.");
  }

  return destination;
}

import fs from "node:fs/promises";
import path from "node:path";

function createValidationError(message, code, status = 400) {
  const error = new Error(message);

  error.name = "ProjectMediaValidationError";
  error.code = code;
  error.status = status;
  error.publicMessage = message;

  return error;
}

function getAuthorizedFilePath(request) {
  const userId = String(request.auth?.userId ?? "").trim();
  const projectId = String(request.params?.projectId ?? "").trim();
  const authorizedProjectId = String(
    request.authorizedProject?.id ?? "",
  ).trim();

  if (!userId || !projectId || authorizedProjectId !== projectId) {
    throw createValidationError(
      "Project media validation is not authorized.",
      "PROJECT_MEDIA_VALIDATION_UNAUTHORIZED",
      403,
    );
  }

  const uploadedFile = request.file;

  if (!uploadedFile?.path) {
    throw createValidationError(
      "Please select a Project media file to upload.",
      "PROJECT_MEDIA_FILE_REQUIRED",
    );
  }

  const expectedDirectory = path.resolve(
    getProjectMediaDirectory(userId, projectId),
  );

  const uploadedPath = path.resolve(uploadedFile.path);

  if (path.dirname(uploadedPath) !== expectedDirectory) {
    throw createValidationError(
      "Project media file location is invalid.",
      "PROJECT_MEDIA_INVALID_FILE_LOCATION",
      403,
    );
  }

  return uploadedPath;
}

async function removeRejectedFile(userId, projectId, filePath) {
  const expectedDirectory = path.resolve(
    getProjectMediaDirectory(userId, projectId),
  );

  const targetPath = path.resolve(filePath);

  if (path.dirname(targetPath) !== expectedDirectory) {
    throw new Error(
      "Rejected Project media file is outside its authorized directory.",
    );
  }

  try {
    await assertSafeProjectMediaDirectory(userId, projectId);

    const stats = await fs.lstat(targetPath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw new Error(
        "Rejected Project media target is not a regular file.",
      );
    }

    await fs.unlink(targetPath);
  } catch (error) {
    if (error?.code !== "ENOENT") {
      throw error;
    }
  }
}

export async function validateProjectMedia(request, response, next) {
  let uploadedPath;

  try {
    uploadedPath = getAuthorizedFilePath(request);

    await assertSafeProjectMediaDirectory(
      request.auth.userId,
      request.params.projectId,
    );

    const uploadedFile = request.file;

    // Reject symbolic links and non-regular files before reading content.
    const stats = await fs.lstat(uploadedPath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw createValidationError(
        "The uploaded media is not a regular file.",
        "PROJECT_MEDIA_INVALID_FILE",
      );
    }

    const detectedType = await fileTypeFromFile(uploadedPath);

    if (
      !detectedType ||
      !ALLOWED_MEDIA_TYPES.has(detectedType.mime)
    ) {
      throw createValidationError(
        "The uploaded file does not contain a supported media signature.",
        "PROJECT_MEDIA_CONTENT_TYPE_NOT_ALLOWED",
      );
    }

    if (detectedType.mime !== uploadedFile.mimetype) {
      throw createValidationError(
        "The uploaded media type does not match its contents.",
        "PROJECT_MEDIA_TYPE_MISMATCH",
      );
    }

    const maximumSize = detectedType.mime.startsWith("image/")
      ? MAX_IMAGE_SIZE
      : MAX_VIDEO_SIZE;


    if (stats.size > maximumSize) {
      throw createValidationError(
        "The uploaded media exceeds its permitted size.",
        "PROJECT_MEDIA_TOO_LARGE",
        413,
      );
    }

    request.projectMediaFile = {
      ...uploadedFile,
      verifiedMimeType: detectedType.mime,
      verifiedExtension: detectedType.ext,
      verifiedSize: stats.size,
    };

    return next();
  } catch (error) {
    if (uploadedPath) {
      try {
        await removeRejectedFile(
          request.auth?.userId,
          request.params?.projectId,
          uploadedPath,
        );
      } catch (cleanupError) {
        console.error(
          "Failed to clean up rejected Project media:",
          cleanupError,
        );

        return next(
          createValidationError(
            "Project media cleanup failed.",
            "PROJECT_MEDIA_CLEANUP_FAILED",
            500,
          ),
        );
      }
    }

    return next(error);
  }
}

function checkUrl(errors, field, value, maximum = 2000) {
  if (value === undefined) return;

import crypto from "node:crypto";
import fs from "node:fs/promises";
import path from "node:path";

import { getEnvironment } from "../../config/environment.js";

function requireIdentifier(value, label) {
  const normalized = String(value ?? "").trim();

  if (!normalized) {
    throw new Error(`${label} is required for Project media storage.`);
  }

  return normalized;
}

function hashIdentifier(value) {
  return crypto.createHash("sha256").update(value).digest("hex");
}

function validateFileName(fileName) {
  const normalized = String(fileName ?? "").trim();

  if (
    !normalized ||
    normalized === "." ||
    normalized === ".." ||
    normalized.includes("/") ||
    normalized.includes("\\") ||
    path.basename(normalized) !== normalized ||
    !/^[a-f0-9-]{36}\.(?:jpg|jpeg|png|webp|avif|gif|mp4|webm)$/i.test(
      normalized,
    )
  ) {
    throw new Error("Invalid Project media filename.");
  }

  return normalized;
}

export function getProjectMediaStorageRoot() {
  const environment = getEnvironment();

  const configuredPath = String(
    environment.PROJECT_MEDIA_STORAGE_PATH || "",
  ).trim();

  if (!configuredPath) {
    throw new Error("PROJECT_MEDIA_STORAGE_PATH is not configured.");
  }

  return path.resolve(configuredPath);
}

export function getProjectMediaUserDirectory(userId) {
  return hashIdentifier(requireIdentifier(userId, "User ID"));
}

export function getProjectMediaProjectDirectory(projectId) {
  return hashIdentifier(requireIdentifier(projectId, "Project ID"));
}

export function getProjectMediaDirectory(userId, projectId) {
  return path.join(
    getProjectMediaStorageRoot(),
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
  );
}

export function createProjectMediaStorageKey(userId, projectId, fileName) {
  const safeFileName = validateFileName(fileName);

  return [
    getProjectMediaUserDirectory(userId),
    getProjectMediaProjectDirectory(projectId),
    safeFileName,
  ].join("/");
}

export function resolveOwnedProjectMediaPath(userId, projectId, storageKey) {
  const normalizedKey = String(storageKey ?? "").trim();

  const expectedUserDirectory = getProjectMediaUserDirectory(userId);
  const expectedProjectDirectory = getProjectMediaProjectDirectory(projectId);

  const parts = normalizedKey.split("/");

  if (
    parts.length !== 3 ||
    parts[0] !== expectedUserDirectory ||
    parts[1] !== expectedProjectDirectory
  ) {
    throw new Error(
      "Project media storage key does not belong to this user and Project.",
    );
  }

  const safeFileName = validateFileName(parts[2]);

  const mediaDirectory = path.resolve(
    getProjectMediaDirectory(userId, projectId),
  );

  const filePath = path.resolve(mediaDirectory, safeFileName);

  if (path.dirname(filePath) !== mediaDirectory) {
    throw new Error("Project media path is outside the expected directory.");
  }

  return filePath;
}

export async function assertSafeProjectMediaDirectory(userId, projectId) {
  const root = getProjectMediaStorageRoot();
  const userDirectory = path.join(root, getProjectMediaUserDirectory(userId));
  const projectDirectory = path.join(
    userDirectory,
    getProjectMediaProjectDirectory(projectId),
  );

  const directories = [root, userDirectory, projectDirectory];

  for (const directory of directories) {
    const stats = await fs.lstat(directory);

    if (stats.isSymbolicLink() || !stats.isDirectory()) {
      throw new Error(
        "Project media storage directory is not a regular directory.",
      );
    }
  }

  const canonicalRoot = await fs.realpath(root);
  const canonicalProjectDirectory = await fs.realpath(projectDirectory);

  const relative = path.relative(
    canonicalRoot,
    canonicalProjectDirectory,
  );

  if (
    relative === "" ||
    relative === ".." ||
    relative.startsWith(`..${path.sep}`) ||
    path.isAbsolute(relative)
  ) {
    throw new Error("Project media directory is outside the storage root.");
  }
}

export async function deleteOwnedProjectMedia(userId, projectId, storageKey) {
  const filePath = resolveOwnedProjectMediaPath(userId, projectId, storageKey);

  try {
    await assertSafeProjectMediaDirectory(userId, projectId);

    const stats = await fs.lstat(filePath);

    if (stats.isSymbolicLink() || !stats.isFile()) {
      throw new Error("Project media target is not a regular file.");
    }

    await fs.unlink(filePath);

    return {
      deleted: true,
      storageKey,
    };
  } catch (error) {
    if (error?.code === "ENOENT") {
      return {
        deleted: false,
        storageKey,
      };
    }

    throw error;
  }
}
