 102: function sortTrainingLibrary(trainingRecords = []) {
 103:   return [...trainingRecords].sort((firstTraining, secondTraining) => {
 104:     const firstCurrent =
 105:       firstTraining?.dates?.isCurrent === true ||
 106:       firstTraining?.completion?.status === "in-progress";
 107: 
 108:     const secondCurrent =
 109:       secondTraining?.dates?.isCurrent === true ||
 110:       secondTraining?.completion?.status === "in-progress";
 111: 
 112:     if (firstCurrent !== secondCurrent) {
 113:       return firstCurrent ? -1 : 1;
 114:     }
 115: 
 116:     const dateDifference =
 117:       getTrainingDateValue(secondTraining) -
 118:       getTrainingDateValue(firstTraining);
 119: 
 120:     if (dateDifference !== 0) {
 121:       return dateDifference;
 122:     }
 123: 
 124:     return String(firstTraining?.title || "").localeCompare(
 125:       String(secondTraining?.title || ""),
 126:       undefined,
 127:       {
 128:         sensitivity: "base",
 129:       },
 130:     );
 131:   });
 132: }
