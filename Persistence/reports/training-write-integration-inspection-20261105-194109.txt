  342: function commitCollections({
  343:   originalTrainingRecords,
  344:   originalCertificationRecords,
  345:   nextTrainingRecords,
  346:   nextCertificationRecords,
  347: }) {
  348:   let trainingWriteCompleted = false;
  349:   let certificationWriteCompleted = false;
  350: 
  351:   try {
  352:     const storedTrainingRecords = replaceStoredTraining(nextTrainingRecords);
  353: 
  354:     trainingWriteCompleted = true;
  355: 
  356:     const storedCertificationRecords = replaceStoredCertifications(
  357:       nextCertificationRecords,
  358:     );
  359: 
  360:     certificationWriteCompleted = true;
  361: 
  362:     return {
  363:       trainingRecords: storedTrainingRecords,
  364:       certificationRecords: storedCertificationRecords,
  365:     };
  366:   } catch (cause) {
  367:     const rollbackErrors = [];
  368: 
  369:     if (trainingWriteCompleted || certificationWriteCompleted) {
  370:       try {
  371:         replaceStoredTraining(originalTrainingRecords);
  372:       } catch (rollbackError) {
  373:         rollbackErrors.push({
  374:           collection: "training",
  375:           error: rollbackError,
  376:         });
  377:       }
  378: 
  379:       try {
  380:         replaceStoredCertifications(originalCertificationRecords);
  381:       } catch (rollbackError) {
  382:         rollbackErrors.push({
  383:           collection: "certifications",
  384:           error: rollbackError,
  385:         });
  386:       }
  387:     }
  388: 
  389:     throw createRelationshipError(
  390:       rollbackErrors.length > 0
  391:         ? "TRAINING_CERTIFICATION_ROLLBACK_FAILED"
  392:         : "TRAINING_CERTIFICATION_TRANSACTION_FAILED",
  393: 
  394:       "Training and Certification relationship synchronization failed.",
  395: 
  396:       rollbackErrors.length > 0
  397:         ? "The relationship update failed and could not be completely restored. Reload both libraries before making more changes."
  398:         : "The Training and Certification relationship could not be saved.",
  399: 
  400:       {
  401:         rollbackErrors,
  402:         trainingWriteCompleted,
  403:         certificationWriteCompleted,
  404:       },
  405: 
  406:       cause,
  407:     );
  408:   }
  409: }

  417: export function saveTrainingWithCertificationSync({
  418:   trainingRecords,
  419:   changedTrainingIds = [],
  420:   deletedTrainingIds = [],
  421: } = {}) {
  422:   const originalTrainingRecords = readStoredTraining();
  423: 
  424:   const originalCertificationRecords = readStoredCertifications();
  425: 
  426:   let nextTrainingRecords = normalizeTrainingCollection(trainingRecords);
  427: 
  428:   let nextCertificationRecords = normalizeCertificationCollection(
  429:     originalCertificationRecords,
  430:   );
  431: 
  432:   const changedIds = createIdSet(changedTrainingIds);
  433: 
  434:   const deletedIds = createIdSet(deletedTrainingIds);
  435: 
  436:   const trainingMap = createRecordMap(nextTrainingRecords);
  437: 
  438:   const certificationMap = createRecordMap(nextCertificationRecords);
  439: 
  440:   /*
  441:    * Remove deleted Training records from every
  442:    * Certification relationship collection.
  443:    */
  444: 
  445:   if (deletedIds.size > 0) {
  446:     nextCertificationRecords = nextCertificationRecords.map((certification) => {
  447:       const currentRelationships = getArray(
  448:         certification?.relatedRecords?.trainingRelationships,
  449:       );
  450: 
  451:       const remainingRelationships = currentRelationships.filter(
  452:         (relationship) => !deletedIds.has(getText(relationship?.trainingId)),
  453:       );
  454: 
  455:       if (remainingRelationships.length === currentRelationships.length) {
  456:         return certification;
  457:       }

  459:       return normalizeCertification({
  460:         ...certification,
  461: 
  462:         relatedRecords: {
  463:           ...certification.relatedRecords,
  464: 
  465:           trainingRelationships: remainingRelationships,
  466:         },
  467: 
  468:         updatedAt: new Date().toISOString(),
  469:       });
  470:     });
  471:   }
  472: 
  473:   /*
  474:    * For a Training save, Training is authoritative:
  475:    * its selected Certification IDs determine which
  476:    * Certification records should point back to it.
  477:    */
  478: 
  479:   changedIds.forEach((trainingId) => {
  480:     const training = requireTraining(trainingMap, trainingId);
  481: 
  482:     const selectedCertificationIds = new Set(
  483:       getTrainingCertificationIds(training),
  484:     );
  485: 
  486:     selectedCertificationIds.forEach((certificationId) => {
  487:       requireCertification(certificationMap, certificationId);
  488:     });
  489: 
  490:     nextCertificationRecords = nextCertificationRecords.map((certification) =>
  491:       synchronizeCertificationWithTraining(
  492:         certification,
  493:         training,
  494:         selectedCertificationIds.has(certification.id),
  495:       ),
  496:     );
  497:   });
  498: 
  499:   /*
  500:    * Refresh both snapshot directions after the
  501:    * relationship membership is synchronized.
  502:    */
  503: 
  504:   const refreshedCertificationMap = createRecordMap(nextCertificationRecords);
  505: 
  506:   nextTrainingRecords = nextTrainingRecords.map((training) =>
  507:     changedIds.has(training.id)
  508:       ? refreshTrainingCertificationRelationships(
  509:           training,
  510:           refreshedCertificationMap,
  511:         )
  512:       : training,
  513:   );
  514: 
  515:   return commitCollections({
  516:     originalTrainingRecords,
  517:     originalCertificationRecords,
  518:     nextTrainingRecords,
  519:     nextCertificationRecords,
  520:   });
  521: }

  431: function normalizeTrainingCertificationRelationships(values) {
  432:   if (!Array.isArray(values)) {
  433:     return [];
  434:   }
  435: 
  436:   const usedCertificationIds = new Set();
  437: 
  438:   return values.reduce((relationships, value) => {
  439:     const relationship = createTrainingCertificationRelationship(
  440:       value,
  441:       relationships.length,
  442:     );
  443: 
  444:     if (!relationship.certificationId) {
  445:       return relationships;
  446:     }
  447: 
  448:     const comparisonId = relationship.certificationId.toLocaleLowerCase();
  449: 
  450:     if (usedCertificationIds.has(comparisonId)) {
  451:       return relationships;
  452:     }
  453: 
  454:     if (
  455:       relationships.length >= TRAINING_FIELD_LIMITS.maximumRelatedCertifications
  456:     ) {
  457:       return relationships;
  458:     }
  459: 
  460:     usedCertificationIds.add(comparisonId);
  461: 
  462:     relationships.push({
  463:       ...relationship,
  464:       order: relationships.length,
  465:     });
  466: 
  467:     return relationships;
  468:   }, []);
  469: }
